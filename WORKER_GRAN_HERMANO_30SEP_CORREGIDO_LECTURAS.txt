function corsHeaders(){
  return {
    "Access-Control-Allow-Origin":"*",

function driveImageDirectUrl_(value){
  const raw=String(value ?? "").trim();
  if(!raw) return raw;

  // Ya es una URL directa compatible.
  if(/googleusercontent\.com\/d\//i.test(raw)){
    return raw;
  }

  let u=null;
  try{
    u=new URL(raw);
  }catch(_){
    return raw;
  }

  const host=String(u.hostname||"").toLowerCase();
  if(
    host!=="drive.google.com" &&
    !host.endsWith(".drive.google.com")
  ){
    return raw;
  }

  let fileId="";

  // https://drive.google.com/file/d/FILE_ID/view
  let m=u.pathname.match(/\/file\/d\/([^/]+)/i);
  if(m && m[1]) fileId=m[1];

  // Variantes /d/FILE_ID/
  if(!fileId){
    m=u.pathname.match(/\/d\/([^/]+)/i);
    if(m && m[1]) fileId=m[1];
  }

  // https://drive.google.com/open?id=FILE_ID
  // https://drive.google.com/uc?id=FILE_ID
  // https://drive.google.com/thumbnail?id=FILE_ID
  if(!fileId){
    fileId=String(u.searchParams.get("id")||"").trim();
  }

  if(!fileId) return raw;

  return `https://lh3.googleusercontent.com/d/${encodeURIComponent(fileId)}`;
}

function isImageLikeField_(key,parentKey=""){
  const k=String(key||"").toLowerCase();
  const p=String(parentKey||"").toLowerCase();

  if(
    /^img\d*$/.test(k) ||
    /^imagen\d*$/.test(k) ||
    /^image\d*$/.test(k) ||
    /^foto\d*$/.test(k) ||
    k==="logo" ||
    k.startsWith("logo_") ||
    k==="poster" ||
    k==="thumbnail" ||
    k==="miniatura" ||
    k==="portada" ||
    k==="banner" ||
    k==="imagen_url" ||
    k==="image_url" ||
    k==="url_imagen" ||
    k==="foto_url" ||
    k==="media_url"
  ){
    return true;
  }

  // Las colecciones de media guardan la imagen/video en una clave genérica "url".
  if(
    k==="url" &&
    [
      "media",
      "media_items",
      "imagenes",
      "images",
      "fotos",
      "galeria",
      "gallery"
    ].includes(p)
  ){
    return true;
  }

  return false;
}

function normalizeDriveImageFieldsDeep_(value,parentKey=""){
  if(Array.isArray(value)){
    return value.map(v=>normalizeDriveImageFieldsDeep_(v,parentKey));
  }

  if(!value || typeof value!=="object"){
    return value;
  }

  const out={};

  for(const [key,val] of Object.entries(value)){
    if(
      typeof val==="string" &&
      isImageLikeField_(key,parentKey)
    ){
      out[key]=driveImageDirectUrl_(val);
      continue;
    }

    out[key]=normalizeDriveImageFieldsDeep_(val,key);
  }

  return out;
}

function jsonResponse(obj, status=200){

  return new Response(JSON.stringify(normalized), {
    status,
    headers:{
      "Content-Type":"application/json",
      ...corsHeaders()
    }
  });
}

function base64UrlEncode(input){
  const bytes = typeof input === "string"
    ? new TextEncoder().encode(input)
    : input;

  let binary = "";
  bytes.forEach(b => binary += String.fromCharCode(b));

  return btoa(binary)
    .replace(/\+/g, "-")
    .replace(/\//g, "_")
    .replace(/=+$/g, "");
}

function pemToArrayBuffer(pem){
  const clean = pem
    .replace(/\\n/g, "\n")
    .replace("-----BEGIN PRIVATE KEY-----", "")
    .replace("-----END PRIVATE KEY-----", "")
    .replace(/\s+/g, "");

  const binary = atob(clean);
  const bytes = new Uint8Array(binary.length);

  for(let i=0;i<binary.length;i++) bytes[i] = binary.charCodeAt(i);
  return bytes.buffer;
}

const GOOGLE_TOKEN_CACHE = {
  token:"",
  expiresAt:0,
  pending:null
};

async function createGoogleAccessToken_(env){
  const projectId = String(env.FIREBASE_PROJECT_ID || "").trim();
  const clientEmail = String(env.FIREBASE_CLIENT_EMAIL || "").trim();
  const privateKey = String(env.FIREBASE_PRIVATE_KEY || "").trim();

  if(!projectId) throw new Error("Falta FIREBASE_PROJECT_ID");
  if(!clientEmail) throw new Error("Falta FIREBASE_CLIENT_EMAIL");
  if(!privateKey) throw new Error("Falta FIREBASE_PRIVATE_KEY");

  const now = Math.floor(Date.now() / 1000);

  const header = {
    alg:"RS256",
    typ:"JWT"
  };

  const payload = {
    iss:clientEmail,
    scope:"https://www.googleapis.com/auth/datastore https://www.googleapis.com/auth/cloud-platform",
    aud:"https://oauth2.googleapis.com/token",
    iat:now,
    exp:now + 3600
  };

  const encodedHeader = base64UrlEncode(JSON.stringify(header));
  const encodedPayload = base64UrlEncode(JSON.stringify(payload));
  const unsignedToken = `${encodedHeader}.${encodedPayload}`;

  const key = await crypto.subtle.importKey(
    "pkcs8",
    pemToArrayBuffer(privateKey),
    {
      name:"RSASSA-PKCS1-v1_5",
      hash:"SHA-256"
    },
    false,
    ["sign"]
  );

  const signature = await crypto.subtle.sign(
    "RSASSA-PKCS1-v1_5",
    key,
    new TextEncoder().encode(unsignedToken)
  );

  const jwt = `${unsignedToken}.${base64UrlEncode(new Uint8Array(signature))}`;

  const tokenResp = await fetch("https://oauth2.googleapis.com/token", {
    method:"POST",
    headers:{
      "Content-Type":"application/x-www-form-urlencoded"
    },
    body:new URLSearchParams({
      grant_type:"urn:ietf:params:oauth:grant-type:jwt-bearer",
      assertion:jwt
    })
  });

  const tokenText = await tokenResp.text();
  let tokenJson = {};
  try{ tokenJson = JSON.parse(tokenText); }catch(_){}

  if(!tokenResp.ok || !tokenJson.access_token){
    throw new Error("No se pudo obtener token Google: " + tokenText.slice(0,500));
  }

  const expiresIn = Number(tokenJson.expires_in || 3600);

  GOOGLE_TOKEN_CACHE.token = tokenJson.access_token;
  GOOGLE_TOKEN_CACHE.expiresAt = Date.now() + Math.max(60, expiresIn - 120) * 1000;

  return GOOGLE_TOKEN_CACHE.token;
}

async function getGoogleAccessToken(env){
  if(
    GOOGLE_TOKEN_CACHE.token &&
    Date.now() < GOOGLE_TOKEN_CACHE.expiresAt
  ){
    return GOOGLE_TOKEN_CACHE.token;
  }

  if(GOOGLE_TOKEN_CACHE.pending){
    return await GOOGLE_TOKEN_CACHE.pending;
  }

  GOOGLE_TOKEN_CACHE.pending = createGoogleAccessToken_(env);

  try{
    return await GOOGLE_TOKEN_CACHE.pending;
  }finally{
    GOOGLE_TOKEN_CACHE.pending = null;
  }
}

function firestoreBaseUrl(env){
  const projectId = String(env.FIREBASE_PROJECT_ID || "").trim();
  if(!projectId) throw new Error("Falta FIREBASE_PROJECT_ID");

  return `https://firestore.googleapis.com/v1/projects/${encodeURIComponent(projectId)}/databases/(default)/documents`;
}

function fsValueToJs(v){
  if(v === null || v === undefined) return null;

  if("stringValue" in v) return v.stringValue;
  if("booleanValue" in v) return v.booleanValue;
  if("integerValue" in v) return Number(v.integerValue);
  if("doubleValue" in v) return Number(v.doubleValue);
  if("timestampValue" in v) return v.timestampValue;
  if("nullValue" in v) return null;

  if("arrayValue" in v){
    const values = v.arrayValue && Array.isArray(v.arrayValue.values)
      ? v.arrayValue.values
      : [];
    return values.map(fsValueToJs);
  }

  if("mapValue" in v){
    return firestoreFieldsToJs(
      v.mapValue && v.mapValue.fields ? v.mapValue.fields : {}
    );
  }

  return null;
}

function firestoreFieldsToJs(fields){
  const out = {};
  for(const [k,v] of Object.entries(fields || {})){
    out[k] = fsValueToJs(v);
  }
  return out;
}

function jsToFsValue(v){
  if(v === null || v === undefined){
    return { nullValue:null };
  }

  if(Array.isArray(v)){
    return {
      arrayValue:{
        values:v.map(jsToFsValue)
      }
    };
  }

  if(typeof v === "boolean"){
    return { booleanValue:v };
  }

  if(typeof v === "number"){
    if(Number.isInteger(v)) return { integerValue:String(v) };
    return { doubleValue:v };
  }

  if(typeof v === "object"){
    const fields = {};
    for(const [k,val] of Object.entries(v)){
      if(val !== undefined) fields[k] = jsToFsValue(val);
    }
    return { mapValue:{ fields } };
  }

  return { stringValue:String(v) };
}

function jsObjectToFirestoreFields(obj){
  const fields = {};
  for(const [k,v] of Object.entries(obj || {})){
    if(v !== undefined){
      fields[k] = jsToFsValue(v);
    }
  }
  return fields;
}

function docNameToId(name){
  return String(name || "").split("/").pop() || "";
}

async function firestoreRequest(env, path, {
  method="GET",
  body=null,
  allow404=false
} = {}){
  const token = await getGoogleAccessToken(env);

  const resp = await fetch(
    `${firestoreBaseUrl(env)}/${path}`,
    {
      method,
      headers:{
        "Authorization":`Bearer ${token}`,
        ...(body !== null ? {"Content-Type":"application/json"} : {})
      },
      ...(body !== null ? {body:JSON.stringify(body)} : {})
    }
  );

  if(allow404 && resp.status === 404){
    return null;
  }

  const text = await resp.text();
  let json = null;

  if(text){
    try{ json = JSON.parse(text); }catch(_){}
  }

  if(!resp.ok){
    throw new Error(
      `Firestore ${method} ${path}: ${resp.status} ${text.slice(0,500)}`
    );
  }

  return json;
}

function firestoreDocumentToJs(doc, forcedId=""){
  if(!doc) return null;

  /** @type {any} */
  const out = {
    id:forcedId || docNameToId(doc.name),
    ...firestoreFieldsToJs(doc.fields || {})
  };

  return out;
}

async function firestoreGetDocument(env, collection, id){
  const doc = await firestoreRequest(
    env,
    `${encodeURIComponent(collection)}/${encodeURIComponent(id)}`,
    {
      method:"GET",
      allow404:true
    }
  );

  return firestoreDocumentToJs(doc, id);
}

async function firestoreListCollection(env, collection, pageSize = 1000){
  let pageToken = "";
  const out = [];

  do{
    const qs = new URLSearchParams({
      pageSize:String(pageSize)
    });

    if(pageToken){
      qs.set("pageToken", pageToken);
    }

    const json = await firestoreRequest(
      env,
      `${encodeURIComponent(collection)}?${qs.toString()}`,
      { method:"GET" }
    ) || {};

    for(const doc of (json.documents || [])){
      out.push(firestoreDocumentToJs(doc));
    }

    pageToken = String(json.nextPageToken || "");
  }while(pageToken);

  return out;
}

async function firestoreQueryEqual(env, collection, field, value, limit=1000){
  const token = await getGoogleAccessToken(env);
  const projectId = String(env.FIREBASE_PROJECT_ID || "").trim();
  if(!projectId) throw new Error("Falta FIREBASE_PROJECT_ID");

  const url =
    `https://firestore.googleapis.com/v1/projects/${encodeURIComponent(projectId)}` +
    `/databases/(default)/documents:runQuery`;

  const resp = await fetch(url, {
    method:"POST",
    headers:{
      "Authorization":`Bearer ${token}`,
      "Content-Type":"application/json"
    },
    body:JSON.stringify({
      structuredQuery:{
        from:[{collectionId:collection}],
        where:{
          fieldFilter:{
            field:{fieldPath:field},
            op:"EQUAL",
            value:jsToFsValue(value)
          }
        },
        limit:Number(limit || 1000)
      }
    })
  });

  const raw = await resp.text();
  let json = [];
  try{ json = raw ? JSON.parse(raw) : []; }catch(_){}

  if(!resp.ok){
    throw new Error(
      `Firestore QUERY ${collection}.${field}: ${resp.status} ${raw.slice(0,500)}`
    );
  }

  const out=[];
  for(const row of (Array.isArray(json) ? json : [])){
    if(row && row.document){
      out.push(firestoreDocumentToJs(row.document));
    }
  }
  return out;
}

async function firestoreGetMany_(env, collection, ids){
  const unique = [...new Set(
    (Array.isArray(ids) ? ids : [])
      .map(v=>String(v||"").trim())
      .filter(Boolean)
  )];

  if(!unique.length) return [];

  const out=[];
  const chunkSize=20;

  for(let i=0;i<unique.length;i+=chunkSize){
    const chunk=unique.slice(i,i+chunkSize);
    const rows=await Promise.all(
      chunk.map(id=>firestoreGetDocument(env,collection,id))
    );
    rows.filter(Boolean).forEach(r=>out.push(r));
  }

  return out;
}

function gldCacheRequest_(key){
  return new Request(
    "https://cache.guialocal.internal/" + encodeURIComponent(String(key||"")),
    {method:"GET"}
  );
}

async function gldCacheGetJson_(key){
  try{
    if(typeof caches === "undefined" || !caches.default) return null;
    const hit = await caches.default.match(gldCacheRequest_(key));
    if(!hit) return null;
    return await hit.json();
  }catch(_){
    return null;
  }
}

async function gldCachePutJson_(key, value, ttlSeconds=3600){
  try{
    if(typeof caches === "undefined" || !caches.default) return;
    const response = new Response(JSON.stringify(value), {
      headers:{
        "Content-Type":"application/json",
        "Cache-Control":`public, max-age=${Math.max(60,Number(ttlSeconds||3600))}`
      }
    });
    await caches.default.put(gldCacheRequest_(key), response);
  }catch(_){}
}

async function gldCacheDelete_(key){
  try{
    if(typeof caches === "undefined" || !caches.default) return;
    await caches.default.delete(gldCacheRequest_(key));
  }catch(_){}
}

function guideCacheKey_(paisId, provinciaId, ciudadId){
  return [
    "commerce-guia-v2",
    String(paisId||"").trim() || "*",
    String(provinciaId||"").trim() || "*",
    String(ciudadId||"").trim() || "*"
  ].join(":");
}

async function refreshGuideCacheForAdvertiser_(
  env,
  advertiserId,
  extraCityIds=[]
){
  const aid=String(advertiserId||"").trim();
  if(!aid) return;

  try{
    // IMPORTANTE:
    // Una modificación puntual NO invalida ni reconstruye toda la ciudad.
    // Sólo actualiza las ciudades donde está este anunciante.
    const sedes = await firestoreQueryEqual(
      env, COLL_SEDES, "anunciante_id", aid
    );

    const cityIds=new Set(
      [
        ...sedes.map(s=>String(s && s.ciudad_id || "").trim()),
        ...((Array.isArray(extraCityIds) ? extraCityIds : [])
          .map(v=>String(v||"").trim()))
      ].filter(Boolean)
    );

    const advertiser=await firestoreGetDocument(
      env,COLL_ANUNCIANTES,aid
    );
    const admin=await firestoreGetDocument(
      env,COLL_ADMIN,aid
    );
    const catalogs=await loadPublicGuideCatalogsLight_(env);

    for(const ciudadId of cityIds){
      const ciudadDoc=await firestoreGetDocument(
        env,COLL_CIUDADES,ciudadId
      );
      const provinciaId=String(
        (ciudadDoc && ciudadDoc.provincia_id) || ""
      ).trim();
      const paisId=String(
        (ciudadDoc && ciudadDoc.pais_id) || ""
      ).trim();

      const cacheKey=guideCacheKey_(paisId,provinciaId,ciudadId);
      const cached=await gldCacheGetJson_(cacheKey);

      // Si la ciudad no está en caché, no la reconstruimos.
      // La próxima carga normal podrá construirla una sola vez.
      if(!Array.isArray(cached)) continue;

      const provinciaDoc=provinciaId
        ? await firestoreGetDocument(env,COLL_PROVINCIAS,provinciaId)
        : null;
      const paisDoc=paisId
        ? await firestoreGetDocument(env,COLL_PAISES,paisId)
        : null;

      const citySedes=(sedes||[]).filter(s =>
        String(s && s.ciudad_id || "").trim()===ciudadId &&
        publicSedeActiva_(s)
      );

      const rows=buildPublicGuideRows_({
        advertisers:advertiser ? [advertiser] : [],
        admins:admin ? [admin] : [],
        sedes:citySedes,
        segmentos:catalogs.segmentos||[],
        categorias:catalogs.categorias||[],
        actividades:catalogs.actividades||[],
        acciones:catalogs.acciones||[],
        nodos:catalogs.nodos||[],
        funcionalidades:[],
        niveles_anunciante:[],
        ciudades:ciudadDoc ? [ciudadDoc] : [],
        provincias:provinciaDoc ? [provinciaDoc] : [],
        paises:paisDoc ? [paisDoc] : []
      });

      const next=cached.filter(row =>
        String(row && row.id || "").trim()!==aid
      );

      if(rows.length) next.push(rows[0]);

      await gldCachePutJson_(cacheKey,next,1800);
    }
  }catch(_){}
}

async function patchCachedArrayItem_(key, itemId, item, options={}){
  const cached=await gldCacheGetJson_(key);
  if(!Array.isArray(cached)) return false;

  const idField=String(options.idField || "id");
  const mode=String(options.mode || "upsert");

  const idx=cached.findIndex(x =>
    String(x && x[idField] || "").trim() === String(itemId || "").trim()
  );

  if(mode==="delete"){
    if(idx<0) return true;
    cached.splice(idx,1);
  }else{
    if(idx<0) cached.push(item);
    else cached[idx]=item;
  }

  await gldCachePutJson_(key,cached,Number(options.ttl || 600));
  return true;
}

async function firestoreSetDocument(env, collection, id, data){
  const doc = await firestoreRequest(
    env,
    `${encodeURIComponent(collection)}/${encodeURIComponent(id)}`,
    {
      method:"PATCH",
      body:{
        fields:jsObjectToFirestoreFields(data)
      }
    }
  );

  return firestoreDocumentToJs(doc, id);
}

async function firestoreDeleteDocument(env, collection, id){
  await firestoreRequest(
    env,
    `${encodeURIComponent(collection)}/${encodeURIComponent(id)}`,
    {
      method:"DELETE",
      allow404:true
    }
  );

  return true;
}

function normalizeAction(v){
  return String(v || "")
    .trim()
    .toLowerCase()
    .replace(/\s+/g,"")
    .replace(/-/g,"_");
}

function splitFeatures(value){
  if(Array.isArray(value)){
    return value
      .map(v => String(v || "").trim())
      .filter(Boolean);
  }

  return String(value || "")
    .split(/[;,|]+/)
    .map(s => s.trim())
    .filter(Boolean);
}

function normalizeFeatureName(v){
  return String(v || "")
    .trim()
    .toLowerCase()
    .normalize("NFD")
    .replace(/[\u0300-\u036f]/g,"")
    .replace(/[\s_-]+/g,"");
}

function efemConfigObject_(adminDoc){
  const raw = adminDoc && adminDoc.funcionalidades_config;
  return raw && typeof raw === "object" && !Array.isArray(raw)
    ? raw
    : {};
}




function buildEfemPermisos(adminDoc){
  const funcs = splitFeatures(
    adminDoc && (
      adminDoc.funcionalidades ||
      adminDoc.funciones ||
      adminDoc.permisos ||
      ""
    )
  );

  const normalized = funcs.map(normalizeFeatureName);

  const hasLocal = normalized.some(v =>
    v.includes("efemerideslocal") ||
    v.includes("efemeridelocal") ||
    v === "eflocal"
  );

  const hasGeneral = normalized.some(v =>
    v.includes("efemeridesgrl") ||
    v.includes("efemeridesgeneral") ||
    v.includes("efemeridegeneral") ||
    v === "efgeneral"
  );

  const hasProv = normalized.some(v =>
    v.includes("efemeridesprov") ||
    v.includes("efememeridesprov") ||
    v.includes("efemeridesprovincial") ||
    v === "efprovincial"
  );

  const cfg = efemConfigObject_(adminDoc);

  const localCfg =
    cfg.EFEMERIDES_LOCAL ||
    cfg.efemerides_local ||
    {};

  const provCfg =
    cfg.EFEMERIDES_PROVINCIAL ||
    cfg.efemerides_provincial ||
    {};

  const ciudades = Array.isArray(localCfg.ciudades)
    ? localCfg.ciudades.map(String).filter(Boolean)
    : [];

  const provincias = Array.isArray(provCfg.provincias)
    ? provCfg.provincias.map(String).filter(Boolean)
    : [];

  return {
    efemerides_local:hasLocal,
    efemerides_grl:hasGeneral,
    efemerides_provincial:hasProv,
    ciudades,
    provincias,
    todas_ciudades:Boolean(localCfg.todas_ciudades),
    todas_provincias:Boolean(provCfg.todas_provincias)
  };
}


function canSeeEfemeride(row, permisos){
  const tipo = String(row.tipo || "").trim().toUpperCase();

  if(tipo === "GENERAL"){
    return permisos.efemerides_grl;
  }

  if(tipo === "PROVINCIAL"){
    if(!permisos.efemerides_provincial) return false;
    if(permisos.todas_provincias) return true;

    const provinciaId = String(row.provincia_id || "").trim();
    return permisos.provincias.includes(provinciaId);
  }

  if(tipo === "LOCAL"){
    if(!permisos.efemerides_local) return false;
    if(permisos.todas_ciudades) return true;

    const ciudadId = String(row.ciudad_id || "").trim();
    return permisos.ciudades.includes(ciudadId);
  }

  return false;
}

function addFrontendCompatibilityFields(row){
  const out = { ...row };

  const tipoFecha = String(out.tipo_fecha || "").trim().toUpperCase();

  if(tipoFecha === "FIJA"){
    const mes = Number(out.mes || 0);
    const dia = Number(out.dia || 0);

    if(mes && dia){
      out.fecha = `2000-${String(mes).padStart(2,"0")}-${String(dia).padStart(2,"0")}`;
    }
  }

  return out;
}

function sanitizeEfemPayload(payload){
  const p = payload && typeof payload === "object" ? payload : {};

  const out = {
    efemeride_id:String(p.efemeride_id || p.id || "").trim(),
    tipo:String(p.tipo || "").trim().toUpperCase(),
    provincia_id:String(p.provincia_id || "").trim(),
    ciudad_id:String(p.ciudad_id || "").trim(),
    tipo_fecha:String(p.tipo_fecha || "").trim().toUpperCase(),
    nombre:String(p.nombre || "").trim(),
    descripcion:String(p.descripcion || ""),
    imagen:String(p.imagen || ""),
    activo:p.activo === undefined ? true : Boolean(p.activo),
    origen:String(p.origen || "WORKER_FIRESTORE")
  };

  if(out.tipo_fecha === "FIJA"){
    let mes = Number(p.mes || 0);
    let dia = Number(p.dia || 0);

    if((!mes || !dia) && p.fecha){
      const fecha = String(p.fecha).trim();

      // Formato actual del login: DD/MM
      let m = fecha.match(/^(\d{1,2})\/(\d{1,2})(?:\/\d{2,4})?$/);
      if(m){
        dia = Number(m[1]);
        mes = Number(m[2]);
      }

      // Compatibilidad con DD-MM
      if((!mes || !dia)){
        m = fecha.match(/^(\d{1,2})-(\d{1,2})(?:-\d{2,4})?$/);
        if(m){
          dia = Number(m[1]);
          mes = Number(m[2]);
        }
      }

      // Compatibilidad con ISO YYYY-MM-DD
      if((!mes || !dia)){
        m = fecha.match(/^\d{4}-(\d{1,2})-(\d{1,2})$/);
        if(m){
          mes = Number(m[1]);
          dia = Number(m[2]);
        }
      }
    }

    out.mes = mes;
    out.dia = dia;
    out.semana_mes = "";
    out.dia_semana = "";
  }
  else if(out.tipo_fecha === "MOVIL"){
    out.mes = Number(p.mes || 0);
    out.semana_mes = String(p.semana_mes ?? "").trim().toUpperCase();
    out.dia_semana = Number(p.dia_semana || 0);
    out.dia = "";
  }

  return out;
}

function validateEfemPayload(data){
  if(!data.tipo) return "Falta tipo";
  if(!data.tipo_fecha) return "Falta tipo_fecha";
  if(!data.nombre) return "Falta nombre";

  if(data.tipo === "LOCAL" && !data.ciudad_id){
    return "Falta ciudad_id";
  }

  if(data.tipo === "PROVINCIAL" && !data.provincia_id){
    return "Falta provincia_id";
  }

  if(data.tipo_fecha === "FIJA"){
    if(!data.mes || !data.dia) return "Faltan mes/dia";
  }

  if(data.tipo_fecha === "MOVIL"){
    if(!data.mes || !data.semana_mes || !data.dia_semana){
      return "Faltan datos de fecha móvil";
    }
  }

  return "";
}

function generateEfemId(){
  return `ef_${Date.now()}_${crypto.randomUUID().replace(/-/g,"").slice(0,10)}`;
}

async function loadEfemContext(env, advertiserId){

  }

  const adminDoc = await firestoreGetDocument(

  };
}


function efemPublicDate_(raw){
  const s = String(raw || "").trim();
  const m = s.match(/^(\d{4})-(\d{2})-(\d{2})$/);
  if(m){
    return new Date(Number(m[1]), Number(m[2])-1, Number(m[3]), 12, 0, 0);
  }
  const now = new Date(Date.now() - 3 * 60 * 60 * 1000);
  return new Date(now.getUTCFullYear(), now.getUTCMonth(), now.getUTCDate(), 12, 0, 0);
}

function efemPublicDayNumber_(value){
  const s = String(value ?? "").trim().toLowerCase()
    .normalize("NFD").replace(/[\u0300-\u036f]/g,"");
  const map = {
    "0":0,"domingo":0,
    "1":1,"lunes":1,
    "2":2,"martes":2,
    "3":3,"miercoles":3,
    "4":4,"jueves":4,
    "5":5,"viernes":5,
    "6":6,"sabado":6
  };
  return Object.prototype.hasOwnProperty.call(map,s) ? map[s] : -1;
}

function efemMatchesPublicDate_(row, date){
  const tipoFecha = String(row.tipo_fecha || "").trim().toUpperCase()
    .normalize("NFD").replace(/[\u0300-\u036f]/g,"");

  if(tipoFecha === "FIJA"){
    return Number(row.mes || 0) === date.getMonth()+1 &&
           Number(row.dia || 0) === date.getDate();
  }

  if(tipoFecha === "MOVIL"){
    const mes = Number(row.mes || 0);
    if(mes !== date.getMonth()+1) return false;

    const diaSemana = efemPublicDayNumber_(row.dia_semana);
    if(diaSemana < 0 || diaSemana !== date.getDay()) return false;

    const semanaRaw = String(row.semana_mes ?? "").trim().toUpperCase()
      .normalize("NFD").replace(/[\u0300-\u036f]/g,"");

    if(semanaRaw === "ULTIMA"){
      const nextSameDay = new Date(date);
      nextSameDay.setDate(nextSameDay.getDate()+7);
      return nextSameDay.getMonth() !== date.getMonth();
    }

    const semana = Number(semanaRaw || 0);
    if(!Number.isInteger(semana) || semana < 1 || semana > 5) return false;
    return Math.floor((date.getDate()-1)/7)+1 === semana;
  }

  return false;
}

async function efemeridesPublicasFirestore_(env, ciudadId, fechaRaw){
  const ciudad = String(ciudadId || "").trim();
  const fecha = efemPublicDate_(fechaRaw);
  const iso = [
    fecha.getFullYear(),
    String(fecha.getMonth()+1).padStart(2,"0"),
    String(fecha.getDate()).padStart(2,"0")
  ].join("-");

  if(!ciudad){
    return {success:false, message:"Falta ciudad_id", ciudad_id:"", fecha:iso, efemerides:[]};
  }

  const [rows, ciudadDoc] = await Promise.all([
    firestoreListCollection(env, COLL_EFEMERIDES),
    firestoreGetDocument(env, COLL_CIUDADES, ciudad)
  ]);

  const provinciaId = String(
    ciudadDoc && (ciudadDoc.provincia_id || ciudadDoc.provinciaId) || ""
  ).trim();

  const visibles = rows
    .filter(r => truthyValue_(r.activo))
    .filter(r => {
      const tipo = String(r.tipo || "").trim().toUpperCase();
      if(tipo === "GENERAL") return true;
      if(tipo === "LOCAL") return String(r.ciudad_id || "").trim() === ciudad;
      if(tipo === "PROVINCIAL"){
        return !!provinciaId && String(r.provincia_id || "").trim() === provinciaId;
      }
      return false;
    })
    .filter(r => efemMatchesPublicDate_(r, fecha))
    .map(addFrontendCompatibilityFields)
    .sort((a,b) => {
      const orden = {LOCAL:1, PROVINCIAL:2, GENERAL:3};
      const d = (orden[String(a.tipo||"").toUpperCase()] || 9) -
                (orden[String(b.tipo||"").toUpperCase()] || 9);
      if(d) return d;
      return String(a.nombre || "").localeCompare(String(b.nombre || ""),"es");
    });

  return {
    success:true,
    ciudad_id:ciudad,
    fecha:iso,
    efemerides:visibles
  };
}

async function handleEfemeridesRequest(request, env){
  try{
    /** @type {any} */
    let body = {};
    const url = new URL(request.url);

    if(request.method === "POST"){
      try{
        body = await request.json();
      }catch(_){
        return jsonResponse({ success:false, message:"Payload inválido" }, 400);
      }
    }
    else{
      url.searchParams.forEach((v,k)=>{
        body[k] = v;
      });
    }

    const action = normalizeAction(body.action || body.accion || "");
    if(
      request.method === "GET" &&
      (action === "efemerides" || action === "publicas" || action === "public")
    ){
      const result = await efemeridesPublicasFirestore_(
        env,
        body.ciudad_id || url.searchParams.get("ciudad_id") || "",
        body.fecha || url.searchParams.get("fecha") || ""
      );
      return jsonResponse(result, result.success ? 200 : 400);
    }

    const advertiserId = String(
      body.advertiserId ||
      body.advertiserID ||
      body.__id ||
      url.searchParams.get("advertiserId") ||
      getAdvertiserIdFromHeaders(request.headers) ||
      getAdvertiserIdFromCookies(request.headers.get("cookie") || "") ||
      ""
    ).trim();

    if(!advertiserId){
      return jsonResponse({
        success:false,
        message:"Falta advertiserId"
      }, 400);
    }

    const ctx = await loadEfemContext(env, advertiserId);

    if(
      action === "getpaneldata" ||
      action === "paneldata" ||
      action === "bootstrap" ||
      action === "listar" ||
      action === "listar_admin" ||
      action === "list"
    ){
      const [efemerides, ciudades, provincias] = await Promise.all([
        firestoreListCollection(env, COLL_EFEMERIDES),
        firestoreListCollection(env, COLL_CIUDADES),
        firestoreListCollection(env, COLL_PROVINCIAS)
      ]);

      const visibles = efemerides
        .filter(r => canSeeEfemeride(r, ctx.permisos))
        .map(addFrontendCompatibilityFields);

      const ciudadesPermitidas = ctx.permisos.todas_ciudades
        ? ciudades
        : ciudades.filter(c => ctx.permisos.ciudades.includes(String(c.ciudad_id || c.id || "")));

      const provinciasPermitidas = ctx.permisos.todas_provincias
        ? provincias
        : provincias.filter(p => ctx.permisos.provincias.includes(String(p.provincia_id || p.id || "")));

      return jsonResponse({
        success:true,
        permisos:ctx.permisos,
        permisos_source:ctx.permisos_source,
        permisos_migrados:ctx.permisos_migrados,
        ciudades:ciudadesPermitidas,
        provincias:provinciasPermitidas,
        efemerides:visibles
      }, 200);
    }

    if(action === "guardar" || action === "save"){
      const payload = body.payload && typeof body.payload === "object"
        ? body.payload
        : body;

      // El login, cuando EDITA, envía solamente los campos que cambiaron.
      // Por eso primero recuperamos el registro existente y luego mezclamos
      // el parche recibido con los datos completos guardados en Firestore.
      const requestedId = String(
        payload.efemeride_id ||
        payload.id ||
        ""
      ).trim();

      let existing = null;

      if(requestedId){
        existing = await firestoreGetDocument(
          env,
          COLL_EFEMERIDES,
          requestedId
        );

        if(!existing){
          return jsonResponse({
            success:false,
            message:"Efeméride no encontrada",
            error:"efemeride_id_no_encontrado"
          }, 404);
        }
      }

      const mergedPayload = existing
        ? { ...existing, ...payload, efemeride_id:requestedId }
        : payload;

      const data = sanitizeEfemPayload(mergedPayload);
      const validationError = validateEfemPayload(data);

      if(validationError){
        return jsonResponse({
          success:false,
          message:validationError
        }, 400);
      }

      if(data.tipo === "GENERAL" && !ctx.permisos.efemerides_grl){
        return jsonResponse({ success:false, message:"Sin permiso para efemérides generales" }, 403);
      }

      if(data.tipo === "PROVINCIAL"){
        if(!ctx.permisos.efemerides_provincial){
          return jsonResponse({ success:false, message:"Sin permiso para efemérides provinciales" }, 403);
        }

        if(
          !ctx.permisos.todas_provincias &&
          !ctx.permisos.provincias.includes(data.provincia_id)
        ){
          return jsonResponse({ success:false, message:"Provincia no autorizada" }, 403);
        }
      }

      if(data.tipo === "LOCAL"){
        if(!ctx.permisos.efemerides_local){
          return jsonResponse({ success:false, message:"Sin permiso para efemérides locales" }, 403);
        }

        if(
          !ctx.permisos.todas_ciudades &&
          !ctx.permisos.ciudades.includes(data.ciudad_id)
        ){
          return jsonResponse({ success:false, message:"Ciudad no autorizada" }, 403);
        }
      }

      const id = requestedId || data.efemeride_id || generateEfemId();
      data.efemeride_id = id;
      data.actualizado_en = new Date().toISOString();

      if(!existing){
        data.creado_en = new Date().toISOString();
        data.creado_por = advertiserId;
      }
      else{
        if(existing.creado_en) data.creado_en = existing.creado_en;
        if(existing.creado_por) data.creado_por = existing.creado_por;
      }

      const saved = await firestoreSetDocument(
        env,
        COLL_EFEMERIDES,
        id,
        data
      );

      return jsonResponse({
        success:true,
        efemeride:addFrontendCompatibilityFields(saved)
      }, 200);
    }

    if(action === "activar" || action === "desactivar"){
      const id = String(
        body.efemeride_id ||
        body.id ||
        (body.payload && (body.payload.efemeride_id || body.payload.id)) ||
        ""
      ).trim();

      if(!id){
        return jsonResponse({
          success:false,
          message:"Falta efemeride_id"
        }, 400);
      }

      const existing = await firestoreGetDocument(env, COLL_EFEMERIDES, id);

      if(!existing){
        return jsonResponse({
          success:false,
          message:"Efeméride no encontrada"
        }, 404);
      }

      if(!canSeeEfemeride(existing, ctx.permisos)){
        return jsonResponse({
          success:false,
          message:"No autorizado para modificar esta efeméride"
        }, 403);
      }

      const updated = {
        ...existing,
        activo:action === "activar",
        actualizado_en:new Date().toISOString()
      };

      delete updated.id;

      const saved = await firestoreSetDocument(
        env,
        COLL_EFEMERIDES,
        id,
        updated
      );

      return jsonResponse({
        success:true,
        efemeride:addFrontendCompatibilityFields(saved)
      }, 200);
    }

    if(action === "eliminar" || action === "delete"){
      const id = String(
        body.efemeride_id ||
        body.id ||
        (body.payload && (body.payload.efemeride_id || body.payload.id)) ||
        ""
      ).trim();

      if(!id){
        return jsonResponse({
          success:false,
          message:"Falta efemeride_id"
        }, 400);
      }

      const existing = await firestoreGetDocument(env, COLL_EFEMERIDES, id);

      if(!existing){
        return jsonResponse({
          success:false,
          message:"Efeméride no encontrada"
        }, 404);
      }

      if(!canSeeEfemeride(existing, ctx.permisos)){
        return jsonResponse({
          success:false,
          message:"No autorizado para eliminar esta efeméride"
        }, 403);
      }

      await firestoreDeleteDocument(env, COLL_EFEMERIDES, id);

      return jsonResponse({
        success:true,
        efemeride_id:id
      }, 200);
    }

    if(action === "permisos" || action === "permissions"){
      return jsonResponse({
        success:true,
        permisos:ctx.permisos,
        permisos_source:ctx.permisos_source,
        permisos_migrados:ctx.permisos_migrados
      }, 200);
    }

    return jsonResponse({
      success:false,
      message:"Acción de Efemérides no reconocida",
      action
    }, 400);

  }catch(err){
    return jsonResponse({
      success:false,
      message:"Error en Efemérides Firestore",
      error:String(err && err.message ? err.message : err)
    }, 500);
  }
}

async function handleSearchRequest(request, env){
  try{
    const url = new URL(request.url);
    const q = (url.searchParams.get("q") || "").trim().toLowerCase();
    const idQuery = (url.searchParams.get("id") || url.searchParams.get("__id") || "").trim();

    const [anunciantes, administraciones] = await Promise.all([
      firestoreListCollection(env, COLL_ANUNCIANTES),
      firestoreListCollection(env, COLL_ADMIN)
    ]);

    const adminById = new Map();

    for(const admin of administraciones){
      const id = String(admin.id || admin.__id || "").trim();
      if(!id) continue;
      adminById.set(id, admin);
    }

    let results = [];

    for(const anunciante of anunciantes){
      const id = String(anunciante.id || anunciante.__id || "").trim();
      if(!id) continue;

      const admin = adminById.get(id) || {};

      if(
        admin.aprobado !== undefined &&
        admin.aprobado !== null &&
        admin.aprobado !== "" &&
        !truthyValue_(admin.aprobado)
      ){
        continue;
      }

      results.push({
        id,
        nombre:String(anunciante.nombre || "").trim(),
        nivel:String(admin.nivel ?? "").trim(),
        subnivel:String(admin.subnivel ?? "").trim(),
        aprobado:admin.aprobado ?? "",
        verificado:admin.verificado ?? "",
        gold:admin.gold ?? "",
        funcionalidades:Array.isArray(admin.funcionalidades)
          ? admin.funcionalidades
          : String(admin.funcionalidades || "")
              .split(/[;,|\n]+/)
              .map(v => v.trim())
              .filter(Boolean),
        segmento_id:String(anunciante.segmento_id || "").trim(),
        categoria_ids:Array.isArray(anunciante.categoria_ids)
          ? anunciante.categoria_ids
          : [],
        actividad:String(anunciante.actividad || "").trim(),
        descripcion:String(anunciante.descripcion || "").trim(),
        tags:String(anunciante.tags || "").trim(),
        adicionales:String(anunciante.adicionales || "").trim(),
        logo:String(anunciante.logo || "").trim(),
        link:String(anunciante.link || "").trim()
      });
    }

    if(idQuery){
      results = results.filter(r => r.id === idQuery);
    }

    if(q){
      results = results.filter(r =>
        [
          r.nombre,
          r.actividad,
          r.tags,
          r.descripcion,
          r.id
        ].some(v => String(v || "").toLowerCase().includes(q))
      );
    }

    results.sort((a,b) =>
      String(a.nombre || "").localeCompare(String(b.nombre || ""), "es")
    );

    return jsonResponse({
      success:true,
      count:results.length,
      results,
      source:"firestore"
    }, 200);

  }catch(err){
    return jsonResponse({
      success:false,
      message:"No se pudieron cargar los anunciantes desde Firestore.",
      error:String(err && err.message ? err.message : err)
    }, 500);
  }
}

function truthyValue_(value){
  if(value === true || value === 1) return true;

  const normalized = String(value ?? "")
    .trim()
    .toLowerCase()
    .normalize("NFD")
    .replace(/[\u0300-\u036f]/g, "");

  return [
    "true",
    "1",
    "si",
    "yes",
    "x",
    "✓",
    "aprobado",
    "ok",
    "activo",
    "activa"
  ].includes(normalized);
}

function getAdvertiserIdFromCookies(cookieHeader){
  if(!cookieHeader) return "";

  const parts = cookieHeader.split(";").map(s => s.trim());

  for(const p of parts){
    if(p.startsWith("advertiserId=")){
      return decodeURIComponent(p.split("=")[1] || "");
    }

    if(p.startsWith("advertiser_id=")){
      return decodeURIComponent(p.split("=")[1] || "");
    }
  }

  return "";
}

function getAdvertiserIdFromHeaders(headers){
  if(!headers) return "";

  return headers.get
    ? (headers.get("x-advertiser-id") || headers.get("x-advertiserid") || "")
    : (headers["x-advertiser-id"] || headers["x-advertiserid"] || "");
}

async function proxySuscriptoresRecoveryToAppsScript_(request, env){
  try{
    const url = new URL(request.url);
    let raw = "";
    let contentType = request.headers.get("content-type") || "";
    /** @type {any} */
    let parsed = {};

    if(request.method === "POST"){
      raw = await request.text();

      if(contentType.includes("application/json")){
        try{ parsed = raw ? JSON.parse(raw) : {}; }
        catch(_){ parsed = {}; }
      }
      else{
        const form = new URLSearchParams(raw);
        for(const [k,v] of form.entries()) parsed[k] = v;
      }
    }
    else{
      url.searchParams.forEach((v,k)=>{ parsed[k] = v; });
    }

    parsed.serverSecret = env.SERVER_SECRET || "";

    const resp = await fetch(APPSCRIPT_SUSCRIPTORES_URL, {
      method:"POST",
      headers:{"Content-Type":"application/json"},
      body:JSON.stringify(parsed)
    });

    const text = await resp.text();
    let json = null;
    try{ json = text ? JSON.parse(text) : null; }catch(_){}

    if(json !== null){
      return new Response(JSON.stringify(json), {
        status:resp.status,
        headers:{
          "Content-Type":"application/json",
          ...corsHeaders()
        }
      });
    }

    return jsonResponse({
      success:false,
      error:"recovery_bridge_invalid_response",
      message:"El puente de recuperación no devolvió una respuesta JSON válida.",
      upstream_status:resp.status
    },502);

  }catch(err){
    return jsonResponse({
      success:false,
      error:"recovery_bridge_failed",
      message:"No se pudo conectar con el puente de recuperación de contraseña.",
      detail:String(err && err.message ? err.message : err)
    },502);
  }
}

    .trim()

function publicSedeActiva_(sede){
  if(!sede) return false;

  ].includes(s);
}

function catalogMapById_(rows, idField){
  const map = new Map();
  for(const row of (rows || [])){
    const id = String(row[idField] || row.id || "").trim();
    if(id) map.set(id,row);
  }
  return map;
}

function resolvePublicClassification_(advertiser, segmentos, categorias){
  const segMap = catalogMapById_(segmentos,"segmento_id");
  const catMap = catalogMapById_(categorias,"categoria_id");

  const segmentoId = String(advertiser.segmento_id || "").trim();
  const categoriaIds = normalizeIdsArray_(
    advertiser.categoria_ids || advertiser.categoria_id
  );

  const seg = segMap.get(segmentoId) || {};
  const categoriaNames = categoriaIds
    .map(id => catMap.get(id))
    .filter(Boolean)
    .map(r => String(r.nombre || "").trim())
    .filter(Boolean);

  return {
    segmento_id:segmentoId,
    segmento:String(seg.nombre || "").trim(),
    categoria_id:categoriaIds.join(";"),
    categoria:categoriaNames.join("; "),
    segmento_ids:segmentoId ? [segmentoId] : [],
    categoria_ids:categoriaIds,
    segmentos:seg.nombre ? [String(seg.nombre).trim()] : [],
    categorias:categoriaNames
  };
}

function publicTerritoryForSede_(sede, cityMap, provinceMap, countryMap){
  const ciudadId = String(sede.ciudad_id || "").trim();
  const ciudad = cityMap.get(ciudadId) || {};

  const provinciaId = String(
    sede.provincia_id ||
    ciudad.provincia_id ||
    ""
  ).trim();

  const provincia = provinceMap.get(provinciaId) || {};

  const paisId = String(
    sede.pais_id ||
    ciudad.pais_id ||
    provincia.pais_id ||
    ""
  ).trim();

  const pais = countryMap.get(paisId) || {};

  return {
    pais_id:paisId,
    pais:String(
      sede.pais ||
      ciudad.pais ||
      provincia.pais ||
      pais.nombre ||
      pais.pais ||
      ""
    ).trim(),
    provincia_id:provinciaId,
    provincia:String(
      sede.provincia ||
      ciudad.provincia ||
      provincia.nombre ||
      provincia.provincia ||
      ""
    ).trim(),
    ciudad_id:ciudadId,
    ciudad:String(
      sede.ciudad ||
      ciudad.nombre ||
      ciudad.ciudad ||
      ""
    ).trim(),
    codigo_postal:String(
      sede.codigo_postal ||
      ciudad.codigo_postal ||
      ""
    ).trim()
  };
}

function publicRelationObjects_(ids, map, idField){
  return normalizeIdsArray_(ids).map(id => {
    const row = map.get(id) || {};
    return {
      [idField]:id,
      nombre:String(row.nombre || "").trim(),
      insignia:String(row.insignia || "").trim()
    };
  });
}

async function loadPublicGuideDataForTerritory_(env,paisId,provinciaId,ciudadId){
  paisId=String(paisId||"").trim();
  provinciaId=String(provinciaId||"").trim();
  ciudadId=String(ciudadId||"").trim();

  // La Guía pública siempre debe trabajar con un territorio. Evita cualquier
  // lectura accidental de anunciantes/sedes de todas las ciudades.
  if(!ciudadId && !provinciaId && !paisId){
    return {
      advertisers:[],admins:[],sedes:[],
      segmentos:[],categorias:[],actividades:[],acciones:[],nodos:[],
      funcionalidades:[],niveles_anunciante:[],
      ciudades:[],provincias:[],paises:[]
    };
  }

  let sedes=[];
  if(ciudadId){
    sedes=await firestoreQueryEqual(env,COLL_SEDES,"ciudad_id",ciudadId);
  }else if(provinciaId){
    sedes=await firestoreQueryEqual(env,COLL_SEDES,"provincia_id",provinciaId);
  }else{
    sedes=await firestoreQueryEqual(env,COLL_SEDES,"pais_id",paisId);
  }
  sedes=(sedes||[]).filter(publicSedeActiva_);

  const advertiserIds=[...new Set(
    sedes.map(s=>String(s.anunciante_id||"").trim()).filter(Boolean)
  )];

  const [advertisers,admins,catalogs] = await Promise.all([
    firestoreGetMany_(env,COLL_ANUNCIANTES,advertiserIds),
    firestoreGetMany_(env,COLL_ADMIN,advertiserIds),
    loadPublicGuideCatalogsLight_(env)
  ]);

  let ciudadDoc=null, provinciaDoc=null, paisDoc=null;

  if(ciudadId){
    ciudadDoc=await firestoreGetDocument(env,COLL_CIUDADES,ciudadId);
    if(!provinciaId){
      provinciaId=String(ciudadDoc && ciudadDoc.provincia_id || "").trim();
    }
    if(!paisId){
      paisId=String(ciudadDoc && ciudadDoc.pais_id || "").trim();
    }
  }

  if(provinciaId){
    provinciaDoc=await firestoreGetDocument(env,COLL_PROVINCIAS,provinciaId);
    if(!paisId){
      paisId=String(provinciaDoc && provinciaDoc.pais_id || "").trim();
    }
  }

  if(paisId){
    paisDoc=await firestoreGetDocument(env,COLL_PAISES,paisId);
  }

  return {
    advertisers,
    admins,
    sedes,
    segmentos:catalogs.segmentos||[],
    categorias:catalogs.categorias||[],
    actividades:catalogs.actividades||[],
    acciones:catalogs.acciones||[],
    nodos:catalogs.nodos||[],
    funcionalidades:[],
    niveles_anunciante:[],
    ciudades:ciudadDoc?[ciudadDoc]:[],
    provincias:provinciaDoc?[provinciaDoc]:[],
    paises:paisDoc?[paisDoc]:[]
  };
}

function buildPublicGuideRows_(data){
  const adminMap = new Map(
    data.admins.map(r => [String(r.id || "").trim(),r])
  );

  const sedesByAdvertiser = new Map();
  for(const sede of data.sedes){
    if(!publicSedeActiva_(sede)) continue;
    const advertiserId = String(sede.anunciante_id || "").trim();
    if(!advertiserId) continue;
    if(!sedesByAdvertiser.has(advertiserId)){
      sedesByAdvertiser.set(advertiserId,[]);
    }
    sedesByAdvertiser.get(advertiserId).push(sede);
  }

  const cityMap = catalogMapById_(data.ciudades,"ciudad_id");
  const provinceMap = catalogMapById_(data.provincias,"provincia_id");
  const countryMap = catalogMapById_(data.paises,"pais_id");
  const activityMap = catalogMapById_(data.actividades,"actividad_id");
  const actionMap = catalogMapById_(data.acciones,"accion_id");
  const nodeMap = catalogMapById_(data.nodos,"nodo_id");

  const rows = [];

  for(const advertiser of data.advertisers){
    const id = String(advertiser.id || "").trim();
    if(!id) continue;

    const admin = adminMap.get(id);
    if(!admin) continue;

    // Sólo publica anunciantes aprobados.
    if(
      admin.aprobado !== undefined &&
      !truthyValue_(admin.aprobado)
    ){
      continue;
    }

    const cls = resolvePublicClassification_(
      advertiser,
      data.segmentos,
      data.categorias
    );

    const sedeBundles = (sedesByAdvertiser.get(id) || []).map(raw => {
      const geo = publicTerritoryForSede_(
        raw,
        cityMap,
        provinceMap,
        countryMap
      );

      const actividades = publicRelationObjects_(
        raw.actividad_ids,
        activityMap,
        "actividad_id"
      );

      const acciones = publicRelationObjects_(
        raw.accion_ids,
        actionMap,
        "accion_id"
      );

      const nodos = publicRelationObjects_(
        raw.nodo_ids,
        nodeMap,
        "nodo_id"
      );

      const sede = {
        sede_id:String(raw.sede_id || raw.id || "").trim(),
        anunciante_id:id,
        nombre_sede:String(raw.nombre_sede || "").trim(),
        direccion:String(raw.direccion || "").trim(),
        pais_id:geo.pais_id,
        pais:geo.pais,
        provincia_id:geo.provincia_id,
        provincia:geo.provincia,
        ciudad_id:geo.ciudad_id,
        ciudad:geo.ciudad,
        codigo_postal:geo.codigo_postal,
        lat:raw.lat || "",
        lng:raw.lng || "",
        maps:raw.maps || "",
        telefono:raw.telefono || "",
        telefono2:raw.telefono2 || "",
        telefono3:raw.telefono3 || "",
        telefono4:raw.telefono4 || "",
        telefono5:raw.telefono5 || "",
        whatsapp:raw.whatsapp || "",
        whatsapp2:raw.whatsapp2 || "",
        whatsapp3:raw.whatsapp3 || "",
        whatsapp4:raw.whatsapp4 || "",
        whatsapp5:raw.whatsapp5 || "",
        mail:raw.mail || "",
        mail2:raw.mail2 || "",
        mail3:raw.mail3 || "",
        instagram:raw.instagram || "",
        facebook:raw.facebook || "",
        youtube:raw.youtube || "",
        tiktok:raw.tiktok || "",
        x:raw.x || "",
        linkedin:raw.linkedin || "",
        estado:raw.estado || "",
        actividades
      };

      for(let i=1;i<=10;i++){
        sede["img"+i] = raw["img"+i] || "";
      }

      return {
        sede,
        actividades,
        acciones,
        nodos
      };
    });

    const uniqueBy = (items, field) => {
      const map = new Map();
      for(const item of items){
        const key = String(item && item[field] || "").trim();
        if(key && !map.has(key)) map.set(key,item);
      }
      return [...map.values()];
    };

    const actividades = uniqueBy(
      sedeBundles.flatMap(x => x.actividades),
      "actividad_id"
    );

    const acciones = uniqueBy(
      sedeBundles.flatMap(x => x.acciones),
      "accion_id"
    );

    const nodos = uniqueBy(
      sedeBundles.flatMap(x => x.nodos),
      "nodo_id"
    );

    const sedesPublicas = sedeBundles.map(x => x.sede);

    const buscador = [
      advertiser.nombre,
      cls.segmento,
      cls.categoria,
      advertiser.actividad,
      advertiser.descripcion,
      advertiser.tags,
      advertiser.adicionales,
      ...sedesPublicas.flatMap(sd => [
        sd.nombre_sede,
        sd.direccion,
        sd.pais,
        sd.provincia,
        sd.ciudad
      ]),
      ...actividades.map(x => x.nombre),
      ...acciones.map(x => x.nombre),
      ...nodos.map(x => x.nombre)
    ].filter(Boolean).join(" | ");

    const first = sedesPublicas[0] || {};

    const out = {
      id,
      nombre:String(advertiser.nombre || "").trim(),
      nivel:admin.nivel || "",
      subnivel:admin.subnivel || "",
      segmento_id:cls.segmento_id,
      segmento:cls.segmento,
      categoria_id:cls.categoria_id,
      categoria:cls.categoria,
      actividad:String(advertiser.actividad || "").trim(),
      descripcion:String(advertiser.descripcion || "").trim(),
      tags:String(advertiser.tags || "").trim(),
      adicionales:String(advertiser.adicionales || "").trim(),
      logo:String(advertiser.logo || "").trim(),
      link:String(advertiser.link || "").trim(),
      sedes:sedesPublicas,
      actividades_clave:actividades,
      acciones,
      nodos,
      capacidades:publicAdmin_(admin),
      buscador_texto:buscador
    };

    [
      "telefono","telefono2","telefono3","telefono4","telefono5",
      "whatsapp","whatsapp2","whatsapp3","whatsapp4","whatsapp5",
      "mail","mail2","mail3",
      "instagram","facebook","youtube","tiktok","x","linkedin"
    ].forEach(k => out[k] = first[k] || "");

    for(let i=1;i<=10;i++){
      out["img"+i] = first["img"+i] || "";
    }

    sedesPublicas.slice(0,6).forEach((sd,i) => {
      const suffix = i === 0 ? "" : String(i+1);
      out["direccion"+suffix] = sd.direccion || "";
      out["maps"+suffix] = sd.maps || "";
      out["lat"+suffix] = sd.lat || "";
      out["lng"+suffix] = sd.lng || "";
      out["nombre_sede"+suffix] = sd.nombre_sede || "";
      out["ciudad"+suffix] = sd.ciudad || "";
      out["provincia"+suffix] = sd.provincia || "";
      out["pais"+suffix] = sd.pais || "";
    });

    rows.push(out);
  }

  return rows;
}

function filterPublicGuideRows_(rows, params){
  let out = rows.slice();

  const q = normalizePublicText_(
    params.get("q") ||
    params.get("buscar") ||
    params.get("search") ||
    ""
  );

  const segmentoId = String(
    params.get("segmento_id") ||
    params.get("segmento") ||
    ""
  ).trim();

  const categoriaId = String(
    params.get("categoria_id") ||
    params.get("categoria") ||
    ""
  ).trim();

  const ciudadId = String(
    params.get("ciudad_id") ||
    params.get("ciudad") ||
    params.get("city") ||
    ""
  ).trim();

  const provinciaId = String(
    params.get("provincia_id") ||
    params.get("provincia") ||
    ""
  ).trim();

  const paisId = String(
    params.get("pais_id") ||
    params.get("pais") ||
    ""
  ).trim();

  if(q){
    out = out.filter(r =>
      normalizePublicText_(
        r.buscador_texto ||
        [
          r.nombre,r.segmento,r.categoria,r.actividad,
          r.descripcion,r.tags,r.adicionales
        ].join(" ")
      ).includes(q)
    );
  }

  if(segmentoId){
    out = out.filter(r =>
      String(r.segmento_id || "").trim() === segmentoId ||
      normalizePublicText_(r.segmento) === normalizePublicText_(segmentoId)
    );
  }

  if(categoriaId){
    out = out.filter(r => {
      const ids = normalizeIdsArray_(r.categoria_id);
      const names = String(r.categoria || "")
        .split(/[;,|\n]+/)
        .map(v => v.trim())
        .filter(Boolean);

      return (
        ids.includes(categoriaId) ||
        names.some(x =>
          normalizePublicText_(x) === normalizePublicText_(categoriaId)
        )
      );
    });
  }

  if(ciudadId || provinciaId || paisId){
    out = out.filter(r => (r.sedes || []).some(sd => {
      const okCiudad = !ciudadId ||
        String(sd.ciudad_id || "").trim() === ciudadId ||
        normalizePublicText_(sd.ciudad) === normalizePublicText_(ciudadId);

      const okProv = !provinciaId ||
        String(sd.provincia_id || "").trim() === provinciaId ||
        normalizePublicText_(sd.provincia) === normalizePublicText_(provinciaId);

      const okPais = !paisId ||
        String(sd.pais_id || "").trim() === paisId ||
        normalizePublicText_(sd.pais) === normalizePublicText_(paisId);

      return okCiudad && okProv && okPais;
    }));
  }

  return out;
}

function buildPublicBootstrap_(data){
  const activityMap = catalogMapById_(data.actividades,"actividad_id");
  const actionMap = catalogMapById_(data.acciones,"accion_id");
  const nodeMap = catalogMapById_(data.nodos,"nodo_id");

  const relacionesSedes = {};

  for(const sede of data.sedes){
    const sedeId = String(sede.sede_id || sede.id || "").trim();
    if(!sedeId) continue;

    relacionesSedes[normalizePublicText_(sedeId)] = {
      actividades:publicRelationObjects_(
        sede.actividad_ids,
        activityMap,
        "actividad_id"
      ),
      acciones:publicRelationObjects_(
        sede.accion_ids,
        actionMap,
        "accion_id"
      ),
      nodos:publicRelationObjects_(
        sede.nodo_ids,
        nodeMap,
        "nodo_id"
      )
    };
  }

  const adminMap = new Map(
    data.admins.map(r => [String(r.id || "").trim(),r])
  );

  const metaPorId = {};
  for(const advertiser of data.advertisers){
    const id = String(advertiser.id || "").trim();
    if(!id) continue;

    const admin = adminMap.get(id) || {};
    metaPorId[id] = {
      link:String(advertiser.link || "").trim(),
      verificado:admin.verificado || "",
      gold:admin.gold || ""
    };
  }

  return {
    success:true,
    paises:data.paises,
    provincias:data.provincias,
    ciudades:data.ciudades,
    segmentos:data.segmentos,
    segmentos_publicos:data.segmentos,
    micrositios:[],
    categorias:data.categorias,
    actividades_clave:data.actividades,
    acciones:data.acciones,
    nodos:data.nodos,
    funcionalidades:data.funcionalidades,
    niveles_anunciante:data.niveles_anunciante||[],
    relaciones_sedes:relacionesSedes,
    meta_por_id:metaPorId,
    source:"firestore"
  };
}

function publicResponse_(data, callback, status=200){
  const cb = String(callback || "").trim();

  if(cb && /^[A-Za-z_$][A-Za-z0-9_$.]*$/.test(cb)){
    const normalized=normalizeDriveImageFieldsDeep_(data);
    return new Response(`${cb}(${JSON.stringify(normalized)});`, {
      status,
      headers:{
        ...corsHeaders(),
        "Content-Type":"application/javascript; charset=utf-8",
        "Cache-Control":"no-store"
      }
    });
  }

  return jsonResponse(data,status);
}

async function handlePublicCommerceGet_(url, env){
  const action = String(url.searchParams.get("action") || "")
    .trim()
    .toLowerCase();

  const callback = url.searchParams.get("callback") || "";

  if(![
    "ubicaciones","locations","bootstrap","guia",
    "paises","provincias","ciudades",
    "segmentos","categorias","actividades_clave",
    "actividadesclave","acciones","nodos","funcionalidades",
    "sedes"
  ].includes(action)){
    return null;
  }

  if(action === "guia"){
    const paisId=String(url.searchParams.get("pais_id")||"").trim();
    const provinciaId=String(url.searchParams.get("provincia_id")||"").trim();
    const ciudadId=String(url.searchParams.get("ciudad_id")||"").trim();
    const cacheKey=guideCacheKey_(paisId,provinciaId,ciudadId);

    let baseRows=await gldCacheGetJson_(cacheKey);

    if(!Array.isArray(baseRows)){
      const data=await loadPublicGuideDataForTerritory_(
        env,paisId,provinciaId,ciudadId
      );
      baseRows=buildPublicGuideRows_(data);
      await gldCachePutJson_(cacheKey,baseRows,1800);
    }

    const rows=filterPublicGuideRows_(
      baseRows,
      url.searchParams
    );
    return publicResponse_(rows,callback);
  }

  if(action === "ubicaciones" || action === "locations"){
    const locations=await loadPublicLocations_(env);
    return publicResponse_({
      success:true,
      paises:locations.paises||[],
      provincias:locations.provincias||[],
      ciudades:locations.ciudades||[],
      source:"firestore"
    },callback);
  }

  const data = await loadPublicGuideData_(env);

  if(action === "bootstrap"){
    return publicResponse_(buildPublicBootstrap_(data),callback);
  }

  if(action === "paises"){
    return publicResponse_({
      success:true,
      count:data.paises.length,
      results:data.paises,
      source:"firestore"
    },callback);
  }

  if(action === "provincias"){
    const pais = String(
      url.searchParams.get("pais_id") ||
      url.searchParams.get("pais") ||
      ""
    ).trim();

    const rows = pais
      ? data.provincias.filter(r =>
          String(r.pais_id || "").trim() === pais ||
          normalizePublicText_(r.pais || r.nombre) === normalizePublicText_(pais)
        )
      : data.provincias;

    return publicResponse_({
      success:true,
      count:rows.length,
      results:rows,
      source:"firestore"
    },callback);
  }

  if(action === "ciudades"){
    const provincia = String(
      url.searchParams.get("provincia_id") ||
      url.searchParams.get("provincia") ||
      ""
    ).trim();

    const pais = String(
      url.searchParams.get("pais_id") ||
      url.searchParams.get("pais") ||
      ""
    ).trim();

    let rows = data.ciudades.slice();

    if(provincia){
      rows = rows.filter(r =>
        String(r.provincia_id || "").trim() === provincia ||
        normalizePublicText_(r.provincia || "") === normalizePublicText_(provincia)
      );
    }

    if(pais){
      rows = rows.filter(r =>
        String(r.pais_id || "").trim() === pais ||
        normalizePublicText_(r.pais || "") === normalizePublicText_(pais)
      );
    }

    return publicResponse_({
      success:true,
      count:rows.length,
      results:rows,
      source:"firestore"
    },callback);
  }

  const catalogs = {
    segmentos:data.segmentos,
    categorias:data.categorias,
    actividades_clave:data.actividades,
    actividadesclave:data.actividades,
    acciones:data.acciones,
    nodos:data.nodos,
    funcionalidades:data.funcionalidades
  };

  if(catalogs[action]){
    return publicResponse_({
      success:true,
      count:catalogs[action].length,
      results:catalogs[action],
      source:"firestore"
    },callback);
  }

  if(action === "sedes"){
    const advertiserId = String(
      url.searchParams.get("id") ||
      url.searchParams.get("anunciante_id") ||
      ""
    ).trim();

    const rows = advertiserId
      ? data.sedes.filter(
          s => String(s.anunciante_id || "").trim() === advertiserId
        )
      : data.sedes;

    return publicResponse_({
      success:true,
      count:rows.length,
      results:rows,
      source:"firestore"
    },callback);
  }

  return null;
}

function normalizeIdsArray_(value){
  if(Array.isArray(value)){
    return [...new Set(value.map(v=>String(v||"").trim()).filter(Boolean))];
  }

  return [...new Set(
    String(value||"")
      .split(/[;,|\n]+/)
      .map(v=>v.trim())
      .filter(Boolean)
  )];
}

function generateAdvertiserId_(){
  return crypto.randomUUID().replace(/-/g,"").slice(0,8);
}

function generateSedeId_(advertiserId){
  const suffix = crypto.randomUUID().replace(/-/g,"").slice(0,6).toUpperCase();
  return `SED-${advertiserId}-${suffix}`;
}

async function loadCommerceCatalogs_(env){
  const cacheKey="commerce:catalogs:v2";
  const cached=await gldCacheGetJson_(cacheKey);
  if(cached && typeof cached==="object"){

  }

  const [
    segmentos,

function resolveClassification_(datos, catalogs){
  const segmentoId = String(datos.segmento_id || "").trim();
  const categoriaIds = normalizeIdsArray_(
    Array.isArray(datos.categoria_ids) ? datos.categoria_ids : datos.categoria_id
  );

  const seg = (catalogs.segmentos || []).find(
    r => String(r.segmento_id || r.id || "").trim() === segmentoId
  ) || {};

  const categoriaMap = new Map(
    (catalogs.categorias || []).map(r => [
      String(r.categoria_id || r.id || "").trim(),
      r
    ])
  );

  const categoriaNames = categoriaIds
    .map(id => categoriaMap.get(id))
    .filter(Boolean)
    .map(r => String(r.nombre || "").trim())
    .filter(Boolean);

  return {
    segmento_id:segmentoId,
    segmento:String(seg.nombre || "").trim(),
    categoria_id:categoriaIds.join(";"),
    categoria:categoriaNames.join("; "),
    segmento_ids:segmentoId ? [segmentoId] : [],
    categoria_ids:categoriaIds,
    segmentos:seg.nombre ? [String(seg.nombre).trim()] : [],
    categorias:categoriaNames
  };
}

}

async function commerceSetDatos_(env, body){
  const id = String(
    body.id ||
    body.__id ||
    body.comercio_id ||
    body.advertiserId ||
    ""
  ).trim();

  if(!id){
    return jsonResponse({success:false,error:"Falta id"},400);
  }

  const current = await firestoreGetDocument(env, COLL_ANUNCIANTES, id);
  if(!current){
    return jsonResponse({
      success:false,
      error:"anunciante_no_encontrado"
    },404);
  }

  const reserved = new Set([
    "action","accion","id","__id","comercio_id","advertiserId"
  ]);

  const patch = {};
  for(const [key,value] of Object.entries(body)){
    if(reserved.has(key)) continue;

    if(key === "categoria_id"){
      patch.categoria_ids = normalizeIdsArray_(value);
      continue;
    }

    patch[key] = value;
  }

  const merged = {
    ...current,
    ...patch,
    id,
    actualizado_en:new Date().toISOString()
  };

  delete merged.__id;

  await firestoreSetDocument(env, COLL_ANUNCIANTES, id, merged);
  await refreshGuideCacheForAdvertiser_(env, id);

  return jsonResponse({
    success:true,
    updated:true,
    id,
    source:"firestore"
  },200);
}

async function commerceSetSedes_(env, body){
  const advertiserId = String(
    body.id ||
    body.__id ||
    body.comercio_id ||
    body.advertiserId ||
    ""
  ).trim();

  if(!advertiserId){
    return jsonResponse({success:false,error:"Falta id"},400);
  }

  const incoming = Array.isArray(body.sedes) ? body.sedes : [];
  const saved = [];
  const oldCityIds = new Set();

  for(const raw of incoming){
    const existingId = String(raw && raw.sede_id || "").trim();
    const sedeId = existingId || generateSedeId_(advertiserId);

    const existing = existingId
      ? await firestoreGetDocument(env, COLL_SEDES, sedeId)
      : null;

    if(existing && existing.ciudad_id){
      oldCityIds.add(String(existing.ciudad_id).trim());
    }

    const sede = {
      ...(existing || {}),
      ...(raw || {}),
      sede_id:sedeId,
      anunciante_id:advertiserId,
      actividad_ids:normalizeIdsArray_(raw && raw.actividad_ids),
      accion_ids:normalizeIdsArray_(raw && raw.accion_ids),
      nodo_ids:normalizeIdsArray_(raw && raw.nodo_ids),
      actualizado_en:new Date().toISOString()
    };

    delete sede.__id;

    await firestoreSetDocument(env, COLL_SEDES, sedeId, sede);
    saved.push(sedeId);
  }

  await refreshGuideCacheForAdvertiser_(
    env,
    advertiserId,
    [...oldCityIds]
  );

  return jsonResponse({
    success:true,
    updated:true,
    id:advertiserId,
    sedes:saved,
    source:"firestore"
  },200);
}

async function handleCommerceRequest(request, env){
  if(request.method === "GET"){
    const url = new URL(request.url);

    const publicResponse = await handlePublicCommerceGet_(url, env);
    if(publicResponse) return publicResponse;

    const action = String(url.searchParams.get("action") || "").trim().toLowerCase();
    const id = String(
      url.searchParams.get("id") ||
      url.searchParams.get("__id") ||
      url.searchParams.get("advertiserId") ||
      ""
    ).trim();

    if(action === "anunciante"){
      return await commerceGetAdvertiser_(env, id);
    }

    if(action === "permisos_panel"){
      return await commerceGetPermissions_(env, id);
    }

    return jsonResponse({
      success:false,
      error:"accion_commerce_no_soportada",
      action
    },400);
  }

  if(request.method !== "POST"){
    return jsonResponse({success:false,error:"Método no permitido"},405);
  }

  let body = {};
  try{
    body = await request.json();
  }catch(_){
    return jsonResponse({success:false,error:"JSON inválido"},400);
  }

  const action = String(body.action || body.accion || "")
    .trim()
    .toLowerCase();

  if(action === "set_datos"){
    return await commerceSetDatos_(env, body);
  }

  if(action === "set_sedes"){
    return await commerceSetSedes_(env, body);
  }

  if(action === "set_relaciones_sede"){
    return await commerceSetRelacionesSede_(env, body);
  }

  if(action === "delete_sede"){
    return await commerceDeleteSede_(env, body);
  }

  if(
    action === "crear_anunciante" ||
    action === "create_advertiser" ||
    action === "duplicar_anunciante" ||
    action === "duplicate_advertiser" ||
    action === "eliminar_anunciante" ||
    action === "delete_advertiser"
  ){
    return jsonResponse({
      success:false,
      error:"accion_requiere_superadmin",
      message:"Esta operación queda deshabilitada hasta implementar autorización de Superadministrador."
    },403);
  }


  return jsonResponse({
    success:false,
    error:"accion_commerce_no_soportada",
    action
  },400);
}

function susNorm_(v){
  return String(v == null ? "" : v).trim().toLowerCase();
}

function susTruthy_(v){
  if(v === true || v === 1) return true;
  return ["true","1","si","sí","yes","x","activo","activa"].includes(susNorm_(v));
}

function susPublic_(obj){
  if(!obj) return null;

    fecha_alta:obj.fecha_alta || "",
    ultimo_acceso:obj.ultimo_acceso || ""
  };
}

function susRandomId_(prefix, length=10){
  const chars = "abcdefghijklmnopqrstuvwxyz0123456789";
  const bytes = new Uint8Array(length);
  crypto.getRandomValues(bytes);

  let out = prefix;
  for(let i=0;i<length;i++){
    out += chars[bytes[i] % chars.length];
  }
  return out;
}

  return rows.find(s => susNorm_(s.mail) === needle) || null;
}

async function susAuthorizedAdvertisers_(env, suscriptorId){
  const sid = String(suscriptorId || "").trim();
  if(!sid) return [];

  const relations = await firestoreQueryEqual(
    env,
    COLL_SUSCRIPTOR_ANUNCIANTE,
    "suscriptor_id",
    sid
  );

  const activeRelations = relations.filter(rel =>
    String(rel.suscriptor_id || "").trim() === sid &&
    String(rel.anunciante_id || "").trim() &&
    susTruthy_(rel.activo)
  );

  const advertiserIds = activeRelations
    .map(rel=>String(rel.anunciante_id||"").trim())
    .filter(Boolean);

  const [advertisers, admins] = await Promise.all([
    firestoreGetMany_(env, COLL_ANUNCIANTES, advertiserIds),
    firestoreGetMany_(env, COLL_ADMIN, advertiserIds)
  ]);

  const advertiserById = new Map(
    advertisers.map(a => [String(a.id || "").trim(), a])
  );
  const adminById = new Map(
    admins.map(a => [String(a.id || "").trim(), a])
  );

  const out = [];

  for(const rel of activeRelations){
    const aid = String(rel.anunciante_id || "").trim();
    const advertiser = advertiserById.get(aid) || {};
    const admin = adminById.get(aid) || {};
    const funcionalidades = splitFeatures(admin.funcionalidades || []);

    out.push({
      suscriptor_id:sid,
      anunciante_id:aid,
      anunciante_nombre:String(
        advertiser.nombre ||
        admin.nombre ||
        rel.anunciante_nombre ||
        ""
      ).trim(),
      rol:String(rel.rol || "").trim().toUpperCase(),
      permisos:funcionalidades.join(", "),
      activo:true,
      fecha_alta:rel.fecha_alta || "",
      actualizado:rel.actualizado || ""
    });
  }

  return out;
}

async function susLogin_(env, p){
  const mail = susNorm_(p.mail);
  const clave = String(p.clave || "").trim();

  if(!mail || !clave){
    return {success:false,message:"Falta mail o clave"};
  }

  const sus = await susFindByMail_(env, mail);
  if(!sus){
    return {success:false,message:"Mail o clave incorrectos"};
  }

  if(
    sus.activo !== "" &&
    sus.activo !== null &&
    sus.activo !== undefined &&
    !susTruthy_(sus.activo)
  ){
    return {success:false,message:"Suscriptor no activo"};
  }

  if(String(sus.clave || "").trim() !== clave){
    return {success:false,message:"Mail o clave incorrectos"};
  }

  // Las cuentas nuevas deben verificar su correo.
  // Las cuentas históricas, que no tienen todavía este campo, siguen ingresando.
  if(sus.email_verificado === false || String(sus.email_verificado || "").toLowerCase() === "false"){
    return {
      success:false,
      message:"Primero verificá tu correo electrónico.",
      requiere_verificacion:true,
      mail:sus.mail || mail
    };
  }

  const sid = String(sus.suscriptor_id || sus.id || "").trim();
  const now = new Date().toISOString();

  const updated = {
    ...sus,
    id:undefined,
    ultimo_acceso:now
  };

  await firestoreSetDocument(env, COLL_SUSCRIPTORES, sid, updated);

  const anunciantes = await susAuthorizedAdvertisers_(env, sid);

  return {
    success:true,
    message:"Login OK",
    suscriptor:susPublic_({...sus,ultimo_acceso:now}),
    anunciantes,
    source:"firestore"
  };
}

  };

async function susAddFavorite_(env, p){

  if(!validacion.ok){
    return {success:false,message:validacion.message};

    const favoriteId = String(found.favorito_id || found.id || "").trim();

      ...found,
      id:undefined,

      favoriteId,

    );

      break;
    }

  };

    env,
    COLL_SUSCRIPTOR_FAVORITOS,
    favoriteId,
    doc
  );

  return {
    success:true,
    message:"Favorito guardado",
    favorito:{
      favorito_id:String(saved.favorito_id || favoriteId).trim(),
      suscriptor_id:String(saved.suscriptor_id || "").trim(),

const COLL_SOLICITUDES_ANUNCIANTE = "solicitudes_anunciante";

async function susSolicitarAnunciante_(env, p){
  const sid = String(p.suscriptor_id || p.id || "").trim();
  const modo = String(p.modo || "CREAR").trim().toUpperCase();
  const nombre = String(p.nombre || "").trim();
  const aid = String(p.anunciante_id || "").trim();

  if(!sid) return {success:false,message:"Falta suscriptor_id"};
  if(!["CREAR","RECLAMAR"].includes(modo)) return {success:false,message:"Tipo de solicitud inválido"};
  if(!nombre) return {success:false,message:"Falta nombre del anunciante"};

  const validacion = await susRequireActive_(env, sid);
  if(!validacion.ok) return {success:false,message:validacion.message};

  if(modo === "RECLAMAR"){
    if(!aid) return {success:false,message:"Seleccioná un anuncio existente"};
    const anunciante = await firestoreGetDocument(env, COLL_ANUNCIANTES, aid);
    if(!anunciante) return {success:false,message:"Anunciante no encontrado"};

    const relaciones = await firestoreQueryEqual(env, COLL_SUSCRIPTOR_ANUNCIANTE, "suscriptor_id", sid, 1000);
    const yaTiene = relaciones.some(r =>
      String(r.anunciante_id || "").trim() === aid && susTruthy_(r.activo)
    );
    if(yaTiene) return {success:false,message:"Ese anuncio ya está asociado a tu cuenta"};
  }

  const existentes = await firestoreQueryEqual(env, COLL_SOLICITUDES_ANUNCIANTE, "suscriptor_id", sid, 1000);
  const duplicada = existentes.find(x =>
    String(x.estado || "").trim().toUpperCase() === "PENDIENTE" &&
    String(x.modo || "").trim().toUpperCase() === modo &&
    (modo === "RECLAMAR"
      ? String(x.anunciante_id || "").trim() === aid
      : susNorm_(x.nombre) === susNorm_(nombre))
  );
  if(duplicada) return {success:false,message:"Ya existe una solicitud pendiente para este anunciante"};

  let solicitudId = "";
  for(let i=0;i<20;i++){
    const candidate = susRandomId_("sol-",10);
    if(!await firestoreGetDocument(env, COLL_SOLICITUDES_ANUNCIANTE, candidate)){
      solicitudId = candidate;
      break;
    }
  }
  if(!solicitudId) return {success:false,message:"No se pudo generar la solicitud"};

  const ahora = new Date().toISOString();
  const doc = {
    solicitud_id:solicitudId,
    tipo:"ANUNCIANTE",
    modo,
    suscriptor_id:sid,
    anunciante_id:modo === "RECLAMAR" ? aid : "",
    nombre,
    ciudad_id:String(p.ciudad_id || p.ciudad || "").trim(),
    estado:"PENDIENTE",
    detalle:{
      actividad:String(p.actividad || "").trim(),
      direccion:String(p.direccion || "").trim(),
      whatsapp:String(p.whatsapp || "").trim(),
      mail:String(p.mail || "").trim(),
      web:String(p.web || "").trim(),
      descripcion:String(p.descripcion || "").trim()
    },
    creado_en:ahora,
    actualizado_en:ahora
  };

  await firestoreSetDocument(env, COLL_SOLICITUDES_ANUNCIANTE, solicitudId, doc);
  return {success:true,solicitud_id:solicitudId,estado:"PENDIENTE",message:"Solicitud enviada. Quedó pendiente de revisión.",source:"firestore"};
}

async function susValidateOwner_(env, p){

async function susDeleteAdvertiserRelations_(env, p){
  const sid = String(p.suscriptor_id || p.id || "").trim();
  const aid = String(p.anunciante_id || p.advertiserId || "").trim();

  if(!sid || !aid){
    return {success:false,message:"Faltan datos"};
  }

  const auth = await susValidateOwner_(env, {
    suscriptor_id:sid,
    anunciante_id:aid
  });

  if(!auth.success || !auth.autorizado){
    return {
      success:false,
      message:"Solo el PROPIETARIO puede eliminar el anunciante"
    };
  }

  const relations = await firestoreListCollection(
    env,
    COLL_SUSCRIPTOR_ANUNCIANTE
  );
  const favorites = await firestoreListCollection(
    env,
    COLL_SUSCRIPTOR_FAVORITOS
  );

  let relDeleted = 0;
  let favDeleted = 0;

  for(const rel of relations){
    if(String(rel.anunciante_id || "").trim() !== aid) continue;
    const id = String(rel.id || "").trim();
    if(!id) continue;
    await firestoreDeleteDocument(env, COLL_SUSCRIPTOR_ANUNCIANTE, id);
    relDeleted++;
  }

  for(const fav of favorites){
    if(String(fav.tipo || "").trim().toUpperCase() !== "ANUNCIANTE") continue;
    if(String(fav.referencia_id || "").trim() !== aid) continue;
    const id = String(fav.id || fav.favorito_id || "").trim();
    if(!id) continue;
    await firestoreDeleteDocument(env, COLL_SUSCRIPTOR_FAVORITOS, id);
    favDeleted++;
  }

  return {
    success:true,
    anunciante_id:aid,
    eliminados:{
      suscriptor_anunciante:relDeleted,
      suscriptor_favoritos:favDeleted
    },
    source:"firestore"
  };
}

async function susParseRequest_(request){
  const url = new URL(request.url);
  const params = {};

  for(const [k,v] of url.searchParams.entries()){
    params[k] = v;
    params[k.toLowerCase()] = v;
  }

  if(request.method === "POST"){
    const contentType = String(
      request.headers.get("Content-Type") || ""
    ).toLowerCase();

    try{
      if(contentType.includes("application/json")){
        const body = await request.json();
        if(body && typeof body === "object"){
          for(const [k,v] of Object.entries(body)){
            params[k] = v;
            params[k.toLowerCase()] = v;
          }
        }
      }else{
        const raw = await request.text();
        const form = new URLSearchParams(raw);
        for(const [k,v] of form.entries()){
          params[k] = v;
          params[k.toLowerCase()] = v;
        }
      }
    }catch(_){}
  }

  return params;
}

async function handleSuscriptoresRequest(request, env){
  // El body de Request sólo puede consumirse una vez.
  // Conservamos una copia exclusivamente para el puente de recuperación por Apps Script.
  const recoveryProxyRequest =
    request.method === "POST"
      ? request.clone()
      : request;

  const p = await susParseRequest_(request);
  const action = susNorm_(p.action || p.accion);

  try{
    if(request.method === "GET"){
      if(action === "perfil" || action === "suscriptor"){
        return jsonResponse(await susGetProfile_(env,p),200);
      }

      if(action === "favoritos"){
        return jsonResponse(await susFavorites_(env,p),200);
      }

      if(action === "es_favorito"){
        return jsonResponse(await susIsFavorite_(env,p),200);
      }

      if(action === "anunciantes_autorizados"){
        return jsonResponse(await susAuthorized_(env,p),200);
      }

      if(action === "validar_propietario_anunciante"){
        return jsonResponse(await susValidateOwner_(env,p),200);
      }

      if(action === "eliminar_relaciones_anunciante"){
        return jsonResponse(await susDeleteAdvertiserRelations_(env,p),200);
      }

      return jsonResponse({
        success:true,
        api:"guia-local-suscriptores-firestore",
        source:"firestore",
        actions:[
          "crear",
          "login",
          "perfil",
          "actualizar_perfil",
          "actualizar_ciudad",
          "cambiar_clave",
          "eliminar_cuenta",
          "favoritos",
          "es_favorito",
          "agregar_favorito",
          "quitar_favorito",
          "anunciantes_autorizados",
          "solicitar_anunciante",
          "validar_propietario_anunciante",
          "eliminar_relaciones_anunciante",
          "solicitar_recuperacion (puente temporal de correo)",
          "restablecer_clave (puente temporal de correo)",
          "solicitar_verificacion (puente temporal de correo)",
          "confirmar_verificacion (puente temporal de correo)"
        ]
      },200);
    }

    if(request.method !== "POST"){
      return jsonResponse({
        success:false,
        message:"Método no permitido"
      },405);
    }

    if(action === "crear"){
      return jsonResponse(await susCreate_(env,p),200);
    }

    if(action === "login"){
      return jsonResponse(await susLogin_(env,p),200);
    }

    if(action === "actualizar_perfil"){
      return jsonResponse(await susUpdateProfile_(env,p),200);
    }

    if(action === "actualizar_ciudad"){
      return jsonResponse(await susUpdateCity_(env,p),200);
    }

    if(action === "cambiar_clave"){
      return jsonResponse(await susChangePassword_(env,p),200);
    }

    if(action === "eliminar_cuenta"){
      return jsonResponse(await susDeleteAccount_(env,p),200);
    }

    if(action === "agregar_favorito"){
      return jsonResponse(await susAddFavorite_(env,p),200);
    }

    if(action === "quitar_favorito"){
      return jsonResponse(await susRemoveFavorite_(env,p),200);
    }

    if(action === "anunciantes_autorizados"){
      return jsonResponse(await susAuthorized_(env,p),200);
    }

    if(action === "solicitar_anunciante"){
      return jsonResponse(await susSolicitarAnunciante_(env,p),200);
    }

    // Recuperación de contraseña:
    // todavía usa Apps Script SOLO como puente de correo/código.
    // El resto del runtime de Suscriptores ya es Firestore.
    if(
      action === "solicitar_recuperacion" ||
      action === "restablecer_clave" ||
      action === "solicitar_verificacion" ||
      action === "confirmar_verificacion"
    ){
      const upstream = await proxySuscriptoresRecoveryToAppsScript_(
        recoveryProxyRequest,
        env
      );

      let upstreamData = null;
      let upstreamText = "";

      try{
        const clone = upstream.clone();
        upstreamText = await clone.text();

        try{
          upstreamData = upstreamText
            ? JSON.parse(upstreamText)
            : null;
        }catch(_){}
      }catch(_){}

      // DIAGNÓSTICO TEMPORAL:
      // si Apps Script devuelve un fallo, mostramos exactamente qué respondió.
      if(
        upstreamData &&
        upstreamData.success === false
      ){
        const detalleError = String(
          upstreamData.error ||
          upstreamData.message ||
          "Sin detalle adicional"
        );

        return jsonResponse({
          success:false,
          message:
            (upstreamData.message ||
             "No se pudo enviar el correo de recuperación.") +
            " | DETALLE: " +
            detalleError,
          error:detalleError,
          diagnostico:{
            origen:"apps_script_recuperacion",
            action,
            status:upstream.status,
            respuesta:upstreamData
          }
        },200);
      }

      // Si el restablecimiento fue exitoso, sincronizamos Firestore.
      if(
        action === "restablecer_clave" &&
        upstreamData &&
        upstreamData.success === true
      ){
        try{
          const mail = susNorm_(p.mail || p.email);
          const claveNueva = String(p.clave_nueva || "").trim();
          const sus = await susFindByMail_(env, mail);

          if(sus && claveNueva){
            const sid = String(
              sus.suscriptor_id || sus.id || ""
            ).trim();

            const next = {
              ...sus,
              id:undefined,
              clave:claveNueva
            };

            await firestoreSetDocument(
              env,
              COLL_SUSCRIPTORES,
              sid,
              next
            );
          }
        }catch(syncErr){
          return jsonResponse({
            success:false,
            message:"La contraseña se cambió, pero falló la sincronización en Firestore.",
            error:String(
              syncErr && syncErr.message
                ? syncErr.message
                : syncErr
            ),
            diagnostico:{
              origen:"worker_sync_firestore",
              action
            }
          },200);
        }
      }

      // Si no pudimos interpretar JSON, devolvemos diagnóstico del contenido bruto.
      if(!upstreamData){
        return jsonResponse({
          success:false,
          message:"Apps Script devolvió una respuesta no válida.",
          error:upstreamText || "Respuesta vacía",
          diagnostico:{
            origen:"apps_script_recuperacion",
            action,
            status:upstream.status
          }
        },200);
      }

      return jsonResponse(upstreamData,200);
    }

    if(action === "reconstruir_indice"){
      return jsonResponse({
        success:true,
        actualizados:0,
        message:"suscriptor_indice ya no se utiliza en Firestore.",
        source:"firestore"
      },200);
    }

    return jsonResponse({
      success:false,
      message:"Acción no reconocida",
      action
    },400);

  }catch(err){
    return jsonResponse({
      success:false,
      error:String(err && err.message ? err.message : err)
    },500);
  }
}

function superadminB64_(input){
  const bytes = typeof input === "string"
    ? new TextEncoder().encode(input)
    : input;

  let binary = "";
  bytes.forEach(b => binary += String.fromCharCode(b));

  return btoa(binary)
    .replace(/\+/g,"-")
    .replace(/\//g,"_")
    .replace(/=+$/g,"");
}

function superadminB64Decode_(s){
  s = String(s || "").replace(/-/g,"+").replace(/_/g,"/");
  while(s.length % 4) s += "=";
  const binary = atob(s);
  const bytes = new Uint8Array(binary.length);
  for(let i=0;i<binary.length;i++) bytes[i] = binary.charCodeAt(i);
  return new TextDecoder().decode(bytes);
}

async function superadminHmac_(env, value){
  const secret = String(env.SERVER_SECRET || "").trim();
  if(!secret) throw new Error("Falta SERVER_SECRET");

  const key = await crypto.subtle.importKey(
    "raw",
    new TextEncoder().encode(secret),
    {name:"HMAC",hash:"SHA-256"},
    false,
    ["sign","verify"]
  );

  const sig = await crypto.subtle.sign(
    "HMAC",
    key,
    new TextEncoder().encode(value)
  );

  return superadminB64_(new Uint8Array(sig));
}

async function superadminIssueToken_(env, suscriptorId){
  const payload = {
    sid:String(suscriptorId || "").trim(),
    exp:Date.now() + 8 * 60 * 60 * 1000
  };

  const body = superadminB64_(JSON.stringify(payload));
  const sig = await superadminHmac_(env, body);
  return body + "." + sig;
}

async function superadminVerifyToken_(env, request){
  const auth = String(request.headers.get("Authorization") || "").trim();
  if(!auth.toLowerCase().startsWith("bearer ")){
    return {ok:false,message:"Sesión de administrador requerida"};
  }

  const token = auth.slice(7).trim();
  const parts = token.split(".");
  if(parts.length !== 2){
    return {ok:false,message:"Sesión inválida"};
  }

  const [body,sig] = parts;
  const expected = await superadminHmac_(env, body);

  if(sig !== expected){
    return {ok:false,message:"Sesión inválida"};
  }

  let payload = null;
  try{
    payload = JSON.parse(superadminB64Decode_(body));
  }catch(_){}

  if(!payload || !payload.sid || !payload.exp){
    return {ok:false,message:"Sesión inválida"};
  }

  if(Date.now() > Number(payload.exp)){
    return {ok:false,message:"Sesión vencida"};
  }

  const sid = String(payload.sid).trim();
  const sus = await firestoreGetDocument(env, COLL_SUSCRIPTORES, sid);
  if(!sus){
    return {ok:false,message:"Suscriptor no encontrado"};
  }

  if(
    sus.activo !== "" &&
    sus.activo !== null &&
    sus.activo !== undefined &&
    !susTruthy_(sus.activo)
  ){
    return {ok:false,message:"Suscriptor no activo"};
  }

  const admin = await superadminPermisoPorSuscriptor_(env, sid);
  if(!admin || !susTruthy_(admin.activo)){
    return {ok:false,message:"No tenés acceso a este panel"};
  }

  const rol = String(admin.rol || "").trim().toUpperCase();
  if(!superadminRolValido_(rol)){
    return {ok:false,message:"Rol administrativo no habilitado"};
  }

  return {
    ok:true,
    admin,
    suscriptor:susPublic_(sus),
    rol,
    sid
  };
}

      .toLowerCase()

  );
}

  }

  await refreshGuideCacheForAdvertiser_(env, aid);
  await gldCacheDelete_("commerce:catalogs:v2");
  return {success:true,anunciante_id:aid};
}

function superadminPuedeAprobar_(auth){
  const rol=String(auth && auth.rol || "").trim().toUpperCase();
  if(["SUPERADMIN_PRINCIPAL","SUPERADMIN"].includes(rol)) return true;
  if(rol==="ADMIN_LOCAL"){
    const p=auth && auth.admin && auth.admin.permisos;
    return !!(p && (p.aprobaciones===true || p.todo===true));
  }
  return false;
}

async function superadminPendientes_(env){
  const [eventos, actividades, solicitudesAnunciante] = await Promise.all([
    firestoreListCollection(env, "eventos"),
    firestoreListCollection(env, "actividades"),
    firestoreListCollection(env, COLL_SOLICITUDES_ANUNCIANTE)
  ]);

  const ev = eventos
    .filter(e => ["PENDIENTE","REVISION"].includes(String(e.estado_moderacion || "").trim().toUpperCase()))
    .map(e => ({
      tipo:"EVENTO", id:String(e.evento_id || e.id || "").trim(),
      nombre:String(e.nombre_evento || e.nombre || "").trim(), ciudad_id:String(e.ciudad_id || "").trim(),
      anunciante_id:String(e.anunciante_id || e.id_anunciante || "").trim(), nivel:String(e.nivel || "").trim(),
      estado:String(e.estado_moderacion || "").trim(), fecha:String(e.fecha_desde || e.fecha_carga || "").trim(),
      motivo:String(e.motivo_revision || "").trim()
    }));

  const ac = actividades
    .filter(a => {
      const estado=String(a.estado || "").trim().toUpperCase();
      const aprobado=truthyValue_(a.aprobado);
      return !aprobado && !["RECHAZADA","RECHAZADO","ELIMINADA","ELIMINADO"].includes(estado);
    })
    .map(a => ({
      tipo:"ACTIVIDAD", id:String(a.actividad_id || a.id || "").trim(), nombre:String(a.nombre || "").trim(),
      ciudad_id:String(a.ciudad_id || "").trim(), anunciante_id:String(a.anunciante_id || "").trim(),
      estado:String(a.estado || "PENDIENTE").trim(), fecha:String(a.creado || a.actualizado || "").trim()
    }));

  const an = solicitudesAnunciante
    .filter(x => String(x.estado || "").trim().toUpperCase() === "PENDIENTE")
    .map(x => ({
      tipo:"ANUNCIANTE", id:String(x.solicitud_id || x.id || "").trim(), modo:String(x.modo || "").trim().toUpperCase(),
      nombre:String(x.nombre || "").trim(), ciudad_id:String(x.ciudad_id || "").trim(),

      estado:"PENDIENTE", fecha:String(x.creado_en || "").trim(), detalle:x.detalle || {}
    }));

  return {
    success:true,
    cantidades:{eventos:ev.length,actividades:ac.length,anunciantes:an.length,total:ev.length+ac.length+an.length},
    eventos:ev, actividades:ac, anunciantes:an
  };
}
async function superadminResolverPendiente_(env, p, auth){
  if(!superadminPuedeAprobar_(auth)){
    return {success:false,message:"No tenés permiso para aprobar contenidos."};
  }

  const tipo=String(p.tipo || "").trim().toUpperCase();
  const id=String(p.id || p.evento_id || p.actividad_id || "").trim();
  const decision=String(p.decision || "").trim().toUpperCase();

  if(!id || !["APROBAR","RECHAZAR"].includes(decision)){
    return {success:false,message:"Falta indicar contenido o decisión."};
  }

  if(tipo==="ANUNCIANTE"){
    const solicitud=await firestoreGetDocument(env,COLL_SOLICITUDES_ANUNCIANTE,id);
    if(!solicitud) return {success:false,message:"Solicitud de anunciante no encontrada."};
    if(String(solicitud.estado || "").trim().toUpperCase() !== "PENDIENTE"){
      return {success:false,message:"La solicitud ya fue resuelta."};
    }

    const ahora=new Date().toISOString();
    const sid=String(solicitud.suscriptor_id || "").trim();
    const modo=String(solicitud.modo || "").trim().toUpperCase();

    if(decision==="RECHAZAR"){
      solicitud.estado="RECHAZADA";
      solicitud.moderado_por=String(auth.sid || "").trim();
      solicitud.moderado_en=ahora;
      solicitud.actualizado_en=ahora;
      await firestoreSetDocument(env,COLL_SOLICITUDES_ANUNCIANTE,id,solicitud);
      return {success:true,tipo,id,decision};
    }

    let aid=String(solicitud.anunciante_id || "").trim();
    if(modo==="CREAR"){
      const nivel=String(p.nivel || "").trim();
      if(!nivel) return {success:false,message:"Seleccioná el nivel del nuevo anunciante."};
      aid=generateAdvertiserId_();
      const detalle=solicitud.detalle || {};
      await firestoreSetDocument(env,COLL_ANUNCIANTES,aid,{
        id:aid,
        nombre:String(solicitud.nombre || "").trim(),
        actividad:String(detalle.actividad || "").trim(),
        descripcion:String(detalle.descripcion || "").trim(),
        direccion:String(detalle.direccion || "").trim(),
        whatsapp:String(detalle.whatsapp || "").trim(),
        mail:String(detalle.mail || "").trim(),
        web:String(detalle.web || "").trim(),
        ciudad_id:String(solicitud.ciudad_id || "").trim(),
        creado_en:ahora, actualizado_en:ahora
      });
      await firestoreSetDocument(env,COLL_ADMIN,aid,{
        id:aid, aprobado:true, nivel, subnivel:0, verificado:false, gold:false,
        funcionalidades:[], creado_en:ahora, actualizado_en:ahora
      });
    }else if(modo==="RECLAMAR"){
      if(!aid || !await firestoreGetDocument(env,COLL_ANUNCIANTES,aid)){
        return {success:false,message:"El anunciante reclamado ya no existe."};
      }
    }else{
      return {success:false,message:"Tipo de solicitud inválido."};
    }

    const relId=`${sid}__${aid}`;
    await firestoreSetDocument(env,COLL_SUSCRIPTOR_ANUNCIANTE,relId,{
      relacion_id:relId, suscriptor_id:sid, anunciante_id:aid, rol:"PROPIETARIO",
      activo:true, fecha_alta:ahora, actualizado:ahora
    });

    solicitud.estado="APROBADA";
    solicitud.anunciante_id=aid;
    solicitud.moderado_por=String(auth.sid || "").trim();
    solicitud.moderado_en=ahora;
    solicitud.actualizado_en=ahora;
    await firestoreSetDocument(env,COLL_SOLICITUDES_ANUNCIANTE,id,solicitud);
    return {success:true,tipo,id,decision,anunciante_id:aid,suscriptor_id:sid};
  }

  if(tipo==="EVENTO"){
    const doc=await firestoreGetDocument(env,"eventos",id);
    if(!doc) return {success:false,message:"Evento no encontrado."};
    const anterior={...doc};

    doc.estado_moderacion=decision==="APROBAR" ? "APROBADO" : "RECHAZADO";
    doc.moderado_por=String(auth.sid || "").trim();
    doc.moderado_en=new Date().toISOString();
    doc.actualizado=new Date().toISOString();

    await firestoreSetDocument(env,"eventos",id,doc);
    await evtPatchPublicCache_(env,anterior,doc,"upsert");
    return {success:true,tipo,id,decision};
  }

  if(tipo==="ACTIVIDAD"){
    const doc=await firestoreGetDocument(env,"actividades",id);
    if(!doc) return {success:false,message:"Actividad no encontrada."};

    doc.aprobado=decision==="APROBAR";
    doc.estado=decision==="APROBAR" ? "ACTIVA" : "RECHAZADA";
    doc.moderado_por=String(auth.sid || "").trim();
    doc.moderado_en=new Date().toISOString();
    doc.actualizado=new Date().toISOString();

    await firestoreSetDocument(env,"actividades",id,doc);
    return {success:true,tipo,id,decision};
  }

  return {success:false,message:"Tipo de contenido no reconocido."};
}

async function handleSuperadminRequest(request, env){
  const p = await susParseRequest_(request);
  const action = susNorm_(p.action || p.accion);

  try{
    if(action === "login"){
      if(request.method !== "POST"){
        return jsonResponse({success:false,message:"Usá POST para iniciar sesión"},405);
      }
      return jsonResponse(await superadminLogin_(env,p),200);
    }

    const auth = await superadminVerifyToken_(env, request);
    if(!auth.ok){
      return jsonResponse({success:false,message:auth.message},401);
    }

    if(action === "resumen"){
      return jsonResponse(await superadminResumen_(env),200);
    }

    if(action === "territorios"){
      return jsonResponse(await superadminTerritorios_(env),200);
    }

    if(action === "guardar_territorio"){
      if(request.method !== "POST") return jsonResponse({success:false,message:"Usá POST para guardar"},405);
      return jsonResponse(await superadminGuardarTerritorio_(env,p),200);
    }

    if(action === "buscar_suscriptores"){
      return jsonResponse(await superadminBuscarSuscriptores_(env,p),200);
    }

    if(action === "buscar_anunciantes"){
      return jsonResponse(await superadminBuscarAnunciantes_(env,p),200);
    }

    if(action === "detalle_suscriptor"){
      return jsonResponse(await superadminDetalleSuscriptor_(env,p),200);
    }

    if(action === "detalle_anunciante"){
      return jsonResponse(await superadminDetalleAnunciante_(env,p),200);
    }

    if(action === "listar_catalogo"){
      return jsonResponse(await superadminListarCatalogo_(env,p),200);
    }

    if(action === "guardar_catalogo"){
      if(request.method !== "POST") return jsonResponse({success:false,message:"Usá POST para guardar"},405);
      return jsonResponse(await superadminGuardarCatalogo_(env,p),200);
    }

    if(action === "actualizar_anunciante_comercial"){
      if(request.method !== "POST") return jsonResponse({success:false,message:"Usá POST para guardar"},405);
      return jsonResponse(await superadminActualizarAnuncianteComercial_(env,p),200);
    }

    if(action === "pendientes"){
      return jsonResponse(await superadminPendientes_(env),200);
    }

    if(action === "resolver_pendiente"){
      if(request.method !== "POST"){
        return jsonResponse({success:false,message:"Usá POST para aprobar o rechazar"},405);
      }
      return jsonResponse(await superadminResolverPendiente_(env,p,auth),200);
    }

    return jsonResponse({
      success:true,
      api:"guia-local-superadmin",
      version:"1",
      mode:"read_only",
      actions:[
        "login",
        "resumen",
        "buscar_suscriptores",
        "buscar_anunciantes",
        "detalle_suscriptor",
        "detalle_anunciante"
      ]
    },200);

  }catch(err){
    return jsonResponse({
      success:false,
      error:String(err && err.message ? err.message : err)
    },500);
  }
}

const COLL_FARMACIAS_CICLOS = "farmacias_ciclos";
const COLL_FARMACIAS_CICLO_SEDES = "farmacias_ciclo_sedes";

function farmText_(v){
  return String(v ?? "").trim();
}

function farmBool_(v){
  return truthyValue_(v);
}

function farmLocalToUtcMs_(fecha, hora){
  const f = farmText_(fecha);
  const h = farmText_(hora) || "00:00";
  const m = f.match(/^(\d{4})-(\d{2})-(\d{2})$/);
  const t = h.match(/^(\d{1,2}):(\d{2})/);
  if(!m || !t) return NaN;

  // Argentina continental: UTC-03:00.
  return Date.UTC(
    Number(m[1]),
    Number(m[2]) - 1,
    Number(m[3]),
    Number(t[1]) + 3,
    Number(t[2]),
    0,
    0
  );
}

function farmUtcMsToLocalParts_(ms){
  // Convertimos UTC a hora Argentina restando 3 horas.
  const d = new Date(ms - 3 * 60 * 60 * 1000);
  return {
    fecha:
      d.getUTCFullYear() + "-" +
      String(d.getUTCMonth()+1).padStart(2,"0") + "-" +
      String(d.getUTCDate()).padStart(2,"0"),
    hora:
      String(d.getUTCHours()).padStart(2,"0") + ":" +
      String(d.getUTCMinutes()).padStart(2,"0")
  };
}

function farmTodayArgentina_(){
  return farmUtcMsToLocalParts_(Date.now()).fecha;
}

async function farmaciasTurnosFirestore_(request, env){
  const url = new URL(request.url);
  const ciudadId = farmText_(url.searchParams.get("ciudad_id"));
  const fecha = farmText_(url.searchParams.get("fecha")) || farmTodayArgentina_();

  if(!ciudadId){
    return jsonResponse({
      success:false,
      error:"Falta ciudad_id"
    },400);
  }

  const cacheKey=`farmacias:turnos:${ciudadId}:${fecha}`;
  const cached=await gldCacheGetJson_(cacheKey);
  if(cached && typeof cached==="object"){
    return jsonResponse(cached,200);
  }

  const ciclos = await firestoreQueryEqual(
    env,
    COLL_FARMACIAS_CICLOS,
    "ciudad_id",
    ciudadId
  );

  const ciclo = ciclos
    .filter(c => farmBool_(c.activo))
    .sort((a,b) =>
      farmText_(b.fecha_inicio).localeCompare(farmText_(a.fecha_inicio))
    )[0];

  if(!ciclo){
    const empty={
      success:true,
      ciudad_id:ciudadId,
      fecha,
      turnos:[]
    };
    await gldCachePutJson_(cacheKey,empty,300);
    return jsonResponse(empty,200);
  }

  const cicloId=farmText_(ciclo.ciclo_id || ciclo.id);

  const relaciones = await firestoreQueryEqual(
    env,
    COLL_FARMACIAS_CICLO_SEDES,
    "ciclo_id",
    cicloId
  );

  const participantes = relaciones
    .filter(r => farmBool_(r.activo))
    .sort((a,b) => Number(a.orden || 0) - Number(b.orden || 0));

  if(!participantes.length){
    const empty={
      success:true,
      ciudad_id:ciudadId,
      fecha,
      turno_id:cicloId,
      turnos:[]
    };
    await gldCachePutJson_(cacheKey,empty,300);
    return jsonResponse(empty,200);
  }

  const sedeIds=[...new Set(
    participantes.map(p=>farmText_(p.sede_id)).filter(Boolean)
  )];

  const sedes=await firestoreGetMany_(
    env,
    COLL_SEDES,
    sedeIds
  );

  const anuncianteIds=[...new Set(
    sedes.map(s=>farmText_(s.anunciante_id)).filter(Boolean)
  )];

  const anunciantes=await firestoreGetMany_(
    env,
    COLL_ANUNCIANTES,
    anuncianteIds
  );

  const inicioMs = farmLocalToUtcMs_(
    ciclo.fecha_inicio,
    ciclo.hora_inicio || "00:00"
  );

  const duracionHoras = Math.max(1, Number(ciclo.duracion_horas || 24));
  const duracionMs = duracionHoras * 60 * 60 * 1000;
  const simultaneas = Math.max(
    1,
    Math.min(
      participantes.length,
      Math.floor(Number(ciclo.farmacias_por_turno || 1))
    )
  );

  const diaInicioMs = farmLocalToUtcMs_(fecha, "00:00");
  const diaFinMs = diaInicioMs + 24 * 60 * 60 * 1000;

  if(!Number.isFinite(inicioMs) || !Number.isFinite(diaInicioMs)){
    return jsonResponse({
      success:false,
      error:"Configuración de ciclo inválida"
    },500);
  }

  let primerPaso = Math.floor((diaInicioMs - inicioMs) / duracionMs) - 1;
  if(primerPaso < 0) primerPaso = 0;

  const pasos = [];
  for(let paso=primerPaso; paso<primerPaso+5; paso++){
    const desdeMs = inicioMs + paso * duracionMs;
    const hastaMs = desdeMs + duracionMs;

    if(hastaMs <= diaInicioMs) continue;
    if(desdeMs >= diaFinMs) break;

    pasos.push({paso,desdeMs,hastaMs});
  }

  const sedeById = new Map(
    sedes.map(s => [farmText_(s.sede_id || s.id), s])
  );
  const anuncianteById = new Map(
    anunciantes.map(a => [farmText_(a.id || a.anunciante_id), a])
  );

  const ahora = Date.now();
  const out = [];

  for(const turno of pasos){
    const inicioIndice = (turno.paso * simultaneas) % participantes.length;

    for(let j=0;j<simultaneas;j++){
      const participante = participantes[(inicioIndice + j) % participantes.length];
      const sedeId = farmText_(participante.sede_id);
      const sede = sedeById.get(sedeId) || {};
      const anuncianteId = farmText_(sede.anunciante_id);
      const anunciante = anuncianteById.get(anuncianteId) || {};

      const desde = farmUtcMsToLocalParts_(turno.desdeMs);
      const hasta = farmUtcMsToLocalParts_(turno.hastaMs);

      out.push({
        ciclo_id:cicloId,
        sede_id:sedeId,
        anunciante_id:anuncianteId,
        orden:Number(participante.orden || 0),
        nombre_ref:farmText_(anunciante.nombre || sede.nombre_ref || sede.nombre || "Farmacia"),
        nombre_sede:farmText_(sede.nombre_sede || sede.nombre || anunciante.nombre || "Farmacia"),
        ciudad_id:farmText_(sede.ciudad_id || ciudadId),
        direccion:farmText_(sede.direccion),
        maps:farmText_(sede.maps),
        telefono:farmText_(sede.telefono || anunciante.telefono),
        telefono2:farmText_(sede.telefono2 || anunciante.telefono2),
        telefono3:farmText_(sede.telefono3 || anunciante.telefono3),
        telefono4:farmText_(sede.telefono4 || anunciante.telefono4),
        telefono5:farmText_(sede.telefono5 || anunciante.telefono5),
        whatsapp:farmText_(sede.whatsapp || anunciante.whatsapp),
        whatsapp2:farmText_(sede.whatsapp2 || anunciante.whatsapp2),
        whatsapp3:farmText_(sede.whatsapp3 || anunciante.whatsapp3),
        whatsapp4:farmText_(sede.whatsapp4 || anunciante.whatsapp4),
        whatsapp5:farmText_(sede.whatsapp5 || anunciante.whatsapp5),
        mail:farmText_(sede.mail || anunciante.mail),
        instagram:farmText_(sede.instagram || anunciante.instagram),
        facebook:farmText_(sede.facebook || anunciante.facebook),
        logo:farmText_(sede.logo || anunciante.logo),
        fecha_desde:desde.fecha,
        hora_desde:desde.hora,
        fecha_hasta:hasta.fecha,
        hora_hasta:hasta.hora,
        observaciones:farmText_(ciclo.observaciones),
        en_turno_ahora: ahora >= turno.desdeMs && ahora < turno.hastaMs
      });
    }
  }

  const result={
    success:true,
    ciudad_id:ciudadId,
    fecha,
    ciclo_id:cicloId,
    turnos:out
  };

  await gldCachePutJson_(cacheKey,result,300);
  return jsonResponse(result,200);
}

function farmFeatureAllowed_(adminDoc){
  const funcs = splitFeatures(
    adminDoc && (
      adminDoc.funcionalidades ||
      adminDoc.funciones ||
      adminDoc.permisos ||
      ""
    )
  ).map(x => normalizeFeatureName(x));

  return funcs.some(x =>
    ["turnosfarma","farmacias","farmacia"].includes(x)
  ) || truthyValue_(adminDoc && adminDoc.turnos_farma);
}

function farmAllowedCities_(adminDoc, ciudades){
  const cfgRoot =
    adminDoc &&
    adminDoc.funcionalidades_config &&
    typeof adminDoc.funcionalidades_config === "object"
      ? adminDoc.funcionalidades_config
      : {};

  const cfg =
    cfgRoot.TURNOS_FARMA ||
    cfgRoot.FARMACIAS ||
    cfgRoot.turnos_farma ||
    {};

  const todas =
    truthyValue_(cfg.todas_ciudades) ||
    farmText_(adminDoc && adminDoc.turnos_farma) === "*";

  const ids = Array.isArray(cfg.ciudades)
    ? cfg.ciudades.map(farmText_).filter(Boolean)
    : farmText_(adminDoc && adminDoc.turnos_farma)
      .split(/[;,|]+/)
      .map(farmText_)
      .filter(v => v && v !== "*");

  if(todas || !ids.length) return ciudades;

  const allowed = new Set(ids);
  return ciudades.filter(c =>
    allowed.has(farmText_(c.ciudad_id || c.id))
  );
}

function farmCategoryLooksLikePharmacy_(categoria){
  const s = [
    categoria && categoria.titulo,
    categoria && categoria.nombre,
    categoria && categoria.categoria
  ].map(farmText_).join(" ").toLowerCase();

  return s.includes("farmacia");
}

function farmAdvertiserCategoryIds_(anunciante){
  const raw =
    anunciante && (
      anunciante.categoria_id ||
      anunciante.categoria_ids ||
      anunciante.categorias ||
      ""
    );

  if(Array.isArray(raw)) return raw.map(farmText_).filter(Boolean);

  return farmText_(raw)
    .split(/[;,|]+/)
    .map(farmText_)
    .filter(Boolean);
}

async function farmPanelDataFirestore_(env, advertiserId){
  const panelCacheKey=`farmacias:panel:${advertiserId}`;
  const cached=await gldCacheGetJson_(panelCacheKey);
  if(cached && typeof cached==="object"){
    return cached;
  }

  const admin = await firestoreGetDocument(env, COLL_ADMIN, advertiserId);

  if(!admin || !farmFeatureAllowed_(admin)){
    return {
      success:false,
      message:"No tenés habilitada la administración de Farmacias de turno."
    };
  }

  const [
    ciudades,
    ciclos,
    relaciones,
    sedes,
    anunciantes,
    categorias
  ] = await Promise.all([
    firestoreListCollection(env, COLL_CIUDADES),
    firestoreListCollection(env, COLL_FARMACIAS_CICLOS),
    firestoreListCollection(env, COLL_FARMACIAS_CICLO_SEDES),
    firestoreListCollection(env, COLL_SEDES),
    firestoreListCollection(env, COLL_ANUNCIANTES),
    firestoreListCollection(env, COLL_CATEGORIAS)
  ]);

  const ciudadesPermitidas = farmAllowedCities_(admin, ciudades)
    .map(c => ({
      ciudad_id:farmText_(c.ciudad_id || c.id),
      ciudad_visible:farmText_(
        c.ciudad_visible ||
        c.nombre ||
        c.ciudad ||
        c.ciudad_id ||
        c.id
      )
    }))
    .filter(c => c.ciudad_id);

  const allowedCityIds = new Set(
    ciudadesPermitidas.map(c => c.ciudad_id)
  );

  const farmaciaCategoryIds = new Set(
    categorias
      .filter(farmCategoryLooksLikePharmacy_)
      .map(c => farmText_(c.id || c.categoria_id))
      .filter(Boolean)
  );

  const anuncianteById = new Map(
    anunciantes.map(a => [farmText_(a.id || a.anunciante_id), a])
  );

  const sedesParticipantes = new Set(
    relaciones.map(r => farmText_(r.sede_id)).filter(Boolean)
  );

  const farmacias = sedes
    .filter(s => {
      const ciudadId = farmText_(s.ciudad_id);
      if(!allowedCityIds.has(ciudadId)) return false;

      const estado = farmText_(s.estado).toLowerCase();
      if(estado && !["activa","activo","true","1","x"].includes(estado)) return false;

      const anuncianteId = farmText_(s.anunciante_id);
      const a = anuncianteById.get(anuncianteId) || {};
      const nombre = farmText_(a.nombre || s.nombre_ref || s.nombre_sede || s.nombre).toLowerCase();

    })

      return {
        sede_id:farmText_(s.sede_id || s.id),
        anunciante_id:anuncianteId,
        nombre_ref:farmText_(a.nombre || s.nombre_ref || "Farmacia"),
        nombre_sede:farmText_(s.nombre_sede || s.nombre || ""),
        ciudad_id:farmText_(s.ciudad_id),
        direccion:farmText_(s.direccion),
        maps:farmText_(s.maps),

        telefono2:farmText_(s.telefono2 || a.telefono2),
        telefono3:farmText_(s.telefono3 || a.telefono3),

    });

function farmRandomId_(){
  return "cic_" +
    crypto.randomUUID().replace(/-/g,"").slice(0,12);
}

async function farmGuardarCicloFirestore_(env, advertiserId, body){
  const admin = await firestoreGetDocument(env, COLL_ADMIN, advertiserId);

  if(!admin || !farmFeatureAllowed_(admin)){
    return {
      success:false,
      message:"No tenés habilitada la administración de Farmacias de turno."
    };
  }

  const payload =
    body && body.payload && typeof body.payload === "object"
      ? body.payload
      : body || {};

  const ciudades = await firestoreListCollection(env, COLL_CIUDADES);
  const permitidas = farmAllowedCities_(admin, ciudades);
  const allowedCityIds = new Set(
    permitidas.map(c => farmText_(c.ciudad_id || c.id))
  );

  let cicloId = farmText_(payload.ciclo_id);
  let actual = null;

  if(cicloId){
    actual = await firestoreGetDocument(
      env,
      COLL_FARMACIAS_CICLOS,
      cicloId
    );
  }

  const merged = {
    ...(actual || {}),
    ...payload
  };

  const ciudadId = farmText_(merged.ciudad_id);

  if(!ciudadId || !allowedCityIds.has(ciudadId)){
    return {
      success:false,
      message:"La ciudad no está habilitada para esta cuenta."
    };
  }

  if(!farmText_(merged.fecha_inicio)){
    return {success:false,message:"Falta fecha de inicio."};
  }

  if(!farmText_(merged.hora_inicio)){
    return {success:false,message:"Falta hora de inicio."};
  }

  const duracion = Math.max(1,Number(merged.duracion_horas || 24));
  const simultaneas = Math.max(
    1,
    Math.floor(Number(merged.farmacias_por_turno || 1))
  );

  if(!cicloId) cicloId = farmRandomId_();

  const doc = {
    ciclo_id:cicloId,
    ciudad_id:ciudadId,
    fecha_inicio:farmText_(merged.fecha_inicio),
    hora_inicio:farmText_(merged.hora_inicio),
    duracion_horas:duracion,
    farmacias_por_turno:simultaneas,
    activo:
      merged.activo === undefined
        ? true
        : truthyValue_(merged.activo),
    observaciones:farmText_(merged.observaciones),
    actualizado:new Date().toISOString()
  };

  await firestoreSetDocument(
    env,
    COLL_FARMACIAS_CICLOS,
    cicloId,
    doc
  );

  if(Array.isArray(payload.participantes)){
    const nuevos = payload.participantes
      .map((p,i) => ({
        ciclo_id:cicloId,
        sede_id:farmText_(p.sede_id),
        orden:Number(p.orden || i+1),
        activo:true,
        actualizado:new Date().toISOString()
      }))
      .filter(p => p.sede_id);

    if(!nuevos.length){
      return {
        success:false,
        message:"Seleccioná al menos una farmacia."
      };
    }

    if(simultaneas > nuevos.length){
      return {
        success:false,
        message:"La cantidad simultánea supera las farmacias seleccionadas."
      };
    }

    const existentesCiclo = await firestoreQueryEqual(
      env,
      COLL_FARMACIAS_CICLO_SEDES,
      "ciclo_id",
      cicloId
    );

    const nuevosIds = new Set(
      nuevos.map(p => cicloId + "__" + p.sede_id)
    );

    for(const r of existentesCiclo){
      const id = farmText_(r.id || (cicloId + "__" + farmText_(r.sede_id)));
      if(id && !nuevosIds.has(id)){
        await firestoreDeleteDocument(
          env,
          COLL_FARMACIAS_CICLO_SEDES,
          id
        );
      }
    }

    for(const p of nuevos){
      const id = cicloId + "__" + p.sede_id;
      await firestoreSetDocument(
        env,
        COLL_FARMACIAS_CICLO_SEDES,
        id,
        p
      );
    }
  }

  await gldCacheDelete_(`farmacias:panel:${advertiserId}`);

  return {
    success:true,
    ciclo_id:cicloId
  };
}

async function handleFarmaciasRequest(request, env){
  const url = new URL(request.url);
  let body = {};
  let action = farmText_(url.searchParams.get("action"));

  if(request.method === "POST"){
    try{
      body = await request.json();
      action = farmText_(body.action || body.accion || action);
    }catch(_){
      body = {};
    }
  }

  if(request.method === "GET" && action === "turnos"){
    try{
      return await farmaciasTurnosFirestore_(request, env);
    }catch(err){
      return jsonResponse({
        success:false,
        error:"Error leyendo Farmacias desde Firestore",
        message:String(err && err.message ? err.message : err)
      },500);
    }
  }

  if(request.method === "POST" && action === "getPanelData"){
    const advertiserId = farmText_(
      body.advertiserId ||
      body.advertiser_id ||
      body.id
    );

    if(!advertiserId){
      return jsonResponse({
        success:false,
        message:"Falta advertiserId"
      },400);
    }

    try{
      return jsonResponse(
        await farmPanelDataFirestore_(env, advertiserId),
        200
      );
    }catch(err){
      return jsonResponse({
        success:false,
        message:String(err && err.message ? err.message : err)
      },500);
    }
  }

  if(request.method === "POST" && action === "guardar_ciclo"){
    const advertiserId = farmText_(
      body.advertiserId ||
      body.advertiser_id ||
      body.id
    );

    if(!advertiserId){
      return jsonResponse({
        success:false,
        message:"Falta advertiserId"
      },400);
    }

    try{
      return jsonResponse(
        await farmGuardarCicloFirestore_(
          env,
          advertiserId,
          body
        ),
        200
      );
    }catch(err){
      return jsonResponse({
        success:false,
        message:String(err && err.message ? err.message : err)
      },500);
    }
  }

  // Compatibilidad temporal sólo para acciones antiguas no migradas.
  return jsonResponse({
    success:false,
    message:"Acción de Farmacias no reconocida."
  },400);
}

const COLL_ACTIVIDADES_MOD = "actividades";
const COLL_ACTIVIDAD_HORARIOS = "actividad_horarios";
const COLL_ACTIVIDADES_CATEGORIAS = "actividades_categorias";

function actText_(v){
  return String(v ?? "").trim();
}

function actBool_(v){
  return truthyValue_(v);
}

function actTodayArgentina_(){
  const d = new Date(Date.now() - 3 * 60 * 60 * 1000);
  return (
    d.getUTCFullYear() + "-" +
    String(d.getUTCMonth()+1).padStart(2,"0") + "-" +
    String(d.getUTCDate()).padStart(2,"0")
  );
}

function actIsPublic_(a){
  if(!a) return false;
  if(!actBool_(a.activo)) return false;
  if(!actBool_(a.aprobado)) return false;

  const estado = actText_(a.estado).toUpperCase();
  if(estado && ["PAUSADA","PAUSADO","INACTIVA","INACTIVO","BORRADOR","ELIMINADA","ELIMINADO"].includes(estado)){
    return false;
  }

  const hoy = actTodayArgentina_();
  const desde = actText_(a.vigente_desde);
  const hasta = actText_(a.vigente_hasta);

  if(desde && desde > hoy) return false;
  if(hasta && hasta < hoy) return false;

  return true;
}

async function actividadesPublicasFirestore_(request, env){
  const url = new URL(request.url);
  const ciudadId = actText_(url.searchParams.get("ciudad_id"));

  if(!ciudadId){
    return jsonResponse({
      success:false,
      message:"Falta ciudad_id"
    },400);
  }

  const horarios = await firestoreQueryEqual(
    env,
    COLL_ACTIVIDAD_HORARIOS,
    "ciudad_id",
    ciudadId
  );

  const horariosActivos = horarios.filter(h=>actBool_(h.activo));
  const actividadIds = [...new Set(
    horariosActivos
      .map(h=>actText_(h.actividad_id))
      .filter(Boolean)
  )];

  const [actividades, categorias] = await Promise.all([
    firestoreGetMany_(env, COLL_ACTIVIDADES_MOD, actividadIds),
    firestoreListCollection(env, COLL_ACTIVIDADES_CATEGORIAS)
  ]);

  const publicas = actividades.filter(actIsPublic_);
  const advertiserIds = [...new Set(
    publicas.map(a=>actText_(a.anunciante_id)).filter(Boolean)
  )];
  const anunciantes = await firestoreGetMany_(
    env,
    COLL_ANUNCIANTES,
    advertiserIds
  );

  const anuncianteById = new Map(
    anunciantes.map(a => [actText_(a.id || a.anunciante_id), a])
  );

  const categoriaByName = new Map(
    categorias.map(c => [
      actText_(c.categoria || c.nombre).toLowerCase(),
      c
    ])
  );

  const horariosPorActividad = new Map();
  horariosActivos.forEach(h => {
    const id = actText_(h.actividad_id);
    if(!id) return;
    if(!horariosPorActividad.has(id)) horariosPorActividad.set(id,[]);
    horariosPorActividad.get(id).push(h);
  });

  const out = publicas
    .filter(a => horariosPorActividad.has(actText_(a.actividad_id || a.id)))
    .map(a => {
      const actividadId = actText_(a.actividad_id || a.id);
      const anuncianteId = actText_(a.anunciante_id);
      const anunciante = anuncianteById.get(anuncianteId) || {};
      const cat = categoriaByName.get(actText_(a.categoria).toLowerCase()) || {};

      const hs = (horariosPorActividad.get(actividadId) || [])
        .slice()
        .sort((x,y) => {
          const dx = actText_(x.dia_semana);
          const dy = actText_(y.dia_semana);
          if(dx !== dy) return dx.localeCompare(dy);
          return actText_(x.hora_desde).localeCompare(actText_(y.hora_desde));
        });

      return {
        ...a,
        actividad_id:actividadId,
        anunciante_id:anuncianteId,
        anunciante_nombre:actText_(
          anunciante.nombre ||
          a.anunciante_nombre
        ),
        logo_anunciante:actText_(
          anunciante.logo ||
          a.logo_anunciante
        ),
        logo_categoria:actText_(
          cat.logo ||
          a.logo_categoria
        ),
        horarios:hs
      };
    })
    .sort((a,b) =>
      actText_(a.nombre).localeCompare(actText_(b.nombre),"es")
    );

  return jsonResponse({
    success:true,
    ciudad_id:ciudadId,
    actividades:out
  },200);
}

function actFeatureAllowed_(adminDoc){
  const funcs = splitFeatures(
    adminDoc && (
      adminDoc.funcionalidades ||
      adminDoc.funciones ||
      adminDoc.permisos ||
      ""
    )
  ).map(x => normalizeFeatureName(x));

  return funcs.includes("actividades") || truthyValue_(adminDoc && adminDoc.actividades);
}

function actCupoMax_(adminDoc){
  const cfgRoot =
    adminDoc &&
    adminDoc.funcionalidades_config &&
    typeof adminDoc.funcionalidades_config === "object"
      ? adminDoc.funcionalidades_config
      : {};

  const cfg =
    cfgRoot.ACTIVIDADES ||
    cfgRoot.actividades ||
    {};

  const candidatos = [
    cfg.cantidad,
    cfg.cupo,
    cfg.max,
    cfg.maximo,
    adminDoc && adminDoc.actividades_cant
  ];

  for(const v of candidatos){
    const n = Number(v);
    if(Number.isFinite(n) && n >= 0) return Math.floor(n);
  }

  return 0;
}

function actAddDays_(isoDate, days){
  const m = actText_(isoDate).match(/^(\d{4})-(\d{2})-(\d{2})$/);
  if(!m) return "";
  const d = new Date(Date.UTC(Number(m[1]),Number(m[2])-1,Number(m[3])));
  d.setUTCDate(d.getUTCDate()+Number(days||0));
  return (
    d.getUTCFullYear()+"-"+
    String(d.getUTCMonth()+1).padStart(2,"0")+"-"+
    String(d.getUTCDate()).padStart(2,"0")
  );
}

function actNowIso_(){
  return new Date().toISOString();
}

function actRandomId_(prefix){
  return prefix + "-" + crypto.randomUUID().replace(/-/g,"").slice(0,12);
}

function actContactoFromDoc_(a){
  if(a && a.contacto && typeof a.contacto === "object") return a.contacto;
  if(a && a.contacto_json && typeof a.contacto_json === "object") return a.contacto_json;

  const out={};
  [
    "telefono","telefono2","telefono3","telefono4","telefono5",
    "whatsapp","whatsapp2","whatsapp3","whatsapp4","whatsapp5",
    "mail","mail2","mail3","instagram","facebook","youtube",
    "tiktok","linkedin","x"
  ].forEach(k=>{
    const v=actText_(a && a[k]);
    if(v) out[k]=v;
  });
  return out;
}

function actFlatContactFields_(contacto){
  const out={};
  if(!contacto || typeof contacto !== "object") return out;

  [
    "telefono","telefono2","telefono3","telefono4","telefono5",
    "whatsapp","whatsapp2","whatsapp3","whatsapp4","whatsapp5",
    "mail","mail2","mail3","instagram","facebook","youtube",
    "tiktok","linkedin","x"
  ].forEach(k=>{
    if(contacto[k] !== undefined) out[k]=actText_(contacto[k]);
  });

  return out;
}

async function actPanelDataFirestore_(env, advertiserId){
  const admin = await firestoreGetDocument(env, COLL_ADMIN, advertiserId);

  if(!admin || !actFeatureAllowed_(admin)){
    return {
      success:false,
      message:"No tenés habilitado el módulo Actividades."
    };
  }

  const [
    actividades,
    horarios,
    ciudades,
    lugares,
    sedes,
    categorias
  ] = await Promise.all([
    firestoreQueryEqual(env, COLL_ACTIVIDADES_MOD, "anunciante_id", advertiserId),
    Promise.resolve([]),
    firestoreListCollection(env, COLL_CIUDADES),
    firestoreListCollection(env, "lugares"),
    firestoreQueryEqual(env, COLL_SEDES, "anunciante_id", advertiserId),
    firestoreListCollection(env, COLL_ACTIVIDADES_CATEGORIAS)
  ]);

  const horariosPorActividad = new Map();
  await Promise.all(
    actividades.map(async a=>{
      const actividadId=actText_(a.actividad_id || a.id);
      if(!actividadId) return;
      const hs=await firestoreQueryEqual(
        env,
        COLL_ACTIVIDAD_HORARIOS,
        "actividad_id",
        actividadId
      );
      horariosPorActividad.set(
        actividadId,
        hs.filter(h=>actBool_(h.activo))
      );
    })
  );

  const propias = actividades.map(a => {
    const actividadId = actText_(a.actividad_id || a.id);
    return {
      ...a,
      actividad_id:actividadId,
      contacto:actContactoFromDoc_(a),
      horarios:(horariosPorActividad.get(actividadId)||[])
        .sort((x,y) => actText_(x.dia_semana).localeCompare(actText_(y.dia_semana)))
    };
  });

  const max = actCupoMax_(admin);
  const activas = propias.filter(a =>
    actBool_(a.activo) &&
    actText_(a.estado).toUpperCase() !== "PAUSADA"
  ).length;

  const total = propias.length;
  const totalMax = max > 0 ? max * 2 : 0;

  return {
    success:true,
    permisos:{
      actividades:true,
      actividades_cant:max
    },
    cupo:{
      max,
      activas,
      total,
      totalMax
    },
    ciudades:ciudades
      .map(c=>({
        ciudad_id:actText_(c.ciudad_id || c.id),
        ciudad_visible:actText_(c.ciudad_visible || c.nombre || c.ciudad || c.id)
      }))
      .filter(c=>c.ciudad_id),
    lugares:lugares
      .filter(l=>actBool_(l.activo))
      .map(l=>({
        lugar_id:actText_(l.lugar_id || l.id),
        ciudad_id:actText_(l.ciudad_id),
        nombre:actText_(l.nombre),
        direccion:actText_(l.direccion),
        maps:actText_(l.maps),
        tipo:actText_(l.tipo),
        activo:true
      })),
    sedes:sedes
      .filter(s =>
        actText_(s.anunciante_id) === advertiserId &&
        (!actText_(s.estado) || ["ACTIVA","ACTIVO"].includes(actText_(s.estado).toUpperCase()))
      )
      .map(s=>({
        ...s,
        sede_id:actText_(s.sede_id || s.id)
      })),
    categorias:categorias
      .filter(c => c.activo === undefined || actBool_(c.activo))
      .map(c => actText_(c.categoria || c.nombre))
      .filter(Boolean),
    actividades:propias
  };
}

async function actGuardarFirestore_(env, advertiserId, body){
  const admin = await firestoreGetDocument(env, COLL_ADMIN, advertiserId);

  if(!admin || !actFeatureAllowed_(admin)){
    return {success:false,message:"No tenés habilitado el módulo Actividades."};
  }

  const payload =
    body && body.payload && typeof body.payload === "object"
      ? body.payload
      : {};

  const propias = await firestoreQueryEqual(
    env,
    COLL_ACTIVIDADES_MOD,
    "anunciante_id",
    advertiserId
  );

  const max = actCupoMax_(admin);
  const totalMax = max > 0 ? max * 2 : 0;

  let actividadId = actText_(payload.actividad_id);
  let anterior = null;

  if(actividadId){
    anterior = await firestoreGetDocument(env, COLL_ACTIVIDADES_MOD, actividadId);
    if(!anterior || actText_(anterior.anunciante_id) !== advertiserId){
      return {success:false,message:"Actividad no encontrada."};
    }
  }else{
    if(max <= 0) return {success:false,error:"actividades_sin_cupo"};
    if(totalMax > 0 && propias.length >= totalMax){
      return {success:false,error:"actividades_maximo_guardadas"};
    }

    const activas = propias.filter(a =>
      actBool_(a.activo) &&
      actText_(a.estado).toUpperCase() !== "PAUSADA"
    ).length;

    if(activas >= max){
      return {success:false,error:"actividades_cupo_activo_completo"};
    }

    actividadId = actRandomId_("ACT");
  }

  const hoy = actTodayArgentina_();
  const contacto =
    payload.contacto && typeof payload.contacto === "object"
      ? payload.contacto
      : actContactoFromDoc_(anterior || {});

  const doc = {
    ...(anterior || {}),
    actividad_id:actividadId,
    anunciante_id:advertiserId,
    nombre:payload.nombre !== undefined ? actText_(payload.nombre) : actText_(anterior && anterior.nombre),
    categoria:payload.categoria !== undefined ? actText_(payload.categoria) : actText_(anterior && anterior.categoria),
    descripcion:payload.descripcion !== undefined ? actText_(payload.descripcion) : actText_(anterior && anterior.descripcion),
    inscripcion:payload.inscripcion !== undefined ? actText_(payload.inscripcion) : actText_(anterior && anterior.inscripcion),
    web:payload.web !== undefined ? actText_(payload.web) : actText_(anterior && anterior.web),
    contacto_sede_id:payload.contacto_sede_id !== undefined ? actText_(payload.contacto_sede_id) : actText_(anterior && anterior.contacto_sede_id),
    contacto,
    contacto_json:contacto,
    ...actFlatContactFields_(contacto),
    img1:payload.img1 !== undefined ? actText_(payload.img1) : actText_(anterior && anterior.img1),
    img2:payload.img2 !== undefined ? actText_(payload.img2) : actText_(anterior && anterior.img2),
    img3:payload.img3 !== undefined ? actText_(payload.img3) : actText_(anterior && anterior.img3),
    img4:payload.img4 !== undefined ? actText_(payload.img4) : actText_(anterior && anterior.img4),
    img5:payload.img5 !== undefined ? actText_(payload.img5) : actText_(anterior && anterior.img5),
    aprobado:anterior ? actBool_(anterior.aprobado) : false,
    activo:anterior ? actBool_(anterior.activo) : true,
    estado:anterior ? actText_(anterior.estado || "ACTIVA") : "PENDIENTE",
    vigente_desde:anterior ? actText_(anterior.vigente_desde || hoy) : hoy,
    vigente_hasta:anterior ? actText_(anterior.vigente_hasta || actAddDays_(hoy,30)) : actAddDays_(hoy,30),
    creado:anterior ? actText_(anterior.creado || actNowIso_()) : actNowIso_(),
    actualizado:actNowIso_()
  };

  if(!doc.nombre) return {success:false,message:"Falta nombre de la actividad."};
  if(!doc.categoria) return {success:false,message:"Falta categoría."};

  await firestoreSetDocument(env, COLL_ACTIVIDADES_MOD, actividadId, doc);

  if(Array.isArray(payload.horarios)){
    const anterioresHorarios = await firestoreQueryEqual(
      env,
      COLL_ACTIVIDAD_HORARIOS,
      "actividad_id",
      actividadId
    );

    for(const h of anterioresHorarios){
      const hid = actText_(h.actividad_horario_id || h.id);
      if(hid){
        await firestoreDeleteDocument(env, COLL_ACTIVIDAD_HORARIOS, hid);
      }
    }

    for(const h of payload.horarios){
      const hid = actRandomId_("AH");
      const hdoc = {
        actividad_horario_id:hid,
        actividad_id:actividadId,
        ciudad_id:actText_(h.ciudad_id),
        tipo_lugar:actText_(h.tipo_lugar),
        sede_id:actText_(h.sede_id),
        lugar_id:actText_(h.lugar_id),
        lugar_texto:actText_(h.lugar_texto),
        direccion:actText_(h.direccion),
        maps:actText_(h.maps),
        sector:actText_(h.sector),
        responsable:actText_(h.responsable),
        dia_semana:actText_(h.dia_semana),
        hora_desde:actText_(h.hora_desde),
        hora_hasta:actText_(h.hora_hasta),
        observaciones:actText_(h.observaciones),
        activo:h.activo === undefined ? true : actBool_(h.activo),
        actualizado:actNowIso_()
      };

      await firestoreSetDocument(env, COLL_ACTIVIDAD_HORARIOS, hid, hdoc);
    }
  }

  return {success:true,actividad_id:actividadId};
}

async function actOwned_(env, advertiserId, actividadId){
  const doc = await firestoreGetDocument(env, COLL_ACTIVIDADES_MOD, actividadId);
  if(!doc || actText_(doc.anunciante_id) !== advertiserId) return null;
  return doc;
}

async function actSimpleActionFirestore_(env, advertiserId, action, actividadId){
  const admin = await firestoreGetDocument(env, COLL_ADMIN, advertiserId);

  if(!admin || !actFeatureAllowed_(admin)){
    return {success:false,message:"No tenés habilitado el módulo Actividades."};
  }

  const doc = await actOwned_(env, advertiserId, actividadId);
  if(!doc) return {success:false,message:"Actividad no encontrada."};

  if(action === "eliminar"){
    const horarios = await firestoreQueryEqual(
      env,
      COLL_ACTIVIDAD_HORARIOS,
      "actividad_id",
      actividadId
    );
    for(const h of horarios){
      const hid = actText_(h.actividad_horario_id || h.id);
      if(hid) await firestoreDeleteDocument(env, COLL_ACTIVIDAD_HORARIOS, hid);
    }

    await firestoreDeleteDocument(env, COLL_ACTIVIDADES_MOD, actividadId);
    return {success:true};
  }

  const hoy = actTodayArgentina_();

  if(action === "pausar"){
    doc.activo = false;
    doc.estado = "PAUSADA";
  }

  if(action === "reanudar" || action === "renovar"){
    const propias = await firestoreQueryEqual(
      env,
      COLL_ACTIVIDADES_MOD,
      "anunciante_id",
      advertiserId
    );

    const max = actCupoMax_(admin);
    const activas = propias.filter(a =>
      actText_(a.actividad_id || a.id) !== actividadId &&
      actBool_(a.activo) &&
      actText_(a.estado).toUpperCase() !== "PAUSADA"
    ).length;

    if(max <= 0 || activas >= max){
      return {success:false,error:"actividades_cupo_activo_completo"};
    }

    doc.activo = true;
    doc.estado = "ACTIVA";

    if(action === "renovar"){
      doc.vigente_desde = hoy;
      doc.vigente_hasta = actAddDays_(hoy,30);
    }
  }

  doc.actualizado = actNowIso_();

  await firestoreSetDocument(env, COLL_ACTIVIDADES_MOD, actividadId, doc);

  return {success:true};
}

async function handleActividadesFirestoreRequest(request, env){
  const url = new URL(request.url);
  let body = {};
  let action = normalizeAction(url.searchParams.get("action"));

  if(request.method === "POST"){
    try{
      body = await request.json();
      action = normalizeAction(body.action || body.accion || action);
    }catch(_){
      body = {};
    }
  }

  try{
    if(request.method === "GET" && action === "publicas"){
      return await actividadesPublicasFirestore_(request, env);
    }

    if(request.method === "GET" && action === "getpaneldata"){
      const advertiserId = actText_(url.searchParams.get("advertiserId") || url.searchParams.get("advertiser_id"));
      if(!advertiserId){
        return jsonResponse({success:false,message:"Falta advertiserId"},400);
      }
      return jsonResponse(await actPanelDataFirestore_(env, advertiserId),200);
    }

    if(request.method === "POST" && action === "guardar"){
      const advertiserId = actText_(body.advertiserId || body.advertiser_id);
      if(!advertiserId){
        return jsonResponse({success:false,message:"Falta advertiserId"},400);
      }
      return jsonResponse(await actGuardarFirestore_(env, advertiserId, body),200);
    }

    if(
      request.method === "POST" &&
      ["renovar","pausar","reanudar","eliminar"].includes(action)
    ){
      const advertiserId = actText_(body.advertiserId || body.advertiser_id);
      const actividadId = actText_(body.actividad_id);

      if(!advertiserId || !actividadId){
        return jsonResponse({success:false,message:"Faltan datos de la actividad"},400);
      }

      return jsonResponse(
        await actSimpleActionFirestore_(env, advertiserId, action, actividadId),
        200
      );
    }

    return jsonResponse({
      success:false,
      message:"Acción de Actividades no reconocida."
    },400);

  }catch(err){
    return jsonResponse({
      success:false,
      message:String(err && err.message ? err.message : err)
    },500);
  }
}

const COLL_PROMOS = "promos";
const COLL_PROMOS_CATEGORIAS = "promos_categorias";

function promoText_(v){
  return String(v ?? "").trim();
}

function promoBool_(v){
  return truthyValue_(v);
}

function promoFeatureAllowed_(adminDoc){
  const funcs = splitFeatures(
    adminDoc && (
      adminDoc.funcionalidades ||
      adminDoc.funciones ||
      adminDoc.permisos ||
      ""
    )
  ).map(x => normalizeFeatureName(x));

  return funcs.includes("promos") || funcs.includes("promociones") ||
    truthyValue_(adminDoc && (adminDoc.promos || adminDoc.promociones));
}

function promoCupoMax_(adminDoc){
  const root =
    adminDoc &&
    adminDoc.funcionalidades_config &&
    typeof adminDoc.funcionalidades_config === "object"
      ? adminDoc.funcionalidades_config
      : {};

  const cfg =
    root.PROMOS ||
    root.promos ||
    root.PROMOCIONES ||
    root.promociones ||
    {};

  const candidatos = [
    cfg.cantidad,
    cfg.cupo,
    cfg.max,
    cfg.maximo,
    adminDoc && adminDoc.promos_cant,
    adminDoc && adminDoc.promociones_cant
  ];

  for(const v of candidatos){
    const n = Number(v);
    if(Number.isFinite(n) && n >= 0) return Math.floor(n);
  }

  return 0;
}

function promoRandomId_(){
  return "PRO-" + crypto.randomUUID().replace(/-/g,"").slice(0,12);
}

function promoNowIso_(){
  return new Date().toISOString();
}

function promoIsPaused_(p){
  return promoBool_(p && p.pausado);
}

function promoAdvertiserId_(p){
  return promoText_(
    p && (
      p.anunciante_id ||
      p.id_comercio ||
      p.comercio_id ||
      p.id_anunciante
    )
  );
}

function promoForAdvertiser_(rows, advertiserId){
  const aid = promoText_(advertiserId);
  return (rows || []).filter(p => promoAdvertiserId_(p) === aid);
}

function promoCityVisible_(c){
  return promoText_(
    c && (
      c.ciudad_visible ||
      c.nombre ||
      c.ciudad ||
      c.localidad ||
      c.ciudad_id ||
      c.id
    )
  );
}

function promoSedeBelongs_(sede, advertiserId){
  return promoText_(
    sede && (
      sede.anunciante_id ||
      sede.id_comercio ||
      sede.comercio_id ||
      sede.anunciante
    )
  ) === promoText_(advertiserId);
}

function promoEnrichRow_(p, advertiserMap, sedeMap, categoriaMap){
  const aid = promoAdvertiserId_(p);
  const sedeId = promoText_(p && p.sede_id);
  const advertiser = advertiserMap.get(aid) || {};
  const sede = sedeMap.get(sedeId) || {};
  const categoriaNombre = promoText_(p && p.categoria);
  const cat = categoriaMap.get(
    normalizeFeatureName(categoriaNombre)
  ) || {};

  return {
    ...p,
    promo_id:promoText_(p && (p.promo_id || p.id)),
    anunciante_id:aid,
    id_comercio:aid,
    anunciante:promoText_(
      (p && p.anunciante) ||
      advertiser.nombre ||
      advertiser.actividad ||
      aid
    ),
    logo:promoText_(
      (p && p.logo) ||
      advertiser.logo
    ),
    sede_id:sedeId,
    ciudad_id:promoText_(
      (p && p.ciudad_id) ||
      sede.ciudad_id
    ),
    logo_categoria:promoText_(
      (p && p.logo_categoria) ||
      cat.logo
    )
  };
}

async function promoLoadContext_(env, options={}){
  const ciudadId=promoText_(options.ciudad_id);
  const advertiserId=promoText_(options.advertiser_id);

  let promosPromise;
  if(advertiserId){
    promosPromise=firestoreQueryEqual(
      env, COLL_PROMOS, "anunciante_id", advertiserId
    );
  }else if(ciudadId){
    promosPromise=firestoreQueryEqual(
      env, COLL_PROMOS, "ciudad_id", ciudadId
    );
  }else{
    promosPromise=firestoreListCollection(env, COLL_PROMOS);
  }

  let sedesPromise=advertiserId
    ? firestoreQueryEqual(env, COLL_SEDES, "anunciante_id", advertiserId)
    : Promise.resolve([]);

  const [promos, sedesBase, ciudades, categorias] = await Promise.all([
    promosPromise,
    sedesPromise,
    firestoreListCollection(env, COLL_CIUDADES),
    firestoreListCollection(env, COLL_PROMOS_CATEGORIAS)
  ]);

  const advertiserIds=[...new Set(
    promos.map(p=>promoAdvertiserId_(p)).filter(Boolean)
  )];

  const sedeIds=[...new Set(
    promos.map(p=>promoText_(p.sede_id)).filter(Boolean)
  )];

  const [anunciantes, sedesPorId] = await Promise.all([
    firestoreGetMany_(env, COLL_ANUNCIANTES, advertiserIds),
    firestoreGetMany_(env, COLL_SEDES, sedeIds)
  ]);

  const sedesMapCombined=new Map();
  [...sedesBase,...sedesPorId].forEach(s=>{
    const sid=promoText_(s.sede_id || s.id);
    if(sid) sedesMapCombined.set(sid,s);
  });
  const sedes=[...sedesMapCombined.values()];

  const advertiserMap = new Map(
    anunciantes.map(a => [
      promoText_(a.id || a.anunciante_id),
      a
    ])
  );

  const sedeMap = new Map(
    sedes.map(s => [
      promoText_(s.sede_id || s.id),
      s
    ])
  );

  const categoriaMap = new Map(
    categorias.map(c => [
      normalizeFeatureName(
        c.titulo || c.nombre || c.categoria || c.id
      ),
      c
    ])
  );

  return {
    promos,
    anunciantes,
    sedes,
    ciudades,
    categorias,
    advertiserMap,
    sedeMap,
    categoriaMap
  };
}

async function promoPublicFirestore_(request, env){
  const url = new URL(request.url);
  const ciudadId = promoText_(url.searchParams.get("ciudad_id"));
  const ctx = await promoLoadContext_(env,{ciudad_id:ciudadId});

  let rows = ctx.promos.map(p =>
    promoEnrichRow_(
      p,
      ctx.advertiserMap,
      ctx.sedeMap,
      ctx.categoriaMap
    )
  );

  if(ciudadId){
    rows = rows.filter(p =>
      promoText_(p.ciudad_id) === ciudadId
    );
  }

  return jsonResponse({
    success:true,
    ciudad_id:ciudadId,
    promos:rows
  },200);
}

async function promoPanelDataFirestore_(env, advertiserId){
  const aid = promoText_(advertiserId);

  const [
    adminDoc,
    advertiser,
    ctx
  ] = await Promise.all([
    firestoreGetDocument(env, COLL_ADMIN, aid),
    firestoreGetDocument(env, COLL_ANUNCIANTES, aid),
    promoLoadContext_(env,{advertiser_id:aid})
  ]);

  if(!adminDoc){
    return {
      success:false,
      message:"No existe la administración del anunciante."
    };
  }

  if(!promoFeatureAllowed_(adminDoc)){
    return {
      success:false,
      message:"El anunciante no tiene Promos habilitado."
    };
  }

  const cupo = promoCupoMax_(adminDoc);

  const sedes = ctx.sedes
    .filter(s => promoSedeBelongs_(s, aid))
    .map(s => ({
      ...s,
      sede_id:promoText_(s.sede_id || s.id)
    }));

  const ciudadIds = new Set(
    sedes
      .map(s => promoText_(s.ciudad_id))
      .filter(Boolean)
  );

  const ciudades = ctx.ciudades
    .filter(c =>
      !ciudadIds.size ||
      ciudadIds.has(promoText_(c.ciudad_id || c.id))
    )
    .map(c => ({
      ...c,
      ciudad_id:promoText_(c.ciudad_id || c.id),
      ciudad_visible:promoCityVisible_(c)
    }));

  const promos = promoForAdvertiser_(ctx.promos, aid)
    .map(p =>
      promoEnrichRow_(
        p,
        ctx.advertiserMap,
        ctx.sedeMap,
        ctx.categoriaMap
      )
    );

  return {
    success:true,
    advertiser:advertiser || {id:aid},
    sedes,
    ciudades,
    categorias:ctx.categorias,
    promos,
    promos_cant:cupo
  };
}

async function promoGetPromosFirestore_(env, advertiserId){
  const aid = promoText_(advertiserId);
  const ctx = await promoLoadContext_(env,{advertiser_id:aid});

  const promos = promoForAdvertiser_(ctx.promos, aid)
    .map(p =>
      promoEnrichRow_(
        p,
        ctx.advertiserMap,
        ctx.sedeMap,
        ctx.categoriaMap
      )
    );

  return {
    success:true,
    promos
  };
}

async function promoEnsureWriteAllowed_(env, advertiserId){
  const aid = promoText_(advertiserId);
  const adminDoc = await firestoreGetDocument(env, COLL_ADMIN, aid);

  if(!adminDoc){
    throw new Error("No existe la administración del anunciante.");
  }

  if(!promoFeatureAllowed_(adminDoc)){
    throw new Error("El anunciante no tiene Promos habilitado.");
  }

  return {
    adminDoc,
    cupo:promoCupoMax_(adminDoc)
  };
}

async function promoCreateFirestore_(env, advertiserId, payload){
  const aid = promoText_(advertiserId);
  const data = payload && typeof payload === "object" ? payload : {};
  const perm = await promoEnsureWriteAllowed_(env, aid);

  if(!promoText_(data.promo)){
    return {success:false,message:"Falta el texto de la promo."};
  }

  const sedeId = promoText_(data.sede_id);
  if(!sedeId){
    return {success:false,message:"Falta la sede de la promo."};
  }

  const sede = await firestoreGetDocument(env, COLL_SEDES, sedeId);
  if(!sede || !promoSedeBelongs_(sede, aid)){
    return {success:false,message:"La sede no pertenece al anunciante."};
  }

  const propios = await firestoreQueryEqual(
    env,
    COLL_PROMOS,
    "anunciante_id",
    aid
  );
  const activos = propios.filter(p => !promoIsPaused_(p)).length;
  const max = Number(perm.cupo || 0);
  const totalMax = max > 0 ? max * 2 : 0;

  if(max <= 0){
    return {success:false,message:"El anunciante no tiene cupo de Promos."};
  }

  if(activos >= max){
    return {
      success:false,
      message:"Alcanzaste el máximo de promos activas."
    };
  }

  if(propios.length >= totalMax){
    return {
      success:false,
      message:"Alcanzaste el máximo de promos guardadas."
    };
  }

  const [advertiser, categorias] = await Promise.all([
    firestoreGetDocument(env, COLL_ANUNCIANTES, aid),
    firestoreListCollection(env, COLL_PROMOS_CATEGORIAS)
  ]);

  const categoriaNombre = promoText_(data.categoria);
  const cat = categorias.find(c =>
    normalizeFeatureName(
      c.titulo || c.nombre || c.categoria || c.id
    ) === normalizeFeatureName(categoriaNombre)
  ) || {};

  const promoId = promoRandomId_();

  const doc = {
    promo_id:promoId,
    anunciante_id:aid,
    id_comercio:aid,
    sede_id:sedeId,
    ciudad_id:promoText_(data.ciudad_id || sede.ciudad_id),

    aprobado:true,
    pausado:false,

    anunciante:promoText_(
      advertiser && (advertiser.nombre || advertiser.actividad)
    ),
    logo:promoText_(advertiser && advertiser.logo),

    categoria:categoriaNombre,
    logo_categoria:promoText_(cat.logo),

    promo:promoText_(data.promo),
    desde:promoText_(data.desde),
    hasta:promoText_(data.hasta),
    otros:promoText_(data.otros),
    img:promoText_(data.img),

    instagram:promoText_(data.instagram),
    facebook:promoText_(data.facebook),
    youtube:promoText_(data.youtube),
    tiktok:promoText_(data.tiktok),
    linkedin:promoText_(data.linkedin),
    x:promoText_(data.x),

    direccion:promoText_(data.direccion),
    maps:promoText_(data.maps),

    telefono:promoText_(data.telefono),
    telefono1:promoText_(data.telefono1),
    telefono2:promoText_(data.telefono2),
    telefono3:promoText_(data.telefono3),
    telefono4:promoText_(data.telefono4),

    whatsapp:promoText_(data.whatsapp),
    whatsapp2:promoText_(data.whatsapp2),
    whatsapp3:promoText_(data.whatsapp3),

    audit:promoNowIso_()
  };

  await firestoreSetDocument(
    env,
    COLL_PROMOS,
    promoId,
    doc
  );

  return {
    success:true,
    created:true,
    promo_id:promoId
  };
}

async function promoUpdateFirestore_(env, advertiserId, payload){
  const aid = promoText_(advertiserId);
  const data = payload && typeof payload === "object" ? payload : {};
  const promoId = promoText_(data.promo_id);

  if(!promoId){
    return {success:false,message:"Falta promo_id."};
  }

  const perm = await promoEnsureWriteAllowed_(env, aid);
  const current = await firestoreGetDocument(env, COLL_PROMOS, promoId);

  if(!current || promoAdvertiserId_(current) !== aid){
    return {success:false,message:"La promo no pertenece al anunciante."};
  }

  const next = {...current};

  const allowed = [
    "promo","categoria","sede_id","ciudad_id","img",
    "desde","hasta","otros",
    "instagram","facebook","youtube","tiktok","linkedin","x",
    "direccion","maps",
    "telefono","telefono1","telefono2","telefono3","telefono4",
    "whatsapp","whatsapp2","whatsapp3"
  ];

  for(const key of allowed){
    if(Object.prototype.hasOwnProperty.call(data,key)){
      next[key] = promoText_(data[key]);
    }
  }

  if(Object.prototype.hasOwnProperty.call(data,"pausado")){
    const quierePausar = promoBool_(data.pausado);

    if(!quierePausar && promoIsPaused_(current)){
      const propios = await firestoreQueryEqual(
        env,
        COLL_PROMOS,
        "anunciante_id",
        aid
      );
      const activos = propios
        .filter(p =>
          promoText_(p.promo_id || p.id) !== promoId &&
          !promoIsPaused_(p)
        ).length;

      if(activos >= Number(perm.cupo || 0)){
        return {
          success:false,
          message:"Alcanzaste el máximo de promos activas."
        };
      }
    }

    next.pausado = quierePausar;
  }

  if(Object.prototype.hasOwnProperty.call(data,"sede_id")){
    const sedeId = promoText_(data.sede_id);
    const sede = await firestoreGetDocument(env, COLL_SEDES, sedeId);

    if(!sede || !promoSedeBelongs_(sede, aid)){
      return {
        success:false,
        message:"La sede no pertenece al anunciante."
      };
    }

    if(!promoText_(next.ciudad_id)){
      next.ciudad_id = promoText_(sede.ciudad_id);
    }
  }

  if(Object.prototype.hasOwnProperty.call(data,"categoria")){
    const categorias = await firestoreListCollection(
      env,
      COLL_PROMOS_CATEGORIAS
    );

    const cat = categorias.find(c =>
      normalizeFeatureName(
        c.titulo || c.nombre || c.categoria || c.id
      ) === normalizeFeatureName(next.categoria)
    ) || {};

    next.logo_categoria = promoText_(cat.logo);
  }

  next.promo_id = promoId;
  next.anunciante_id = aid;
  next.id_comercio = aid;
  next.audit = promoNowIso_();

  await firestoreSetDocument(
    env,
    COLL_PROMOS,
    promoId,
    next
  );

  return {success:true,promo_id:promoId};
}

async function promoDeleteFirestore_(env, advertiserId, payload){
  const aid = promoText_(advertiserId);
  const data = payload && typeof payload === "object" ? payload : {};
  const promoId = promoText_(data.promo_id);

  if(!promoId){
    return {success:false,message:"Falta promo_id."};
  }

  await promoEnsureWriteAllowed_(env, aid);

  const current = await firestoreGetDocument(env, COLL_PROMOS, promoId);

  if(!current || promoAdvertiserId_(current) !== aid){
    return {success:false,message:"La promo no pertenece al anunciante."};
  }

  await firestoreDeleteDocument(env, COLL_PROMOS, promoId);

  return {success:true,promo_id:promoId};
}

async function handlePromosFirestoreRequest(request, env){
  try{
    const url = new URL(request.url);

    if(request.method === "GET"){
      return await promoPublicFirestore_(request, env);
    }

    if(request.method !== "POST"){
      return jsonResponse({
        success:false,
        message:"Método no permitido."
      },405);
    }

    let body = {};
    try{
      body = await request.json();
    }catch(_){
      body = {};
    }

    const action = promoText_(
      body.action ||
      body.accion ||
      url.searchParams.get("action")
    );

    const advertiserId = promoText_(
      body.advertiserId ||
      body.advertiser_id ||
      body.id
    );

    if(!advertiserId){
      return jsonResponse({
        success:false,
        message:"Falta advertiserId."
      },400);
    }

    if(action === "getPanelData"){
      return jsonResponse(
        await promoPanelDataFirestore_(env, advertiserId),
        200
      );
    }

    if(action === "getPromos"){
      return jsonResponse(
        await promoGetPromosFirestore_(env, advertiserId),
        200
      );
    }

    if(action === "createPromo"){
      return jsonResponse(
        await promoCreateFirestore_(
          env,
          advertiserId,
          body.payload || {}
        ),
        200
      );
    }

    if(action === "updatePromo"){
      return jsonResponse(
        await promoUpdateFirestore_(
          env,
          advertiserId,
          body.payload || {}
        ),
        200
      );
    }

    if(action === "deletePromo"){
      return jsonResponse(
        await promoDeleteFirestore_(
          env,
          advertiserId,
          body.payload || {}
        ),
        200
      );
    }

    return jsonResponse({
      success:false,
      message:"Acción de Promos no reconocida."
    },400);

  }catch(err){
    return jsonResponse({
      success:false,
      message:String(
        err && err.message ? err.message : err
      )
    },500);
  }
}

const COLL_EVENTOS = "eventos";
const COLL_EVENTO_PROGRAMACION = "evento_programacion";
const COLL_EVENTOS_CATEGORIAS = "eventos_categorias";
const COLL_EVENTOS_MODERACION = "eventos_moderacion_palabras";
const COLL_LUGARES_GLOBAL = "lugares";

function evtText_(v){
  return String(v ?? "").trim();
}

function evtNorm_(v){
  return evtText_(v)
    .toLowerCase()
    .normalize("NFD")
    .replace(/[\u0300-\u036f]/g,"")
    .replace(/[\s_-]+/g,"");
}

function evtBool_(v){
  return truthyValue_(v);
}

function evtLevel_(e){
  return evtText_(e && e.nivel).toUpperCase();
}

function evtIsFree_(e){
  return evtLevel_(e) === "FREE";
}

function evtIsVip_(e){
  const nivel=evtLevel_(e);
  return nivel === "VIP" || (nivel && nivel !== "FREE");
}

function evtAdvertiserId_(e){
  return evtText_(
    e && (
      e.anunciante_id ||
      e.id_anunciante ||
      e.advertiser_id
    )
  );
}

function evtFeatureEnabled_(adminDoc, feature){
  const target=evtNorm_(feature);
  const funcs=splitFeatures(
    adminDoc && adminDoc.funcionalidades
  ).map(evtNorm_);

  if(funcs.includes(target)) return true;

  if(target === "eventos"){
    return evtBool_(
      adminDoc && (
        adminDoc.eventos ||
        adminDoc.EVENTOS
      )
    );
  }

  if(target === "eventosfree"){
    return evtBool_(
      adminDoc && (
        adminDoc.eventos_free ||
        adminDoc.EVENTOS_FREE
      )
    );
  }

  return false;
}

function evtCupo_(adminDoc, feature){
  const root=
    adminDoc &&
    adminDoc.funcionalidades_config &&
    typeof adminDoc.funcionalidades_config === "object"
      ? adminDoc.funcionalidades_config
      : {};

  const aliases=
    evtNorm_(feature)==="eventosfree"
      ? ["EVENTOS_FREE","eventos_free","EVENTOSFREE","eventosfree"]
      : ["EVENTOS","eventos"];

  let cfg={};
  for(const key of aliases){
    if(root[key] && typeof root[key] === "object"){
      cfg=root[key];
      break;
    }
  }

  const candidates=[
    cfg.cupo,
    cfg.cantidad,
    cfg.max,
    cfg.maximo
  ];

  if(evtNorm_(feature)==="eventos"){
    candidates.push(adminDoc && adminDoc.eventos_cant);
  }else{
    candidates.push(adminDoc && adminDoc.eventos_free_cant);
  }

  for(const v of candidates){
    const n=Number(v);
    if(Number.isFinite(n) && n>=0) return Math.floor(n);
  }

  return 0;
}

function evtRandomId_(prefix){
  return String(prefix||"evt") + "-" +
    Date.now() + "-" +
    Math.floor(Math.random()*9000+1000);
}

function evtNowIso_(){
  return new Date().toISOString();
}

function evtIsPublished_(e){
  const estado=evtText_(e && e.estado_moderacion).toUpperCase();

  if(
    estado &&
    !["PUBLICADO","APROBADO","OK"].includes(estado)
  ){
    return false;
  }

  return !evtBool_(e && e.pausado);
}

function evtCategoryList_(rows){
  return (rows||[])
    .map(c=>evtText_(c.titulo || c.nombre || c.categoria))
    .filter(Boolean)
    .sort((a,b)=>a.localeCompare(b,"es"));
}

async function evtProgramacionForEvent_(env, eventoId){

    "evento_id",
    evtText_(eventoId)
  );

  return rows
    .filter(r=>r.activo === undefined || evtBool_(r.activo))
    .sort((a,b)=>Number(a.orden||0)-Number(b.orden||0));

    env,

  );

        evtText_(r.evento_programacion_id || r.id)

}

async function evtSaveProgramacion_(env, eventoId, programacion){
  await evtDeleteProgramacion_(env,eventoId);

  const rows=Array.isArray(programacion) ? programacion : [];

  let orden=0;
  for(const raw of rows){
    orden++;
    const pid=evtText_(raw.evento_programacion_id) || evtRandomId_("evp");

    await firestoreSetDocument(
      env,
      COLL_EVENTO_PROGRAMACION,
      pid,
      {
        evento_programacion_id:pid,
        evento_id:evtText_(eventoId),
        fecha:evtText_(raw.fecha),
        hora_desde:evtText_(raw.hora_desde),
        hora_hasta:evtText_(raw.hora_hasta),
        ciudad_id:evtText_(raw.ciudad_id),
        tipo_lugar:evtText_(raw.tipo_lugar),
        sede_id:evtText_(raw.sede_id),
        lugar_id:evtText_(raw.lugar_id),
        lugar_texto:evtText_(raw.lugar_texto),
        direccion:evtText_(raw.direccion),
        maps:evtText_(raw.maps),
        actividad:evtText_(raw.actividad),
        detalle:evtText_(raw.detalle),
        observaciones:evtText_(raw.observaciones),
        activo:raw.activo === false ? false : true,
        orden:Number(raw.orden || orden),
        actualizado:evtNowIso_()
      }
    );
  }
}

async function evtModerateFree_(env, payload){
  const searchable=[
    payload.nombre_evento,
    payload.categoria,
    payload.lugar,
    payload.descripcion,
    payload.organizador
  ]
    .map(evtText_)
    .join(" ")
    .toLowerCase()
    .normalize("NFD")
    .replace(/[\u0300-\u036f]/g,"");

  if(!searchable){
    return {
      nivel_riesgo:"BAJO",
      estado_moderacion:"PUBLICADO",
      motivo_revision:""
    };
  }

  const key="events:moderation:v1";
  let reglas=await gldCacheGetJson_(key);

  if(!Array.isArray(reglas)){
    reglas=await firestoreListCollection(
      env,
      COLL_EVENTOS_MODERACION
    );
    await gldCachePutJson_(key,reglas,3600);
  }

  let maxRisk="BAJO";
  let motivo="";

  const weight={BAJO:1,MEDIO:2,ALTO:3};

  for(const r of reglas){
    if(r.activo !== undefined && !evtBool_(r.activo)) continue;

    const termino=evtText_(r.termino)
      .toLowerCase()
      .normalize("NFD")
      .replace(/[\u0300-\u036f]/g,"");

    if(!termino) continue;

    if(searchable.includes(termino)){
      const riesgo=evtText_(r.nivel_riesgo).toUpperCase() || "MEDIO";
      if((weight[riesgo]||2) > (weight[maxRisk]||1)){
        maxRisk=riesgo;
        motivo=evtText_(r.motivo);
      }
    }
  }

  return {
    nivel_riesgo:maxRisk,
    estado_moderacion:maxRisk==="ALTO" ? "REVISION" : "PUBLICADO",
    motivo_revision:motivo
  };
}

  const freeEvents=ownEvents.filter(evtIsFree_);

async function evtVipCreate_(env, advertiserId, payload){
  const aid=evtText_(advertiserId);
  const data=payload && typeof payload==="object" ? payload : {};

  const perm=await evtEnsureVipAllowed_(env,aid);

  const own=await firestoreQueryEqual(
    env,
    COLL_EVENTOS,
    "anunciante_id",
    aid
  );

  const vip=own.filter(evtIsVip_);
  const activos=vip.filter(e=>!evtBool_(e.pausado)).length;
  const max=Number(perm.cupo||0);
  const totalMax=max>0 ? max*2 : 0;

  if(max<=0){
    return {success:false,message:"El anunciante no tiene cupo de Eventos."};
  }

  if(activos>=max){
    return {success:false,message:"Alcanzaste el máximo de eventos activos."};
  }

  if(vip.length>=totalMax){
    return {success:false,message:"Alcanzaste el máximo de eventos guardados."};
  }

  const advertiser=await firestoreGetDocument(
    env,
    COLL_ANUNCIANTES,
    aid
  );

  const eventoId=evtRandomId_("vip");
  const now=evtNowIso_();

  const doc={
    ...data,
    nivel:"VIP",
    estado_moderacion:"PENDIENTE",
    evento_id:eventoId,
    suscriptor_id:evtText_(data.suscriptor_id),
    id_anunciante:aid,
    anunciante_id:aid,
    acepta_responsabilidad:true,
    fecha_aceptacion:now,
    fecha_carga:now,
    actualizado:now,
    pausado:false,
    organizador:evtText_(
      data.organizador ||
      (advertiser && advertiser.nombre)
    ),
    logo_organizador:evtText_(
      data.logo_organizador ||
      (advertiser && advertiser.logo)
    )
  };

  delete doc.programacion;
  delete doc.__partner_error;
  delete doc.__programacion_error;

  await firestoreSetDocument(
    env,
    COLL_EVENTOS,
    eventoId,
    doc
  );

  await evtSaveProgramacion_(
    env,
    eventoId,
    data.programacion
  );

  await evtPatchPublicCache_(env,null,doc,"upsert");

  return {
    success:true,
    created:true,
    evento_id:eventoId
  };
}

async function evtVipUpdate_(env, advertiserId, payload){
  const aid=evtText_(advertiserId);
  const data=payload && typeof payload==="object" ? payload : {};
  const eventoId=evtText_(data.evento_id);

  if(!eventoId){
    return {success:false,message:"Falta evento_id."};
  }

  await evtEnsureVipAllowed_(env,aid);

  const current=await firestoreGetDocument(
    env,
    COLL_EVENTOS,
    eventoId
  );

  if(!current || evtAdvertiserId_(current)!==aid || !evtIsVip_(current)){
    return {success:false,message:"El evento no pertenece al anunciante."};
  }

  const next={...current};

  for(const [k,v] of Object.entries(data)){
    if([
      "evento_id",
      "__partner_error",
      "__programacion_error",
      "programacion"
    ].includes(k)) continue;

    next[k]=v;
  }

  next.evento_id=eventoId;
  next.id_anunciante=aid;
  next.anunciante_id=aid;
  next.nivel="VIP";
  next.actualizado=evtNowIso_();

  await firestoreSetDocument(
    env,
    COLL_EVENTOS,
    eventoId,
    next
  );

  if(Object.prototype.hasOwnProperty.call(data,"programacion")){
    await evtSaveProgramacion_(
      env,
      eventoId,
      data.programacion
    );
  }

  await evtPatchPublicCache_(env,current,next,"upsert");

  return {
    success:true,
    updated:true,
    evento_id:eventoId
  };
}

async function evtVipPause_(env, advertiserId, payload){
  const aid=evtText_(advertiserId);
  const eventoId=evtText_(payload && payload.evento_id);

  if(!eventoId){
    return {success:false,message:"Falta evento_id."};
  }

  const perm=await evtEnsureVipAllowed_(env,aid);
  const current=await firestoreGetDocument(
    env,
    COLL_EVENTOS,
    eventoId
  );

  if(!current || evtAdvertiserId_(current)!==aid || !evtIsVip_(current)){
    return {success:false,message:"El evento no pertenece al anunciante."};
  }

  const pause=evtBool_(
    payload && (
      payload.pause !== undefined
        ? payload.pause
        : payload.pausado
    )
  );

  if(!pause && evtBool_(current.pausado)){
    const own=await firestoreQueryEqual(
      env,
      COLL_EVENTOS,
      "anunciante_id",
      aid
    );

    const activos=own
      .filter(evtIsVip_)
      .filter(e=>
        evtText_(e.evento_id||e.id)!==eventoId &&
        !evtBool_(e.pausado)
      ).length;

    if(activos>=Number(perm.cupo||0)){
      return {
        success:false,
        message:"Alcanzaste el máximo de eventos activos."
      };
    }
  }

  current.pausado=pause;
  current.actualizado=evtNowIso_();

  await firestoreSetDocument(
    env,
    COLL_EVENTOS,
    eventoId,
    current
  );

  await evtPatchPublicCache_(env,current,current,"upsert");

  return {success:true,evento_id:eventoId};
}

async function evtFreeSimilar_(env, payload){
  const ciudadId=evtText_(payload.ciudad_id);
  if(!ciudadId) return [];

  const rows=await firestoreQueryEqual(
    env,
    COLL_EVENTOS,
    "ciudad_id",
    ciudadId
  );

  const wantedName=evtNorm_(payload.nombre_evento);
  const wantedPlace=evtNorm_(payload.lugar);
  const wantedDate=evtText_(payload.fecha_desde);

  return rows
    .filter(evtIsFree_)
    .filter(e=>{
      const sameName=
        wantedName &&
        evtNorm_(e.nombre_evento)===wantedName;

      const samePlace=
        wantedPlace &&
        evtNorm_(e.lugar)===wantedPlace;

      const sameDate=
        wantedDate &&
        evtText_(e.fecha_desde)===wantedDate;

      return sameDate && (sameName || samePlace);
    })
    .slice(0,5);
}

async function evtFreeUpdate_(env, advertiserId, payload){

  if(!eventoId){
    return {success:false,message:"Falta evento_id."};
  }

  await evtEnsureFreeAllowed_(env,aid);

  const current=await firestoreGetDocument(
    env,
    COLL_EVENTOS,
    eventoId
  );

  if(!current || evtAdvertiserId_(current)!==aid || !evtIsFree_(current)){
    return {success:false,message:"El evento free no pertenece al anunciante."};
  }

  const next={...current,...data};
  const moderation=await evtModerateFree_(env,next);

  next.evento_id=eventoId;
  next.id_anunciante=aid;
  next.anunciante_id=aid;
  next.nivel="FREE";
  next.nivel_riesgo=moderation.nivel_riesgo;
  next.estado_moderacion=moderation.estado_moderacion;
  next.motivo_revision=moderation.motivo_revision;
  next.actualizado=evtNowIso_();

  await firestoreSetDocument(
    env,
    COLL_EVENTOS,
    eventoId,
    next
  );

  await evtPatchPublicCache_(env,current,next,"upsert");

  return {
    success:true,
    updated:true,
    evento_id:eventoId
  };
}

async function evtPatchPublicCache_(env, previous, next, mode="upsert"){
  const prevCity=evtText_(previous && previous.ciudad_id);
  const nextCity=evtText_(next && next.ciudad_id);
  const prevId=evtText_(previous && (previous.evento_id || previous.id));
  const nextId=evtText_(next && (next.evento_id || next.id));
  const id=nextId || prevId;

  if(prevCity && (!nextCity || prevCity!==nextCity)){
    await patchCachedArrayItem_(
      "events:public:"+prevCity,
      prevId,
      null,
      {idField:"evento_id",mode:"delete",ttl:600}
    );
  }

  if(nextCity && id){
    await patchCachedArrayItem_(
      "events:public:"+nextCity,
      id,
      next,
      {idField:"evento_id",mode:mode,ttl:600}
    );
  }
}

}

async function handleEventsFirestoreRequest(request, env){
  try{
    const url=new URL(request.url);

    if(request.method==="GET"){
      const action=evtText_(
        url.searchParams.get("action") ||
        "getevents"
      );

      if(action==="getevents"){
        const events=await evtPublicEvents_(
          env,
          url.searchParams.get("ciudad_id")
        );

        return jsonResponse({
          success:true,
          events,
          data:events
        },200);
      }

      if(action==="getcities"){
        const catalogs=await evtCachedCatalogs_(env);

        return jsonResponse({
          success:true,
          cities:catalogs.ciudades,
          ciudades:catalogs.ciudades
        },200);
      }

      if(action==="getPlaces"){
        const catalogs=await evtCachedCatalogs_(env);
        const sedes=await evtCachedVenueSedes_(env);
        const ciudadId=evtText_(url.searchParams.get("ciudad_id"));

        const lugares=[...catalogs.lugares,...sedes]
          .filter(x=>!ciudadId || evtText_(x.ciudad_id)===ciudadId);

        return jsonResponse({
          success:true,
          lugares
        },200);
      }

      return jsonResponse({
        success:false,
        message:"Acción GET de Eventos no reconocida."
      },400);
    }

    if(request.method!=="POST"){
      return jsonResponse({
        success:false,
        message:"Método no permitido."
      },405);
    }

    let body={};
    try{
      body=await request.json();
    }catch(_){
      body={};
    }

    const action=evtText_(body.action || body.accion);
    const advertiserId=evtText_(
      body.advertiserId ||
      body.advertiser_id ||
      body.id
    );
    const payload=
      body.payload && typeof body.payload==="object"
        ? body.payload
        : {};

    if(!advertiserId){
      return jsonResponse({
        success:false,
        message:"Falta advertiserId."
      },400);
    }

    if(action==="getPanelData"){
      return jsonResponse(
        await evtPanelData_(env,advertiserId),
        200
      );
    }

    if(action==="getVipEvents"){
      return jsonResponse(
        await evtGetVipEvents_(env,advertiserId),
        200
      );
    }

    if(action==="createVipEvent"){
      return jsonResponse(
        await evtVipCreate_(env,advertiserId,payload),
        200
      );
    }

    if(action==="updateVipEvent"){
      return jsonResponse(
        await evtVipUpdate_(env,advertiserId,payload),
        200
      );
    }

    if(action==="pauseVipEvent"){
      return jsonResponse(
        await evtVipPause_(env,advertiserId,payload),
        200
      );
    }

    if(action==="deleteVipEvent"){
      return jsonResponse(
        await evtVipDelete_(env,advertiserId,payload),
        200
      );
    }

    if(action==="createFreeEvent"){
      return jsonResponse(
        await evtFreeCreate_(env,advertiserId,payload),
        200
      );
    }

    if(action==="updateFreeEvent"){
      return jsonResponse(
        await evtFreeUpdate_(env,advertiserId,payload),
        200
      );
    }

    if(action==="pauseFreeEvent"){
      return jsonResponse(
        await evtFreePause_(env,advertiserId,payload),
        200
      );
    }

    if(action==="deleteFreeEvent"){
      return jsonResponse(
        await evtFreeDelete_(env,advertiserId,payload),
        200
      );
    }

    return jsonResponse({
      success:false,
      message:"Acción de Eventos no reconocida."
    },400);

  }catch(err){
    return jsonResponse({
      success:false,
      message:String(
        err && err.message ? err.message : err
      )
    },500);
  }
}

const COLL_PUBLICIDADES = "publicidades";
const COLL_PUBLICIDAD_MEDIA = "publicidad_media";
const COLL_PUBLICIDAD_SEGMENTACION = "publicidad_segmentacion";
const COLL_PUBLICIDAD_UBICACIONES = "publicidad_ubicaciones";
const COLL_PUBLICIDAD_PRIORIDADES = "publicidad_prioridades";
const COLL_PUBLICIDAD_CATEGORIAS = "publicidad_categorias";
const COLL_PUBLICIDAD_CAMBIOS = "publicidad_cambios_activos";

function pubText_(v){
  return String(v ?? "").trim();
}

function pubNorm_(v){
  return pubText_(v)
    .toLowerCase()
    .normalize("NFD")
    .replace(/[\u0300-\u036f]/g,"")
    .replace(/[\s_-]+/g,"");
}

function pubBool_(v){
  return truthyValue_(v);
}

function pubNowIso_(){
  return new Date().toISOString();
}

function pubDateKeyArgentina_(){
  const parts=new Intl.DateTimeFormat("en-CA",{
    timeZone:"America/Argentina/Buenos_Aires",
    year:"numeric",
    month:"2-digit",
    day:"2-digit"
  }).formatToParts(new Date());

  const map={};
  for(const p of parts) map[p.type]=p.value;

  return `${map.year}-${map.month}-${map.day}`;
}

function pubRandomId_(prefix){
  return String(prefix||"pub") + "-" +
    Date.now() + "-" +
    Math.floor(Math.random()*9000+1000);
}

function pubFeatureEnabled_(admin){
  const funcs=splitFeatures(
    admin && admin.funcionalidades
  ).map(pubNorm_);

  if(funcs.includes("publicidad")) return true;

  return pubBool_(
    admin && (
      admin.publicidad ||
      admin.PUBLICIDAD
    )
  );
}

function pubConfig_(admin){
  const root=
    admin &&
    admin.funcionalidades_config &&
    typeof admin.funcionalidades_config === "object"
      ? admin.funcionalidades_config
      : {};

  const cfg=
    root.PUBLICIDAD &&
    typeof root.PUBLICIDAD === "object"
      ? root.PUBLICIDAD
      : {};

  return {
    activas_max:Number(cfg.activas_max || 0),
    guardadas_max:Number(cfg.guardadas_max || 0),
    ciudades_max:Number(cfg.ciudades_max || 0),
    categorias_max:Number(cfg.categorias_max || 0),
    tags_max:Number(cfg.tags_max || 0),
    cruces_max:Number(cfg.cruces_max || 0),
    permite_imagen:pubBool_(cfg.permite_imagen),
    permite_carrusel:pubBool_(cfg.permite_carrusel),
    permite_short:pubBool_(cfg.permite_short),
    prioridad_id:pubText_(cfg.prioridad_id).toUpperCase(),
    cambios_activos_por_dia_max:Number(cfg.cambios_activos_por_dia_max || 0)
  };
}

async function pubCatalogs_(env){
  const key="publicidad:catalogos:v2";
  const hit=await gldCacheGetJson_(key);
  if(hit && typeof hit==="object") return hit;

  const [ciudades,categorias,ubicaciones,prioridades]=await Promise.all([
    firestoreListCollection(env,COLL_CIUDADES),
    firestoreListCollection(env,COLL_PUBLICIDAD_CATEGORIAS),
    firestoreListCollection(env,COLL_PUBLICIDAD_UBICACIONES),
    firestoreListCollection(env,COLL_PUBLICIDAD_PRIORIDADES)
  ]);

  const out={
    ciudades:ciudades.map(c=>({
      ...c,
      ciudad_id:pubText_(c.ciudad_id || c.id),
      ciudad_visible:pubText_(
        c.ciudad_visible ||
        c.nombre ||
        c.ciudad ||
        c.ciudad_id ||
        c.id
      )
    })),
    categorias:categorias.filter(c=>
      c.activo===undefined || pubBool_(c.activo)
    ),
    ubicaciones:ubicaciones.filter(u=>
      u.activo===undefined || pubBool_(u.activo)
    ),
    prioridades:prioridades.filter(p=>
      p.activo===undefined || pubBool_(p.activo)
    ).sort((a,b)=>Number(a.orden||0)-Number(b.orden||0))
  };

  await gldCachePutJson_(key,out,3600);
  return out;
}

async function pubMediaFor_(env, publicidadId){
  const rows=await firestoreQueryEqual(
    env,
    COLL_PUBLICIDAD_MEDIA,
    "publicidad_id",
    pubText_(publicidadId)
  );

  return rows
    .filter(r=>r.activo===undefined || pubBool_(r.activo))
    .sort((a,b)=>Number(a.orden||0)-Number(b.orden||0));
}

async function pubSegmentacionFor_(env, publicidadId){
  const rows=await firestoreQueryEqual(
    env,
    COLL_PUBLICIDAD_SEGMENTACION,
    "publicidad_id",
    pubText_(publicidadId)
  );

  return rows
    .filter(r=>r.activo===undefined || pubBool_(r.activo));
}

async function pubEnrichMany_(env, publicidades){
  const rows=Array.isArray(publicidades)?publicidades:[];

  await Promise.all(rows.map(async p=>{
    const id=pubText_(p.publicidad_id || p.id);
    const [media,segmentacion]=await Promise.all([
      pubMediaFor_(env,id),
      pubSegmentacionFor_(env,id)
    ]);
    p.media=media;
    p.segmentacion=segmentacion;
  }));

  return rows;
}

function pubIsWithinValidity_(p){
  const now=Date.now();

  const parse=v=>{
    const s=pubText_(v);
    if(!s) return null;
    const d=new Date(s);
    return Number.isFinite(d.getTime()) ? d.getTime() : null;
  };

  const desde=parse(p.vigente_desde);
  const hasta=parse(p.vigente_hasta);

  if(desde!==null && now<desde) return false;
  if(hasta!==null && now>hasta) return false;

  return true;
}

function pubIsPublic_(p){
  return (
    pubText_(p.estado).toUpperCase()==="ACTIVA" &&
    pubBool_(p.aprobado) &&
    pubIsWithinValidity_(p)
  );
}

function pubModuleToUbicacion_(catalogs, modulo){
  const wanted=pubNorm_(modulo);
  return (catalogs.ubicaciones||[])
    .filter(u=>pubNorm_(u.modulo)===wanted)
    .map(u=>pubText_(u.ubicacion_id))
    .filter(Boolean);
}

async function pubPublicas_(env, url){
  const ciudadId=pubText_(url.searchParams.get("ciudad_id"));
  const modulo=pubText_(url.searchParams.get("modulo")).toUpperCase();
  const categoriaId=pubText_(url.searchParams.get("categoria_id"));

  if(!ciudadId || !modulo){
    return {
      success:false,
      message:"Faltan ciudad_id o modulo.",
      publicidades:[]
    };
  }

  const cacheKey=[
    "publicidad:publicas:v3",
    ciudadId,
    modulo,
    categoriaId || "*"
  ].join(":");

  const hit=await gldCacheGetJson_(cacheKey);
  if(hit && typeof hit==="object") return hit;

  const catalogs=await pubCatalogs_(env);
  const ubicacionesPermitidas=new Set(
    pubModuleToUbicacion_(catalogs,modulo)
  );

  if(!ubicacionesPermitidas.size){
    const empty={success:true,publicidades:[]};
    await gldCachePutJson_(cacheKey,empty,300);
    return empty;
  }

  // Consulta principal por ciudad: evita leer toda la segmentación.
  const segCiudad=await firestoreQueryEqual(
    env,
    COLL_PUBLICIDAD_SEGMENTACION,
    "ciudad_id",
    ciudadId
  );

  const segFiltrada=segCiudad.filter(s=>{
    if(s.activo!==undefined && !pubBool_(s.activo)) return false;

    const ubic=pubText_(s.ubicacion_id);
    if(!ubicacionesPermitidas.has(ubic)) return false;

    if(categoriaId){
      return pubText_(s.categoria_id)===categoriaId;
    }

    return true;
  });

  const publicidadIds=[...new Set(
    segFiltrada
      .map(s=>pubText_(s.publicidad_id))
      .filter(Boolean)
  )];

  if(!publicidadIds.length){
    const empty={success:true,publicidades:[]};
    await gldCachePutJson_(cacheKey,empty,300);
    return empty;
  }

  const publicaciones=await firestoreGetMany_(
    env,
    COLL_PUBLICIDADES,
    publicidadIds
  );

  const activas=publicaciones.filter(pubIsPublic_);

  await pubEnrichMany_(env,activas);

  // Dejamos únicamente la segmentación que aplica al contexto consultado.
  for(const p of activas){
    p.segmentacion=(p.segmentacion||[]).filter(s=>{
      if(pubText_(s.ciudad_id)!==ciudadId) return false;
      if(!ubicacionesPermitidas.has(pubText_(s.ubicacion_id))) return false;
      if(categoriaId && pubText_(s.categoria_id)!==categoriaId) return false;
      return true;
    });
  }

  const prioridadOrden=new Map(
    (catalogs.prioridades||[]).map(p=>[
      pubText_(p.prioridad_id).toUpperCase(),
      Number(p.orden||999)
    ])
  );

  activas.sort((a,b)=>{
    const sa=(a.segmentacion||[])[0]||{};
    const sb=(b.segmentacion||[])[0]||{};
    const oa=prioridadOrden.get(pubText_(sa.prioridad_id).toUpperCase()) ?? 999;
    const ob=prioridadOrden.get(pubText_(sb.prioridad_id).toUpperCase()) ?? 999;
    if(oa!==ob) return oa-ob;
    return pubText_(a.publicidad_id).localeCompare(pubText_(b.publicidad_id));
  });

  const ubicCfg=(catalogs.ubicaciones||[]).find(u=>
    ubicacionesPermitidas.has(pubText_(u.ubicacion_id))
  );

  const max=Number(ubicCfg && ubicCfg.max_publicidades_pagina || 0);
  const out=max>0 ? activas.slice(0,max) : activas;

  const result={
    success:true,
    publicidades:out
  };

  await gldCachePutJson_(cacheKey,result,300);
  return result;
}

async function pubCambiosHoy_(env, advertiserId, config){
  const aid=pubText_(advertiserId);
  const dateKey=pubDateKeyArgentina_();
  const docId=aid+"_"+dateKey;

  const doc=await firestoreGetDocument(
    env,
    COLL_PUBLICIDAD_CAMBIOS,
    docId
  );

  const usados=Number(doc && doc.usados || 0);
  const max=Number(config.cambios_activos_por_dia_max || 0);

  return {
    doc_id:docId,
    fecha:dateKey,
    usados,
    max,
    disponibles:max>0 ? Math.max(0,max-usados) : 0
  };
}

async function pubPanelData_(env, advertiserId){
  const aid=pubText_(advertiserId);

  const [admin,publicidades,catalogs]=await Promise.all([
    firestoreGetDocument(env,COLL_ADMIN,aid),
    firestoreQueryEqual(
      env,
      COLL_PUBLICIDADES,
      "anunciante_id",
      aid
    ),
    pubCatalogs_(env)
  ]);

  if(!admin){
    return {
      success:false,
      message:"No existe la administración del anunciante."
    };
  }

  if(!pubFeatureEnabled_(admin)){
    return {
      success:false,
      message:"El anunciante no tiene PUBLICIDAD habilitada."
    };
  }

  const config=pubConfig_(admin);
  await pubEnrichMany_(env,publicidades);

  const activas=publicidades.filter(p=>
    pubText_(p.estado).toUpperCase()==="ACTIVA"
  ).length;

  const cambios=await pubCambiosHoy_(
    env,
    aid,
    config
  );

  return {
    success:true,
    config,
    cupo:{
      activas,
      activas_max:config.activas_max,
      guardadas:publicidades.length,
      guardadas_max:config.guardadas_max
    },
    cambios_activos:cambios,
    ciudades:catalogs.ciudades,
    categorias:catalogs.categorias,
    ubicaciones:catalogs.ubicaciones,
    prioridades:catalogs.prioridades,
    publicidades:publicidades.sort((a,b)=>
      pubText_(a.nombre_interno||a.titulo||a.publicidad_id)
        .localeCompare(
          pubText_(b.nombre_interno||b.titulo||b.publicidad_id),
          "es"
        )
    )
  };
}

function pubValidatePayload_(payload, config){
  const formato=pubText_(payload.formato).toUpperCase();
  const media=Array.isArray(payload.media)?payload.media.filter(x=>pubText_(x.url)):[];
  const ciudades=Array.isArray(payload.ciudades)?[...new Set(payload.ciudades.map(pubText_).filter(Boolean))]:[];
  const categorias=Array.isArray(payload.categorias)?[...new Set(payload.categorias.map(pubText_).filter(Boolean))]:[];

  if(!formato){
    return "formato_requerido";
  }

  if(formato==="IMAGEN" && !config.permite_imagen){
    return "formato_imagen_no_autorizado";
  }

  if(formato==="CARRUSEL" && !config.permite_carrusel){
    return "formato_carrusel_no_autorizado";
  }

  if(formato==="SHORT" && !config.permite_short){
    return "formato_short_no_autorizado";
  }

  if(formato==="IMAGEN" && media.length!==1){
    return "media_imagen_invalida";
  }

  if(formato==="CARRUSEL" && media.length<2){
    return "media_carrusel_invalida";
  }

  if(formato==="SHORT" && media.length!==1){
    return "media_short_invalida";
  }

  if(config.ciudades_max>0 && ciudades.length>config.ciudades_max){
    return "supera_ciudades_max";
  }

  if(config.categorias_max>0 && categorias.length>config.categorias_max){
    return "supera_categorias_max";
  }

  const cruces=ciudades.length*categorias.length;

  if(config.cruces_max>0 && cruces>config.cruces_max){
    return "supera_cruces_max";
  }

  return "";
}

function pubParseCategoriaKey_(value){
  const raw=pubText_(value);
  const idx=raw.indexOf(":");

  if(idx>0){
    return {
      categoria_key:raw,
      ubicacion_id:raw.slice(0,idx),
      categoria_id:raw.slice(idx+1)
    };
  }

  return {
    categoria_key:raw,
    ubicacion_id:"",
    categoria_id:raw
  };
}

async function pubDeleteChildren_(env, publicidadId){
  const [media,segmentacion]=await Promise.all([
    pubMediaFor_(env,publicidadId),
    pubSegmentacionFor_(env,publicidadId)
  ]);

  await Promise.all([
    ...media.map(m=>
      firestoreDeleteDocument(
        env,
        COLL_PUBLICIDAD_MEDIA,
        pubText_(m.media_id || m.id)
      )
    ),
    ...segmentacion.map(s=>
      firestoreDeleteDocument(
        env,
        COLL_PUBLICIDAD_SEGMENTACION,
        pubText_(s.segmentacion_id || s.id)
      )
    )
  ]);
}

async function pubSaveChildren_(env, publicidadId, payload, config){
  await pubDeleteChildren_(env,publicidadId);

  const media=Array.isArray(payload.media)?payload.media.filter(x=>pubText_(x.url)):[];
  let orden=0;

  for(const m of media){
    orden++;
    const id=pubRandomId_("MED");

    await firestoreSetDocument(
      env,
      COLL_PUBLICIDAD_MEDIA,
      id,
      {
        media_id:id,
        publicidad_id:publicidadId,
        tipo_media:pubText_(payload.formato).toUpperCase(),
        url:pubText_(m.url),
        poster:pubText_(m.poster),
        orden,
        activo:true,
        actualizado:pubNowIso_()
      }
    );
  }

  const categorias=Array.isArray(payload.categorias)
    ? [...new Set(payload.categorias.map(pubText_).filter(Boolean))]
    : [];

  const ciudades=Array.isArray(payload.ciudades)
    ? [...new Set(payload.ciudades.map(pubText_).filter(Boolean))]
    : [];

  for(const ciudadId of ciudades){
    for(const categoriaKey of categorias){
      const cat=pubParseCategoriaKey_(categoriaKey);
      if(!cat.ubicacion_id || !cat.categoria_id) continue;

      const id=pubRandomId_("SEG");

      await firestoreSetDocument(
        env,
        COLL_PUBLICIDAD_SEGMENTACION,
        id,
        {
          segmentacion_id:id,
          publicidad_id:publicidadId,
          ciudad_id:ciudadId,
          ubicacion_id:cat.ubicacion_id,
          categoria_id:cat.categoria_id,
          categoria_key:categoriaKey,
          prioridad_id:pubText_(config.prioridad_id).toUpperCase(),
          activo:true,
          creado:pubNowIso_(),
          actualizado:pubNowIso_()
        }
      );
    }
  }
}

async function pubGuardar_(env, advertiserId, payload){
  const aid=pubText_(advertiserId);
  const admin=await firestoreGetDocument(env,COLL_ADMIN,aid);

  if(!admin){
    return {success:false,message:"No existe la administración del anunciante."};
  }

  if(!pubFeatureEnabled_(admin)){
    return {success:false,message:"El anunciante no tiene PUBLICIDAD habilitada."};
  }

  const config=pubConfig_(admin);
  const data=payload && typeof payload==="object" ? payload : {};
  const publicidadId=pubText_(data.publicidad_id);

  let current=null;
  let merged={};

  if(publicidadId){
    current=await firestoreGetDocument(
      env,
      COLL_PUBLICIDADES,
      publicidadId
    );

    if(
      !current ||
      pubText_(current.anunciante_id)!==aid
    ){
      return {
        success:false,
        message:"La publicidad no pertenece al anunciante."
      };
    }

    merged={...current,...data};
  }else{
    const propias=await firestoreQueryEqual(
      env,
      COLL_PUBLICIDADES,
      "anunciante_id",
      aid
    );

    if(
      config.guardadas_max>0 &&
      propias.length>=config.guardadas_max
    ){
      return {
        success:false,
        message:"publicidades_guardadas_completas"
      };
    }

    merged={...data};
  }

  // En edición el frontend envía sólo los campos modificados.
  // Media y segmentación viven en colecciones separadas, por lo que
  // hay que completar el payload con los valores actuales antes de validar.
  /** @type {any} */
  let validationPayload={...merged};

  if(current){
    const [existingMedia,existingSeg]=await Promise.all([
      pubMediaFor_(env,publicidadId),
      pubSegmentacionFor_(env,publicidadId)
    ]);

    if(!Object.prototype.hasOwnProperty.call(data,"media")){
      validationPayload.media=existingMedia.map(m=>({
        url:pubText_(m.url),
        poster:pubText_(m.poster)
      }));
    }

    if(!Object.prototype.hasOwnProperty.call(data,"ciudades")){
      validationPayload.ciudades=[...new Set(
        existingSeg
          .map(s=>pubText_(s.ciudad_id))
          .filter(Boolean)
      )];
    }

    if(!Object.prototype.hasOwnProperty.call(data,"categorias")){
      validationPayload.categorias=[...new Set(
        existingSeg
          .map(s=>
            pubText_(s.categoria_key) ||
            (
              pubText_(s.ubicacion_id) &&
              pubText_(s.categoria_id)
                ? pubText_(s.ubicacion_id)+":"+pubText_(s.categoria_id)
                : ""
            )
          )
          .filter(Boolean)
      )];
    }
  }

  const validation=pubValidatePayload_(validationPayload,config);
  if(validation){
    return {success:false,message:validation};
  }

  const id=publicidadId || pubRandomId_("PUB");
  const now=pubNowIso_();

  const doc={
    ...(current||{}),
    nombre_interno:pubText_(merged.nombre_interno),
    titulo:pubText_(merged.titulo),
    formato:pubText_(merged.formato).toUpperCase(),
    cta_texto:pubText_(merged.cta_texto),
    cta_tipo:pubText_(merged.cta_tipo).toUpperCase(),
    cta_destino:pubText_(merged.cta_destino),
    publicidad_id:id,
    anunciante_id:aid,
    aprobado:current ? pubBool_(current.aprobado) : true,
    estado:current ? pubText_(current.estado).toUpperCase() : "INACTIVA",
    vigente_desde:current ? pubText_(current.vigente_desde) : "",
    vigente_hasta:current ? pubText_(current.vigente_hasta) : "",
    creado:current ? pubText_(current.creado) : now,
    actualizado:now
  };

  await firestoreSetDocument(
    env,
    COLL_PUBLICIDADES,
    id,
    doc
  );

  // Solo rehace media/segmentación si el payload las trae.
  if(
    Object.prototype.hasOwnProperty.call(data,"media") ||
    Object.prototype.hasOwnProperty.call(data,"ciudades") ||
    Object.prototype.hasOwnProperty.call(data,"categorias") ||
    !current
  ){
    const existingMedia=current ? await pubMediaFor_(env,id) : [];
    const existingSeg=current ? await pubSegmentacionFor_(env,id) : [];

    const normalizedPayload={
      ...merged,
      media:Object.prototype.hasOwnProperty.call(data,"media")
        ? data.media
        : existingMedia.map(m=>({
            url:pubText_(m.url),
            poster:pubText_(m.poster)
          })),
      ciudades:Object.prototype.hasOwnProperty.call(data,"ciudades")
        ? data.ciudades
        : [...new Set(existingSeg.map(s=>pubText_(s.ciudad_id)).filter(Boolean))],
      categorias:Object.prototype.hasOwnProperty.call(data,"categorias")
        ? data.categorias
        : [...new Set(existingSeg.map(s=>
            pubText_(s.categoria_key) ||
            (
              pubText_(s.ubicacion_id) &&
              pubText_(s.categoria_id)
                ? pubText_(s.ubicacion_id)+":"+pubText_(s.categoria_id)
                : ""
            )
          ).filter(Boolean))]
    };

    await pubSaveChildren_(
      env,
      id,
      normalizedPayload,
      config
    );
  }

  return {
    success:true,
    publicidad_id:id,
    created:!current,
    updated:!!current
  };
}

async function pubActualizarActivos_(env, advertiserId, ids){
  const aid=pubText_(advertiserId);
  const admin=await firestoreGetDocument(env,COLL_ADMIN,aid);

  if(!admin){
    return {success:false,message:"No existe la administración del anunciante."};
  }

  if(!pubFeatureEnabled_(admin)){
    return {success:false,message:"El anunciante no tiene PUBLICIDAD habilitada."};
  }

  const config=pubConfig_(admin);
  const selected=[...new Set(
    (Array.isArray(ids)?ids:[])
      .map(pubText_)
      .filter(Boolean)
  )];

  if(config.activas_max<=0){
    return {success:false,message:"sin_cupo_publicidades_activas"};
  }

  if(selected.length>config.activas_max){
    return {success:false,message:"supera_publicidades_activas_max"};
  }

  const propias=await firestoreQueryEqual(
    env,
    COLL_PUBLICIDADES,
    "anunciante_id",
    aid
  );

  const ownIds=new Set(
    propias.map(p=>pubText_(p.publicidad_id||p.id)).filter(Boolean)
  );

  if(selected.some(id=>!ownIds.has(id))){
    return {success:false,message:"publicidad_no_pertenece_al_anunciante"};
  }

  const actuales=propias
    .filter(p=>pubText_(p.estado).toUpperCase()==="ACTIVA")
    .map(p=>pubText_(p.publicidad_id||p.id))
    .sort();

  const next=[...selected].sort();

  if(JSON.stringify(actuales)===JSON.stringify(next)){
    return {
      success:true,
      sin_cambios:true
    };
  }

  const cambios=await pubCambiosHoy_(
    env,
    aid,
    config
  );

  if(
    cambios.max<=0 ||
    cambios.usados>=cambios.max
  ){
    return {
      success:false,
      message:"cambio_activos_diario_agotado"
    };
  }

  await Promise.all(
    propias.map(async p=>{
      const id=pubText_(p.publicidad_id||p.id);
      const shouldBeActive=selected.includes(id);

      if(
        (pubText_(p.estado).toUpperCase()==="ACTIVA")===
        shouldBeActive
      ){
        return;
      }

      p.estado=shouldBeActive ? "ACTIVA" : "INACTIVA";
      p.actualizado=pubNowIso_();

      await firestoreSetDocument(
        env,
        COLL_PUBLICIDADES,
        id,
        p
      );
    })
  );

  await firestoreSetDocument(
    env,
    COLL_PUBLICIDAD_CAMBIOS,
    cambios.doc_id,
    {
      cambio_id:cambios.doc_id,
      anunciante_id:aid,
      fecha:cambios.fecha,
      usados:cambios.usados+1,
      actualizado:pubNowIso_()
    }
  );

  return {
    success:true,
    sin_cambios:false
  };
}

async function pubEliminar_(env, advertiserId, publicidadId){
  const aid=pubText_(advertiserId);
  const id=pubText_(publicidadId);

  if(!id){
    return {success:false,message:"Falta publicidad_id."};
  }

  const admin=await firestoreGetDocument(env,COLL_ADMIN,aid);

  if(!admin || !pubFeatureEnabled_(admin)){
    return {success:false,message:"El anunciante no tiene PUBLICIDAD habilitada."};
  }

  const current=await firestoreGetDocument(
    env,
    COLL_PUBLICIDADES,
    id
  );

  if(
    !current ||
    pubText_(current.anunciante_id)!==aid
  ){
    return {success:false,message:"La publicidad no pertenece al anunciante."};
  }

  if(pubText_(current.estado).toUpperCase()==="ACTIVA"){
    return {success:false,message:"publicidad_activa_no_eliminable"};
  }

  await pubDeleteChildren_(env,id);
  await firestoreDeleteDocument(
    env,
    COLL_PUBLICIDADES,
    id
  );

  return {
    success:true,
    deleted:true,
    publicidad_id:id
  };
}

async function handlePublicidadFirestoreRequest(request, env){
  try{
    const url=new URL(request.url);

    if(request.method==="GET"){
      const action=pubText_(
        url.searchParams.get("action")
      ).toLowerCase();

      if(action==="publicas"){
        return jsonResponse(
          await pubPublicas_(env,url),
          200
        );
      }

      if(action==="getpaneldata"){
        const advertiserId=pubText_(
          url.searchParams.get("advertiserId") ||
          url.searchParams.get("advertiser_id") ||
          url.searchParams.get("id")
        );

        if(!advertiserId){
          return jsonResponse({
            success:false,
            message:"Falta advertiserId."
          },400);
        }

        return jsonResponse(
          await pubPanelData_(env,advertiserId),
          200
        );
      }

      return jsonResponse({
        success:false,
        message:"Acción GET de Publicidad no reconocida."
      },400);
    }

    if(request.method!=="POST"){
      return jsonResponse({
        success:false,
        message:"Método no permitido."
      },405);
    }

    let body={};
    try{
      body=await request.json();
    }catch(_){
      body={};
    }

    const action=pubText_(
      body.action || body.accion
    ).toLowerCase();

    const advertiserId=pubText_(
      body.advertiserId ||
      body.advertiser_id ||
      body.id
    );

    if(!advertiserId){
      return jsonResponse({
        success:false,
        message:"Falta advertiserId."
      },400);
    }

    if(action==="guardar"){
      return jsonResponse(
        await pubGuardar_(
          env,
          advertiserId,
          body.payload || {}
        ),
        200
      );
    }

    if(action==="actualizar_activos"){
      return jsonResponse(
        await pubActualizarActivos_(
          env,
          advertiserId,
          body.publicidad_ids
        ),
        200
      );
    }

    if(action==="eliminar"){
      return jsonResponse(
        await pubEliminar_(
          env,
          advertiserId,
          body.publicidad_id
        ),
        200
      );
    }

    if(action==="getpaneldata"){
      return jsonResponse(
        await pubPanelData_(env,advertiserId),
        200
      );
    }

    return jsonResponse({
      success:false,
      message:"Acción de Publicidad no reconocida."
    },400);

  }catch(err){
    return jsonResponse({
      success:false,
      message:String(
        err && err.message ? err.message : err
      )
    },500);
  }
}

function normalizedPath_(pathname){
  if(!pathname || pathname === "/") return "/";
  return pathname.endsWith("/")
    ? pathname.slice(0,-1)
    : pathname;
}

export default {
  async fetch(request, env){
    const url = new URL(request.url);
    const path = normalizedPath_(url.pathname);

    if(request.method === "OPTIONS"){
      return new Response(null, {
        status:204,
        headers:corsHeaders()
      });
    }

    if(path === "/"){
      return jsonResponse({
        ok:true,
        msg:"Worker activo"
      },200);
    }

    if(path === "/search" && request.method === "GET"){
      return await handleSearchRequest(request, env);
    }

    // Anunciantes / Commerce ya trabaja directo con Firestore.
    if(
      path === "/commerce" &&
      (request.method === "GET" || request.method === "POST")
    ){
      return await handleCommerceRequest(request, env);
    }

    // Efemérides ya trabaja directo con Firestore.
    if(
      path === "/efemerides" &&
      (request.method === "GET" || request.method === "POST")
    ){
      return await handleEfemeridesRequest(request, env);
    }

    // Suscriptores: Firestore directo.
    if(
      path === "/suscriptores" &&
      (request.method === "GET" || request.method === "POST")
    ){
      return await handleSuscriptoresRequest(request, env);
    }

    // Farmacias: público + panel administrativo directo en Firestore.
    if(
      path === "/farmacias" &&
      (request.method === "GET" || request.method === "POST")
    ){
      return await handleFarmaciasRequest(request, env);
    }

    // Actividades: público + panel administrativo directo en Firestore.
    if(
      path === "/actividades" &&
      (request.method === "GET" || request.method === "POST")
    ){
      return await handleActividadesFirestoreRequest(request, env);
    }

    // Publicidad: panel y lectura pública directo en Firestore.
    if(
      path === "/publicidad" &&
      (request.method === "GET" || request.method === "POST")
    ){
      return await handlePublicidadFirestoreRequest(request, env);
    }

    // Eventos: VIP, FREE, público y programación directo en Firestore.
    if(
      path === "/events-new" &&
      (request.method === "GET" || request.method === "POST")
    ){
      return await handleEventsFirestoreRequest(request, env);
    }

    // Promos: público + panel administrativo directo en Firestore.
    if(
      path === "/promos" &&
      (request.method === "GET" || request.method === "POST")
    ){
      return await handlePromosFirestoreRequest(request, env);
    }

    // Super Admin / Gran Hermano.
    if(
      path === "/superadmin" &&
      (request.method === "GET" || request.method === "POST")
    ){
      return await handleSuperadminRequest(request, env);
    }

    return new Response(
      "Not found",
      {
        status:404,
        headers:corsHeaders()
      }
    );
  }
};
